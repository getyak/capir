  await pool!.query("INSERT INTO users(id,account_id,email,display_name,kind) VALUES($1,$2,$3,'Synthetic owner','simulated_human')", [auth.userId, auth.accountId, auth.userEmail]);

    [auth.sessionId, auth.accountId, auth.userId, randomUUID()],
  );
  return auth;
}

/** A provider whose call can be held, released or made non-cancellable. */
function heldProvider(options: { cancellable?: boolean } = {}) {
  const started: Array<() => void> = [];
  let drainResult: RemoteChatAnswerResult | null = null;
  const pending: Array<{ resolve: (value: RemoteChatAnswerResult) => void; reject: (error: unknown) => void; aborted?: boolean }> = [];
  return {
    provider: {
      providerId: "zhipu-chat-completions",
      model: "synthetic",
      supportsImageInput: false,
      answer: async (request: { signal?: AbortSignal }) => {
        let arrived!: () => void;
        started.push(() => arrived());
        void new Promise<void>((resolve) => { arrived = resolve; });
        if (drainResult) return drainResult;
        if (options.cancellable !== false && request.signal) {
          const abort = () => {
            const entry = pending.find((item) => !item.aborted);
            if (entry) { entry.aborted = true; entry.reject(new Error("provider cancelled")); }
          };
          if (request.signal.aborted) abort();
          else request.signal.addEventListener("abort", abort, { once: true });
        }
        return await new Promise<RemoteChatAnswerResult>((resolve, reject) => {
          pending.push({ resolve, reject });
        });
      },
    } satisfies RemoteChatAnswerProviding,
    started,
    drain(value = answerResult("Synthetic cleanup settlement.")) {
      drainResult=value;
      for (const entry of pending.splice(0)) if (!entry.aborted) entry.resolve(value);
    },
    resolveAll(value = answerResult("Synthetic completed answer.")) {
      for (const entry of pending.splice(0)) if (!entry.aborted) entry.resolve(value);
    },
  };
}

  ] as const)("preserves first %s concurrent Memory probe failure through %s", async (adapter, late) => {
    const auth = await account("get9-a1-overlap"); const f = await wikiFixture(auth);
    const first = new Error("SYNTHETIC_A1_FIRST_CONCURRENT_PROBE_FAILURE");
    const second = new Error("SYNTHETIC_A1_SECOND_CONCURRENT_PROBE_FAILURE");
    let releaseRecall!: () => void, releaseFirst!: () => void, releaseSecond!: () => void;
    const recallGate = new Promise<void>((resolve) => { releaseRecall = resolve; });
    const firstGate = new Promise<void>((resolve) => { releaseFirst = resolve; });
    const secondGate = new Promise<void>((resolve) => { releaseSecond = resolve; });
    let recalled = 0, probing = 0, entered = 0;
    const observed: Array<{ value?: unknown; error?: unknown }> = [];
    const watched = a1Watch(async (sql, execute) => {
      if (sql.includes("FROM memory_items m")) {
        const result = await execute();
        recalled += 1;
        if (recalled === 2) releaseRecall();
        await recallGate;
        return result;
      }
      if (recalled === 2 && probing < 2 && sql.includes("SELECT a.auth_actor_kind")) {
        probing += 1;
        if (probing === 1) { await firstGate; throw first; }

        if (late === "late-failure") throw second;
      }

    });

    };
    const provider = adapter === "sdk-adapter"
