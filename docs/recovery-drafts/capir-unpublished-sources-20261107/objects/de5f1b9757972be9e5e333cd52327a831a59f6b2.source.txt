import XCTest
@testable import TalentSignal

/// Real production wire proof for the delta chain: the authenticated
/// `AgentSessionSyncClient` (with its stable RuntimeEndpoint/account/user
/// namespace and durable transfer store) drives raw GET overlay, planning and
/// exact-byte frame PATCH through a stubbed URLSession. Server-owned fields
/// the DTO cannot model survive into the declared intent and its digest.
final class AgentSessionSyncClientDeltaTests: XCTestCase {
    func testAuthenticatedFactoryKeepsExactFrameAcrossBearerRotation() throws {
        let baseURL = URL(string: "https://synthetic.invalid")!
        let accountID = UUID().uuidString, userID = UUID().uuidString
        let endpoint = RuntimeEndpoint.scope(baseURL)
        let first = AgentSessionSyncClient.authenticated(baseURL: baseURL, bearerToken: "first",
            accountID: accountID, userID: userID, session: .shared)
        let rotated = AgentSessionSyncClient.authenticated(baseURL: baseURL, bearerToken: "rotated",
            accountID: accountID, userID: userID, session: .shared)
        let store = try XCTUnwrap(first.transferStore)
        defer { try? store.releasePartition(accountID: accountID, userID: userID, endpointScope: endpoint) }
        let frame = AgentSessionDeltaInFlightFrame(sessionID: UUID(), transferID: UUID(), frameIndex: 1,
            idempotencyKey: UUID(), expectedRevision: 0, targetPayloadDigest: String(repeating: "a", count: 64),
            requestBody: Data("exact retained request".utf8), isFinal: false, createdAt: Date(),
            sessionExpiresAt: Date(timeIntervalSinceNow: 600))
        try store.retain(frame, accountID: accountID, userID: userID, endpointScope: endpoint)
        let rotatedStore = try XCTUnwrap(rotated.transferStore)
        XCTAssertEqual(try rotatedStore.inFlightFrame(accountID: accountID, userID: userID,
            endpointScope: endpoint, sessionID: frame.sessionID), frame)
    }

    private final class DeltaWireStub: URLProtocol, @unchecked Sendable {
        static var handler: ((URLRequest) throws -> (Int, Data))?
        static var capturedBodies: [Data] = []
        override class func canInit(with request: URLRequest) -> Bool { true }
        override class func canonicalRequest(for request: URLRequest) -> URLRequest { request }
        override func startLoading() {
            var request = self.request
            if request.httpBody == nil, let stream = request.httpBodyStream {
                stream.open(); defer { stream.close() }
                var body = Data(), buffer = [UInt8](repeating: 0, count: 4096)
                while stream.hasBytesAvailable {
                    let count = stream.read(&buffer, maxLength: buffer.count)
                    if count <= 0 { break }
                    body.append(contentsOf: buffer.prefix(count))
                }
                request.httpBodyStream = nil
                request.httpBody = body
            }
            do {
                guard let handler = Self.handler, let url = request.url else {
                    throw URLError(.badServerResponse)
                }
                if let body = request.httpBody { Self.capturedBodies.append(body) }
                let (status, data) = try handler(request)
                let response = HTTPURLResponse(
                    url: url, statusCode: status, httpVersion: "HTTP/1.1",
                    headerFields: ["Content-Type": "application/json"]
                )!
                client?.urlProtocol(self, didReceive: response, cacheStoragePolicy: .notAllowed)
                client?.urlProtocol(self, didLoad: data)
                client?.urlProtocolDidFinishLoading(self)
            } catch {
                client?.urlProtocol(self, didFailWithError: error)
            }
        }
        override func stopLoading() {}
    }

    private var client: AgentSessionSyncClient!
    private let sessionID = UUID(uuidString: "7BA9A4B2-8E31-4C0A-9D2E-55AA7C4E1B10")!

    override func setUp() {
        super.setUp()
        DeltaWireStub.capturedBodies = []
        let configuration = URLSessionConfiguration.ephemeral
        configuration.protocolClasses = [DeltaWireStub.self]
        client = AgentSessionSyncClient(
            baseURL: URL(string: "https://synthetic.invalid")!,
            bearerToken: "synthetic-token-never-persisted",
            deltaNamespace: (
                endpointScope: "https://synthetic.invalid",
                accountID: "account-a",
                userID: "user-a"
            ),
            transferStore: EphemeralAgentSessionDeltaTransferStore(),
            session: URLSession(configuration: configuration)
        )
    }

    override func tearDown() {
        DeltaWireStub.handler = nil
        DeltaWireStub.capturedBodies = []
        super.tearDown()
    }

    private func rawRecordJSON(payload: [String: Any]?, deleted: Bool = false) -> Data {
        var session: [String: Any] = [
            "session_id": sessionID.uuidString.lowercased(),
            "revision": 3,
            "updated_at": "2026-10-08T03:07:24.000Z",
            "expires_at": "2099-01-01T00:00:00.000Z",
            "deleted_at": deleted ? "2026-10-08T03:07:24.000Z" : NSNull(),
            "payload": payload ?? NSNull(),
            "display_authority": "stale_unconfirmed",
        ]
        session["sync"] = [
            "state": "complete",
            "transfer_id": NSNull(),
            "target_payload_digest": NSNull(),
            "started_at": NSNull(),
            "frame_count": 0,
        ] as [String: Any]
        let envelope: [String: Any] = [
            "contract_version": TalentSignalAPIContract.version,
            "session": session,
        ]
        return try! JSONSerialization.data(withJSONObject: envelope)
    }

    private func frameResponseJSON(complete: Bool) -> Data {
        let envelope: [String: Any] = [
            "contract_version": TalentSignalAPIContract.version,
            "session": [
                "session_id": sessionID.uuidString.lowercased(),
                "revision": 4,
                "updated_at": "2026-10-08T03:07:25.000Z",
                "expires_at": "2099-01-01T00:00:00.000Z",
                "deleted_at": NSNull(),
                "payload": NSNull(),
                "display_authority": "stale_unconfirmed",
                "sync": [
                    "state": complete ? "complete" : "incomplete",
                    "transfer_id": complete ? NSNull() : UUID().uuidString.lowercased(),
                    "target_payload_digest": complete ? NSNull() : String(repeating: "0", count: 64),
                    "started_at": NSNull(),
                    "frame_count": 1,
                ] as [String: Any],
            ] as [String: Any],
            "transfer": [
                "transfer_id": UUID().uuidString.lowercased(),
                "frame_index": 1,
                "replayed": false,
                "complete": complete,
            ] as [String: Any],
        ]
        return try! JSONSerialization.data(withJSONObject: envelope)
    }

    func testRawSessionKeepsUnknownServerFieldsAndReportsAbsentRecords() async throws {
        let payload: [String: Any] = [
            "id": sessionID.uuidString.lowercased(),
            "scopeKind": "unresolved_intent",
            "personDisplayLabel": "New session",
            "contextDisplayLabel": "Conversation",
            "title": "Raw overlay",
            "updatedAt": "2026-10-08T03:07:24.000Z",
            "isUnread": false,
            "turns": [[String: Any]](),
            "probe_unknown_root_field": ["kept": true],
        ]
        DeltaWireStub.handler = { _ in (200, self.rawRecordJSON(payload: payload)) }
        let fetchedRaw = try await client.rawSession(id: sessionID)
        let raw = try XCTUnwrap(fetchedRaw)
        XCTAssertEqual(raw.record.sessionID, sessionID)
        XCTAssertNotNil((raw.raw["payload"] as? [String: Any])?["probe_unknown_root_field"])

        DeltaWireStub.handler = { _ in (404, try JSONSerialization.data(withJSONObject: [
            "error": ["code": "AGENT_SESSION_NOT_FOUND", "message": "gone", "request_id": "t"],
        ])) }
        let missing = try await client.rawSession(id: sessionID)
        XCTAssertNil(missing, "an absent record is first creation, not an error")
    }

    func testDeltaFrameDispatchesExactBytesAndDecodesSyncState() async throws {
        var seenBody: Data?
        DeltaWireStub.handler = { request in
            seenBody = request.httpBody
            return (200, self.frameResponseJSON(complete: true))
        }
        let bytes = Data(#"{"probe":"exact"}"#.utf8)
        let response = try await client.sendDeltaFrame(id: sessionID, requestBody: bytes)
        XCTAssertEqual(seenBody, bytes, "the exact serialized frame intent hits the wire")
        XCTAssertTrue(response.transfer.complete)
        XCTAssertEqual(response.session.sync?.state, "complete")
    }

    func testProductionChainOverlaysRawServerFieldsBeforePlanAndHash() async throws {
        let turnID = "0D90B077-2BD1-4C13-82CA-8F4DD366DB1B"
        let rawTurn: [String: Any] = [
            "id": turnID,
            "objective": "original message",
            "createdAt": "2026-10-08T03:07:24.123Z",
            "steeredMessages": [["id": UUID().uuidString.lowercased(), "objective": "steered"]] as [Any],
            "images": [["id": UUID().uuidString.lowercased(), "file_name": "shot.png"]] as [Any],
            "response": [
                "contractVersion": TalentSignalAPIContract.version,
                "taskID": UUID().uuidString.lowercased(),
                "contextManifestID": "none-unbound-conversation",
                "knowledgeSnapshotID": "none-unbound-conversation",
                "disposition": "answered",
                "createdAt": "2026-10-08T03:07:24.200Z",
                "execution": ["tools": [["name": "search", "completed_at": "2026-10-08T03:07:25.000Z"]]] as [String: Any],
                "probe_unknown_response_field": ["kept": true],
            ] as [String: Any],
        ]
        let rawPayload: [String: Any] = [
            "id": sessionID.uuidString.lowercased(),
            "scopeKind": "unresolved_intent",
            "personDisplayLabel": "New session",
            "contextDisplayLabel": "Conversation",
            "title": "Old title",
            "updatedAt": "2026-10-08T03:07:24.000Z",
            "isUnread": false,
            "turns": [rawTurn],
        ]
        // DTO-shaped desired value: no execution/steered/images at all.
        var desired: [String: Any] = rawPayload
        desired["title"] = "New title"
        var turns = desired["turns"] as! [[String: Any]]
        turns[0] = [
            "id": turnID,
            "objective": "original message",
            "createdAt": "2026-10-08T03:07:24.123Z",
            "response": [
                "contractVersion": TalentSignalAPIContract.version,
                "taskID": (rawTurn["response"] as! [String: Any])["taskID"]!,
                "contextManifestID": "none-unbound-conversation",
                "knowledgeSnapshotID": "none-unbound-conversation",
                "disposition": "answered",
                "createdAt": "2026-10-08T03:07:24.200Z",
            ] as [String: Any],
        ]
        turns.append([
            "id": UUID().uuidString.lowercased(),
            "objective": "second message",
            "createdAt": "2026-10-08T03:07:26.000Z",
            "response": [
                "contractVersion": TalentSignalAPIContract.version,
                "taskID": UUID().uuidString.lowercased(),
                "contextManifestID": "none-unbound-conversation",
                "knowledgeSnapshotID": "none-unbound-conversation",
                "disposition": "answered",
                "createdAt": "2026-10-08T03:07:26.100Z",
            ] as [String: Any],
        ])
        desired["turns"] = turns

        DeltaWireStub.handler = { request in
            if request.httpMethod == "GET" {
                return (200, self.rawRecordJSON(payload: rawPayload))
            }
            return (200, self.frameResponseJSON(complete: true))
        }
        let fetchedDelta = try await client.deltaServing()
        let delta = try XCTUnwrap(fetchedDelta)
        let progress = try await delta.synchronizeSessionDelta(
            sessionID: sessionID, desired: desired, skeleton: nil, isCurrent: { true }
        )
        XCTAssertTrue(progress.synchronized)

        let dispatched = try XCTUnwrap(DeltaWireStub.capturedBodies.last)
        let request = try XCTUnwrap(
            try JSONSerialization.jsonObject(with: dispatched) as? [String: Any]
        )
        // The declared target digest covers the raw server-owned fields the
        // DTO lost: execution, folded steering and image manifests survive.
        let merged = AgentSessionDeltaClient.mergeDesiredOverRaw(
            rawPayload: rawPayload, desired: desired
        )
        XCTAssertEqual(
            request["target_payload_digest"] as? String,
            try AgentSessionDeltaCanonical.sessionIntentDigest(merged)
        )
        XCTAssertNotEqual(
            request["target_payload_digest"] as? String,
            try AgentSessionDeltaCanonical.sessionIntentDigest(desired),
            "a DTO-only digest would silently drop server-owned provenance"
        )
        // The emitted frame never rewrites server-owned provenance.
        for operation in try XCTUnwrap(request["operations"] as? [[String: Any]]) {
            let path = operation["path"] as? String ?? ""
            XCTAssertFalse(path.contains("execution"))
            XCTAssertFalse(path.contains("steeredMessages"))
            XCTAssertFalse(path.contains("images"))
        }
    }

    func testProducedIncompleteSyncDecodesAndKeepsTransferIdentity() async throws {
        // The produced wire state is decoded faithfully: an incomplete record
        // keeps its transfer identity and never counts as complete.
        let incompleteJSON = try JSONSerialization.data(withJSONObject: [
            "contract_version": TalentSignalAPIContract.version,
            "session": [
                "session_id": sessionID.uuidString.lowercased(),
                "revision": 5,
                "updated_at": "2026-10-08T03:07:24.000Z",
                "expires_at": "2099-01-01T00:00:00.000Z",
                "deleted_at": NSNull(),
                "payload": NSNull(),
                "display_authority": "stale_unconfirmed",
                "sync": [
                    "state": "incomplete",
                    "transfer_id": "8BA9A4B2-8E31-4C0A-9D2E-55AA7C4E1B10",
                    "target_payload_digest": String(repeating: "a", count: 64),
                    "started_at": "2026-10-08T03:07:20.000Z",
                    "frame_count": 7,
                ] as [String: Any],
            ] as [String: Any],
        ])
        struct ProducedEnvelope: Decodable {
            let contractVersion: String
            let session: AgentSessionRemoteRecord
            enum CodingKeys: String, CodingKey {
                case contractVersion = "contract_version", session
            }
        }
        let decoded = try JSONDecoder.agentSession.decode(
            ProducedEnvelope.self, from: incompleteJSON
        )
        XCTAssertEqual(decoded.session.sync?.state, "incomplete")
        XCTAssertTrue(decoded.session.isRemotelyIncomplete)
        XCTAssertEqual(decoded.session.sync?.frameCount, 7)
        XCTAssertEqual(
            decoded.session.sync?.transferID?.uuidString.lowercased(),
            "8ba9a4b2-8e31-4c0a-9d2e-55aa7c4e1b10"
        )
    }

    @MainActor
    func testFirstCreateAcceptsRealStoreDTOTimestampsAndKeepsOriginalDeadline() async throws {
        // A real archive DTO always emits fractional-second timestamps; the
        // first creation deadline must parse them and preserve the ORIGINAL
        // Session deadline in the retained frame.
        let archive = AgentSessionStore(persistence: nil)
        let localID = try XCTUnwrap(archive.beginUnscopedSession(objective: "First create"))
        let dto = PersistedAgentSession(try XCTUnwrap(archive.session(id: localID)))
        var desired = try JSONSerialization.jsonObject(
            with: try JSONEncoder.agentSession.encode(dto)
        ) as! [String: Any]
        desired["id"] = sessionID.uuidString.lowercased()
        XCTAssertTrue((desired["createdAt"] as? String ?? "").contains("."), "real DTO timestamps carry fractional seconds")

        let storeDirectory = FileManager.default.temporaryDirectory
            .appending(path: "delta-wire-\(UUID().uuidString)", directoryHint: .isDirectory)
        defer { try? FileManager.default.removeItem(at: storeDirectory) }
        let store = AgentSessionDeltaTransferStore(directory: storeDirectory)

        var patchedBody: Data?
        DeltaWireStub.handler = { request in
            if request.httpMethod == "GET" {
                return (404, try JSONSerialization.data(withJSONObject: [
                    "error": ["code": "AGENT_SESSION_NOT_FOUND", "message": "gone", "request_id": "t"],
                ]))
            }
            patchedBody = request.httpBody
            // The frame is intentionally left uncertain so the retained intent
            // can be inspected against its ORIGINAL deadline.
            return (500, try JSONSerialization.data(withJSONObject: [
                "error": ["code": "UNAVAILABLE", "message": "later", "request_id": "t"],
            ]))
        }
        let delta = AgentSessionDeltaClient(
            transport: client, store: store,
            accountID: "account-a", userID: "user-a", endpointScope: "https://synthetic.invalid"
        )
        do {
            _ = try await delta.synchronize(
                sessionID: sessionID, desired: desired,
                skeleton: AgentSessionDeltaClient.emptySkeleton(from: desired)
            )
            XCTFail("the failed dispatch must surface as uncertainty")
        } catch {}
        let request = try XCTUnwrap(
            try JSONSerialization.jsonObject(with: XCTUnwrap(patchedBody)) as? [String: Any]
        )
        XCTAssertNotNil(request["skeleton"], "first creation carries the canonical skeleton")
        let retained = try XCTUnwrap(
            store.inFlightFrame(
                accountID: "account-a", userID: "user-a",
                endpointScope: "https://synthetic.invalid", sessionID: sessionID
            )
        )
        let created = try XCTUnwrap(AgentSessionDeltaClient.parseTimestamp(desired["createdAt"] as! String))
        let expectedDeadline = AgentSessionDeltaTransferStore.originalDeadline(createdAt: created)
        XCTAssertEqual(
            retained.sessionExpiresAt?.timeIntervalSince1970 ?? 0,
            expectedDeadline.timeIntervalSince1970,
            accuracy: 2
        )
    }

    @MainActor
    func testStoreProducedUppercaseDTOOverLowercaseServerPayloadKeepsProvenance() async throws {
        // Real Store-produced DTO encodes UPPERCASE UUID identifiers while the
        // backend canonical payload is lowercase: identity comparison must not
        // be case-sensitive or known turns get replaced by the lossy DTO.
        let archive = AgentSessionStore(persistence: nil)
        let localID = try XCTUnwrap(archive.beginUnscopedSession(objective: "Case identity"))
        let response = RelationshipAskResponse(contractVersion: TalentSignalAPIContract.version,
            taskID: "none-synthetic", contextManifestID: "none-unbound-conversation", knowledgeSnapshotID: "none-unbound-conversation",
            disposition: "answered", blocks: [], media: [], createdAt: ISO8601DateFormatter().string(from: Date()), citations: [])
        XCTAssertTrue(archive.recordUnscopedChat(sessionID: localID, objective: "Case identity", response: response))
        let dto = PersistedAgentSession(try XCTUnwrap(archive.session(id: localID)))
        var desired = try JSONSerialization.jsonObject(
            with: try JSONEncoder.agentSession.encode(dto)
        ) as! [String: Any]
        desired["id"] = sessionID.uuidString.lowercased()
        desired["title"] = "New title from the local archive"
        let uppercaseTurnID = ((desired["turns"] as? [[String: Any]])?.first?["id"] as? String) ?? ""
        XCTAssertEqual(uppercaseTurnID, uppercaseTurnID.uppercased())

        // Lowercase server canonical payload with server-owned fields.
        var raw = AgentSessionDeltaCanonical.canonicalizingIdentityStrings(desired) as! [String: Any]
        raw["title"] = "Old title"
        raw["turns"] = ((raw["turns"] as? [[String: Any]]) ?? []).map { turn -> [String: Any] in
            var turn = turn
            turn["steeredMessages"] = [["id": "10000000-0000-4000-8000-000000000001", "objective": "steered"]] as [Any]
            turn["images"] = [["id": "30000000-0000-4000-8000-000000000001", "file_name": "shot.png"]] as [Any]
            var response = turn["response"] as? [String: Any] ?? [:]
            response["execution"] = [
                "tools": [["name": "search", "completed_at": "2026-10-08T03:07:25.000Z"]],
            ] as [String: Any]
            response["probe_unknown_response_field"] = ["kept": true]
            turn["response"] = response
            return turn
        }

        DeltaWireStub.handler = { request in
            if request.httpMethod == "GET" {
                return (200, self.rawRecordJSON(payload: raw))
            }
            return (200, self.frameResponseJSON(complete: true))
        }
        let merged = AgentSessionDeltaClient.mergeDesiredOverRaw(rawPayload: raw, desired: desired)
        let mergedTurn = (merged["turns"] as? [[String: Any]])?.first
        XCTAssertNotNil(mergedTurn?["steeredMessages"], "known turns keep their raw provenance")
        XCTAssertNotNil(mergedTurn?["images"])
        XCTAssertNotNil((mergedTurn?["response"] as? [String: Any])?["execution"])

        let fetchedDelta = try await client.deltaServing()
        let delta = try XCTUnwrap(fetchedDelta)
        let progress = try await delta.synchronizeSessionDelta(
            sessionID: sessionID, desired: desired, skeleton: nil, isCurrent: { true }
        )
        XCTAssertTrue(progress.synchronized)
        let dispatched = try XCTUnwrap(DeltaWireStub.capturedBodies.last)
        let request = try XCTUnwrap(
            try JSONSerialization.jsonObject(with: dispatched) as? [String: Any]
        )
        XCTAssertEqual(
            request["target_payload_digest"] as? String,
            try AgentSessionDeltaCanonical.sessionIntentDigest(merged),
            "the declared intent covers the raw provenance across identifier casing"
        )
    }
