function pty(env,actions,args=[],options={}){return new Promise((resolve,reject)=>{const child=spawn('python3',[fileURLToPath(new URL('./model-pty.py',import.meta.url))],{env,stdio:['pipe','pipe','pipe']});let out='',err='';child.stdout.on('data',x=>out+=x);child.stderr.on('data',x=>err+=x);child.once('error',reject);child.once('close',code=>code!==0?reject(new Error(err)):resolve(JSON.parse(out)));child.stdin.end(JSON.stringify({node:process.execPath,binary,actions,args,...options}));});}

async function blockedWriterProbe(f,fd=1){
  const mark=join(f.root,'pending-owned-write'),closed=join(f.root,'owned-writer-closed'),loader=join(f.root,'pending-write-probe.mjs');
  await writeFile(loader,`import fs from 'node:fs';
import cp from 'node:child_process';
import {syncBuiltinESMExports} from 'node:module';
const original=cp.spawn;const marker=${JSON.stringify(mark)},closed=${JSON.stringify(closed)};
cp.spawn=(command,args,options)=>{
 const child=original(command,args,options);
 if(!args?.[0]?.endsWith('/ttyWriter.js')||options?.stdio?.[3]!==${fd})return child;
 let ack='',timer,pending;
 child.stdout.on('data',chunk=>{
  ack+=chunk.toString('ascii');
  while(ack.includes('\\n')){
   const end=ack.indexOf('\\n'),line=ack.slice(0,end);ack=ack.slice(end+1);
   const [id,state]=line.split(':');
   if(state==='start'){pending=id;timer=setTimeout(()=>{if(pending===id)fs.writeFileSync(marker,'actual TTY completion ACK pending after50ms\\n');},50);}
   if(state==='done'&&pending===id){pending=undefined;clearTimeout(timer);fs.rmSync(marker,{force:true});}
  }
 });
 child.once('close',(code,signal)=>{clearTimeout(timer);fs.writeFileSync(closed,JSON.stringify({pid:child.pid,code,signal}));});
 return child;
};syncBuiltinESMExports();\n`);
  return {mark,closed,env:{...f.env,NODE_OPTIONS:[f.env.NODE_OPTIONS,'--import='+new URL('file://'+loader).href].filter(Boolean).join(' ')}};
}
async function assertPhysicalWriterClosed(probe){
  const receipt=JSON.parse(await readFile(probe.closed,'utf8'));
  assert.equal(receipt.signal,'SIGKILL');assert.ok(receipt.pid>0);
  assert.throws(()=>process.kill(receipt.pid,0),error=>error.code==='ESRCH');
}
