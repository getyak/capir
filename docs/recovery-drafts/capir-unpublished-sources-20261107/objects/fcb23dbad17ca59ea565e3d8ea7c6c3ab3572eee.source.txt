function fakeAppleVerifier(): AppleTokenVerifying {
  return {
    async verify(identityToken: string): Promise<AppleIdentityToken> {
      const [subject, email, verified, nonceHash] = identityToken.split("|");
      if (!subject) throw new Error("fixture token missing subject");
      return {
        audience: "com.talentsignal.app",
        email: email && email !== "-" ? email : null,
        emailVerified: verified === "true",
        expiresAt: new Date(Date.now() + 3_600_000),
        issuer: "https://appleid.apple.com",
        nonce: nonceHash ?? "",
        subject,
      };
    },
  };
}

const fakeGoogleVerifier: GoogleVerifier = async (identityToken) => {
  const [subject, email, nonce] = identityToken.split("|");
  if (!subject || !email) throw new Error("fixture token missing claims");
  return {
    subject,
    email,
    name: "Fixture User",
    nonce: nonce ?? "",
    expiresAt: new Date(Date.now() + 3_600_000),
  };
};

const deps: CredentialChangeDeps = {
  appleVerifier: fakeAppleVerifier(),
  googleVerify: fakeGoogleVerifier,
};

function googleToken(subject: string, email: string, nonce: string) {
  return `${subject}|${email}|${nonce}`;
}

function appleToken(subject: string, nonce: string, email = "-", verified = "false") {
  return `${subject}|${email}|${verified}|${nonce}`;
}

const newVerifier = () => randomBytes(32).toString("base64url");
const newState = () => randomBytes(32).toString("base64url");

type Attempt = { attempt_id: string; pairing_secret: string; pending_path: string; expires_at: string };

let app: FastifyInstance;

beforeAll(async () => {
  if (!pool) return;
  app = Fastify();
  app.setErrorHandler((error, request, reply) => {
    const requestId = String((request as { id?: string }).id ?? "test-request");
    if (error instanceof ApiError) {
      void reply.status(error.statusCode).send({
        error: {
          code: error.code,
          message: error.message,
          request_id: requestId,
          details: error.details ?? null,
        },
      });
      return;
    }
    const validation = (error as { validation?: unknown }).validation;
    if (validation) {
      void reply.status(400).send({
        error: {
          code: "REQUEST_VALIDATION_FAILED",
          message: "The request does not match the versioned contract.",
          request_id: requestId,
          details: validation,
        },
      });
      return;
    }
    void reply.status((error as { statusCode?: number }).statusCode ?? 500).send({
      error: {
        code: (error as { code?: string }).code ?? "INTERNAL",
        message: (error as Error).message,
        request_id: requestId,
        details: null,
      },
    });
  });
  registerDesktopAuth(app, pool, config, deps);
  await app.ready();
});

afterAll(async () => {
  await app?.close();
  await pool?.end();
});

function inject(
  method: "GET" | "POST",
  url: string,
  options: {
    payload?: Record<string, unknown> | undefined;
    token?: string | undefined;
    origin?: string | null | undefined;
    destination?: string | null | undefined;
  } = {},
) {
  return app.inject({
    method,
    url,
    headers: {
      ...(options.origin === null ? {} : { origin: options.origin ?? ORIGIN }),
      ...(options.destination === null ? {} : { "sec-fetch-dest": options.destination ?? "document" }),
      ...(options.token ? { authorization: `Bearer ${options.token}` } : {}),
    },
    ...(options.payload ? { payload: options.payload } : {}),
  });
}

async function prepare(input: Record<string, unknown> & { verifier: string; state: string; token?: string | undefined }) {
  const { verifier, state, token, ...fields } = input;
  const response = await inject("POST", "/v1/desktop-auth/prepare", {
    token,
    payload: {
      client_label: "talent-signal-macos-test",
      web_origin: ORIGIN,
      code_challenge: desktopCodeChallenge(verifier),
      state,
      ...fields,
    },
  });
  expect(response.statusCode, response.body).toBe(200);
  return response.json() as Attempt;
}

async function authorize(attempt: Attempt, state: string, token?: string | undefined) {
  const response = await inject("POST", "/v1/desktop-auth/authorize", {
    token,
    payload: { attempt_id: attempt.attempt_id, state },
  });
  expect(response.statusCode, response.body).toBe(200);
  return response.json() as {
    provider_challenge: { challenge_id: string; nonce: string; expires_at: string };
    oidc_nonce: string;
    role: string;
    purpose: string;
  };
}

async function approve(input: {
  attempt: Attempt;
  state: string;
  challengeId: string;
  identityToken: string;
  token?: string | undefined;
}) {
  return inject("POST", "/v1/desktop-auth/approve", {
    token: input.token,
    payload: {
      attempt_id: input.attempt.attempt_id,
      state: input.state,
      challenge_id: input.challengeId,
      identity_token: input.identityToken,
    },
  });
}

/** Full system-browser half of a fixture provider round. */
async function providerRound(input: {
  attempt: Attempt;
  state: string;
  provider?: "google" | "apple" | undefined;
  subject: string;
  email: string;
  emailVerified?: boolean | undefined;
  token?: string | undefined;
}) {
  const provider = input.provider ?? "google";
  const round = await authorize(input.attempt, input.state, input.token);
  const identityToken =
    provider === "google"
      ? googleToken(input.subject, input.email, round.oidc_nonce)
      : appleToken(input.subject, round.oidc_nonce, input.email, String(input.emailVerified ?? true));
  const response = await approve({
    attempt: input.attempt,
    state: input.state,
    challengeId: round.provider_challenge.challenge_id,
    identityToken,
    token: input.token,
  });
  expect(response.statusCode, response.body).toBe(200);
  return round;
}

async function codeFor(attempt: Attempt, state: string, token?: string | undefined) {
  const response = await inject("POST", "/v1/desktop-auth/complete", {
    token,
    payload: { attempt_id: attempt.attempt_id, state },
  });
  expect(response.statusCode, response.body).toBe(200);
  return (response.json() as { code: string }).code;
}

async function consume(input: {
  attempt: Attempt;
  verifier: string;
  code: string;
  pairingSecret?: string | undefined;
  token?: string | undefined;
  attemptSecret?: string | undefined;
}) {
  return inject("POST", "/v1/desktop-auth/consume", {
    token: input.token,
    payload: {
      attempt_id: input.attempt.attempt_id,
      code: input.code,
      verifier: input.verifier,
      pairing_secret: input.pairingSecret ?? input.attempt.pairing_secret,
      ...(input.attemptSecret ? { attempt_secret: input.attemptSecret } : {}),
    },
  });
}

/** Run a complete relayed round and return its consume response. */
async function runRound(input: {
  provider?: "google" | "apple";
  subject: string;
  email: string;
  emailVerified?: boolean;
  prepare: Record<string, unknown>;
  verifier?: string;
  state?: string;
  token?: string | undefined;
  attemptSecret?: string | undefined;
}) {
  const verifier = input.verifier ?? newVerifier();
  const state = input.state ?? newState();
  const attempt = await prepare({ ...input.prepare, verifier, state, token: input.token });
  await providerRound({
    attempt,
    state,
    provider: input.provider ?? "google",
    subject: input.subject,
    email: input.email,
    emailVerified: input.emailVerified,
    token: input.token,
  });
  const code = await codeFor(attempt, state, input.token);
  const response = await consume({
    attempt,
    verifier,
    code,
    token: input.token,
    attemptSecret: input.attemptSecret,
  });
  return { attempt, verifier, state, response, code };
}

async function createAccount(options: {
  email: string;
  displayName?: string;
  kind?: "password_human" | "google_human" | "apple_human";
  password?: string;
}) {
  const client = await pool!.connect();
  const accountId = randomUUID();
  const userId = randomUUID();
  try {
    await client.query("BEGIN");
    await createRealIdentity(client, {
      accountId,
      userId,
      accountName: `${options.displayName ?? "Fixture"}'s workspace`,
      accountSlug: newAccountSlug(),
      email: options.email,
      displayName: options.displayName ?? "Fixture",
      kind: options.kind ?? "password_human",
      emailVerifiedAt: new Date(),
    });
    if (options.password) {
      await client.query(
        `INSERT INTO password_credentials(account_id, user_id, password_scrypt) VALUES ($1, $2, $3)`,
        [accountId, userId, await encodePasswordCredential(options.password)],
      );
    }
    await client.query("COMMIT");
  } catch (error) {
    await client.query("ROLLBACK");
    throw error;
  } finally {
    client.release();
  }
  return { accountId, userId };
}

async function linkIdentity(accountId: string, userId: string, provider: "google" | "apple", subject: string, email: string) {
  await pool!.query(
    `INSERT INTO auth_identities(id, account_id, user_id, provider, subject_hash, email_hint)
     VALUES ($1, $2, $3, $4, $5, $6)`,
    [
      randomUUID(),
      accountId,
      userId,
      provider,
      providerSubjectHash(provider, provider === "google" ? "https://accounts.google.com" : "https://appleid.apple.com", subject),
      email,
    ],
  );
}

async function sessionTokenFor(
  accountId: string,
  userId: string,
  clientLabel: string,
): Promise<{ token: string; sessionId: string }> {
  const client = await pool!.connect();
  try {
    await client.query("BEGIN");
    const row = (
      await pool!.query<{
        display_name: string;
        email: string;
        kind: string;
        username: string | null;
        account_name: string;
        account_slug: string;
        account_role: "admin" | "member";
      }>(
        `SELECT users.display_name, users.email, users.kind, users.username,
                accounts.name AS account_name, accounts.slug AS account_slug,
                users.account_role
         FROM users JOIN accounts ON accounts.id = users.account_id
         WHERE accounts.id = $1 AND users.id = $2`,
        [accountId, userId],
      )
    ).rows[0]!;
    const session = await insertSession(
      client,
      config,
      {
        accountId,
        accountName: row.account_name,
        accountSlug: row.account_slug,
        displayName: row.display_name,
        role: row.account_role,
        userEmail: row.email,
        userId,
        userKind: row.kind as "password_human",
        username: row.username,
      },
      clientLabel,
    );
    await client.query("COMMIT");
    const sessionId = (
      await pool!.query<{ id: string }>(
        `SELECT id FROM sessions WHERE account_id = $1 AND user_id = $2
         ORDER BY created_at DESC LIMIT 1`,
        [accountId, userId],
      )
    ).rows[0]!.id;
    return { token: session.access_token, sessionId };
  } catch (error) {
    await client.query("ROLLBACK");
    throw error;
  } finally {
    client.release();
  }
}

/** The phase-one Server Action side: a fresh reauth challenge for the frozen operation. */
async function reauthChallenge(provider: "google" | "apple"): Promise<string> {
  const client = await pool!.connect();
  try {
    const created =
      provider === "google"
        ? await createGoogleChallenge(client, config, "talent-signal-web")
        : await createAppleLoginChallenge(client, config, { client_label: "talent-signal-web" });
    return created.challenge_id;
  } finally {
    client.release();
  }
}

async function currentScope(accountId: string, userId: string) {
  const rows = await pool!.query<{ settings_revision: number; profile_revision: number }>(
    `SELECT accounts.settings_revision, users.profile_revision
     FROM accounts JOIN users ON users.account_id = accounts.id
     WHERE accounts.id = $1 AND users.id = $2`,
    [accountId, userId],
  );
  return {
    expected_account_id: accountId,
    expected_user_id: userId,
    expected_account_revision: Number(rows.rows[0]!.settings_revision),
    expected_user_revision: Number(rows.rows[0]!.profile_revision),
  };
}

function roundPrepare(input: {
  scope: Record<string, unknown>;
  intent: "link_provider" | "unlink_provider" | "set_password" | "change_password";
  targetProvider?: "apple" | "google" | "password";
  challengeId?: string;
  credentialAttemptId?: string;
  role?: "current" | "target";
  provider?: "google" | "apple";
  flowRef?: string;
  roundRef?: string;
  flowDeadlineAt?: string;
  originalProofAt?: string;
}) {
  const now = Date.now();
  return {
    provider: input.provider ?? "google",
    purpose: "credential_round",
    role: input.role ?? "current",
    intent: input.intent,
    ...(input.targetProvider ? { target_provider: input.targetProvider } : {}),
    flow_ref: input.flowRef ?? `flow-${randomUUID()}`,
    round_ref: input.roundRef ?? `round-${randomUUID()}`,
    flow_deadline_at: input.flowDeadlineAt ?? new Date(now + 590_000).toISOString(),
    original_proof_at: input.originalProofAt ?? new Date(now - 10_000).toISOString(),
    ...(input.challengeId ? { provider_challenge_id: input.challengeId } : {}),
    ...(input.credentialAttemptId ? { credential_attempt_id: input.credentialAttemptId } : {}),
    ...input.scope,
  };
}

const describeDB = database ? describe : describe.skip;

describeDB("desktop login handoff endpoints (real PostgreSQL)", () => {
  it("logs in an existing provider identity with an independent device session", async () => {
    const subject = `google-sub-existing-${RUN}`;
    const account = await createAccount({ email: `existing-${RUN}@example.test`, kind: "google_human" });
    await linkIdentity(account.accountId, account.userId, "google", subject, `existing-${RUN}@example.test`);
    await sessionTokenFor(account.accountId, account.userId, "pre-existing");
    const verifier = newVerifier();
    const state = newState();
    const attempt = await prepare({
      provider: "google",
      purpose: "login",
      role: "login",
      verifier,
      state,
    });
    await providerRound({
      attempt,
      state,
      provider: "google",
      subject,
      email: `existing-${RUN}@example.test`,
    });
    const code = await codeFor(attempt, state);
    const consumed = await consume({ attempt, verifier, code });
    expect(consumed.statusCode, consumed.body).toBe(200);
    const result = consumed.json() as {
      status: string;
      session: { account: { id: string }; user: { id: string } };
    };
    expect(result.status).toBe("logged_in");
    expect(result.session.account.id).toBe(account.accountId);
    expect(result.session.user.id).toBe(account.userId);
    const sessions = await pool!.query<{ count: string }>(
      `SELECT count(*)::text AS count FROM sessions WHERE account_id = $1 AND user_id = $2 AND revoked_at IS NULL`,
      [account.accountId, account.userId],
    );
    expect(Number(sessions.rows[0]!.count)).toBeGreaterThanOrEqual(2);
  });

  it("never promotes an unverified provider hint into verified ownership (P1-2)", async () => {
    const subject = `apple-sub-unverified-${RUN}`;
    const account = await createAccount({ email: `unverified-owner-${RUN}@example.test`, kind: "apple_human" });
    await linkIdentity(account.accountId, account.userId, "apple", subject, `unverified-owner-${RUN}@example.test`);
    const verifier = newVerifier();
    const state = newState();
    const attempt = await prepare({ provider: "apple", purpose: "login", role: "login", verifier, state });
    await providerRound({
      attempt,
      state,
      provider: "apple",
      subject,
      email: `unverified-alias-${RUN}@example.test`,
      emailVerified: false,
    });
    const code = await codeFor(attempt, state);
    const consumed = await consume({ attempt, verifier, code });
    expect(consumed.statusCode, consumed.body).toBe(200);
    const result = consumed.json() as { status: string; session: { account: { id: string } } };
    // Ordinary known-subject login still works without a verified email.
    expect(result.status).toBe("logged_in");
    expect(result.session.account.id).toBe(account.accountId);
    // The unverified hint is a label: no reservation, no verified provenance.
    const reservation = await pool!.query(
      `SELECT normalized_email, verification_source, verified_at FROM account_email_reservations
       WHERE normalized_email = $1`,
      [`unverified-alias-${RUN}@example.test`],
    );
    expect(reservation.rows).toHaveLength(0);
    // The canonical primary email is unchanged.
    const user = await pool!.query<{ email: string }>(`SELECT email FROM users WHERE account_id = $1`, [account.accountId]);
    expect(user.rows[0]!.email).toBe(`unverified-owner-${RUN}@example.test`);
  });

  it("rejects cross-origin, iframe and foreign-identity entries", async () => {
    const verifier = newVerifier();
    const state = newState();
    const payload = {
      provider: "google",
      purpose: "login",
      role: "login",
      code_challenge: desktopCodeChallenge(verifier),
      state,
      client_label: "talent-signal-macos-test",
      web_origin: ORIGIN,
    };
    for (const options of [
      { origin: "https://evil.test" },
      { origin: null },
      { destination: "iframe" },
    ]) {
      const response = await inject("POST", "/v1/desktop-auth/prepare", { ...options, payload });
      expect(response.statusCode).toBe(403);
    }
  });
});

describeDB("credential proof rounds (real PostgreSQL)", () => {
  it("provider-only first password: current round mints one attempt and the sealed continuation replays as the same attempt", async () => {
    const subject = `google-sub-first-password-${RUN}`;
    const account = await createAccount({ email: `first-password-${RUN}@example.test`, kind: "google_human" });
    await linkIdentity(account.accountId, account.userId, "google", subject, `first-password-${RUN}@example.test`);
    const { token } = await sessionTokenFor(account.accountId, account.userId, "round-test");
    const challengeId = await reauthChallenge("google");
    const scope = await currentScope(account.accountId, account.userId);
    const flowRef = `flow-${randomUUID()}`;

    const run = await runRound({
      subject,
      email: `first-password-${RUN}@example.test`,
      token,
      prepare: roundPrepare({
        scope,
        intent: "set_password",
        targetProvider: "password",
        challengeId,
        flowRef,
      }),
    });
    expect(run.response.statusCode, run.response.body).toBe(200);
    const body = run.response.json() as {
      status: string;
      continuation: {
        credential_attempt_id: string;
        attempt_secret: string;
        intent: string;
        provider_challenge: unknown;
      };
    };
    expect(body.status).toBe("awaiting_password");
    expect(body.continuation.intent).toBe("set_password");
    expect(body.continuation.provider_challenge).toBeNull();

    // Exactly one credential attempt exists and no login cookie was created.
    const attempts = await pool!.query<{ count: string }>(
      `SELECT count(*)::text AS count FROM credential_change_attempts WHERE account_id = $1`,
      [account.accountId],
    );
    expect(Number(attempts.rows[0]!.count)).toBe(1);

    // Durable paired recovery: the SAME secret is re-derived and no second
    // attempt is minted when the consume response is lost.
    const result = await inject("POST", "/v1/desktop-auth/result", {
      token,
      payload: {
        attempt_id: run.attempt.attempt_id,
        pairing_secret: run.attempt.pairing_secret,
        verifier: run.verifier,
      },
    });
    expect(result.statusCode, result.body).toBe(200);
    const recovery = result.json() as {
      status: string;
      continuation: { credential_attempt_id: string; attempt_secret: string } | null;
    };
    expect(recovery.status).toBe("pending_continuation");
    expect(recovery.continuation!.credential_attempt_id).toBe(body.continuation.credential_attempt_id);
    expect(recovery.continuation!.attempt_secret).toBe(body.continuation.attempt_secret);
    // HKDF derivation matches the stored hash exactly.
    expect(
      deriveDesktopCredentialSecret(
        run.attempt.pairing_secret,
        flowRef,
        run.attempt.attempt_id,
        body.continuation.credential_attempt_id,
      ),
    ).toBe(body.continuation.attempt_secret);
    const attemptsAfter = await pool!.query<{ count: string }>(
      `SELECT count(*)::text AS count FROM credential_change_attempts WHERE account_id = $1`,
      [account.accountId],
    );
    expect(Number(attemptsAfter.rows[0]!.count)).toBe(1);

    // The paired acknowledgment confirms the WK continuation.
    const ack = await inject("POST", "/v1/desktop-auth/ack", {
      token,
      payload: {
        attempt_id: run.attempt.attempt_id,
        pairing_secret: run.attempt.pairing_secret,
        verifier: run.verifier,
        credential_attempt_id: body.continuation.credential_attempt_id,
      },
    });
    expect(ack.statusCode, ack.body).toBe(200);
    expect(ack.json()).toMatchObject({ acknowledged: true });
  });

  it("linking: current round mints the grant, target round completes it, canonical account/user unchanged", async () => {
    const subject = `apple-sub-link-round-${RUN}`;
    const account = await createAccount({ email: `link-round-${RUN}@example.test`, kind: "apple_human" });
    await linkIdentity(account.accountId, account.userId, "apple", subject, `link-round-${RUN}@example.test`);
    const { token } = await sessionTokenFor(account.accountId, account.userId, "link-round-test");
    const challengeId = await reauthChallenge("apple");
    const scope = await currentScope(account.accountId, account.userId);
    const flowRef = `flow-${randomUUID()}`;

    const current = await runRound({
      provider: "apple",
      subject,
      email: `link-round-${RUN}@example.test`,
      token,
      prepare: roundPrepare({
        scope,
        intent: "link_provider",
        targetProvider: "google",
        provider: "apple",
        challengeId,
        flowRef,
      }),
    });
    const currentBody = current.response.json() as {
      status: string;
      continuation: {
        credential_attempt_id: string;
        attempt_secret: string;
        provider_challenge: { challenge_id: string; oidc_nonce: string };
      };
    };
    expect(currentBody.status).toBe("awaiting_target");
    const targetChallenge = currentBody.continuation.provider_challenge;
    expect(targetChallenge.challenge_id).toBeTruthy();

    // Pre-ack rejection: the target round cannot begin before the current
    // round's WK continuation is acknowledged (ADR 0020 ordering).
    const preAck = await inject("POST", "/v1/desktop-auth/prepare", {
      token,
      payload: {
        client_label: "talent-signal-macos-test",
        web_origin: ORIGIN,
        code_challenge: desktopCodeChallenge(newVerifier()),
        state: newState(),
        ...roundPrepare({
          scope,
          intent: "link_provider",
          targetProvider: "google",
          role: "target",
          provider: "google",
          challengeId: targetChallenge.challenge_id,

        credential_attempt_id: currentBody.continuation.credential_attempt_id,

    });

    const state = newState();

        targetProvider: "google",

        provider: "google",

        flowRef,

      }),

      token,

      email: aliasEmail,
      token,

    // advisory lock (its account/user/session locks are already held).
    const blocker = await pool!.connect();
    await blocker.query("BEGIN");
    await blocker.query(`SELECT pg_advisory_xact_lock(hashtextextended($1, 0))`, [
      `account-email:${aliasEmail}`,
    ]);
    const consumePromise = consume({
      attempt: targetAttempt,
      verifier,
      code,
      token,
      attemptSecret: currentBody.continuation.attempt_secret,
    });
    // Give consume time to acquire its locks and block on the alias lock.
    await new Promise((resolve) => setTimeout(resolve, 500));

    // A concurrent governed mutation cannot slip between consume's check and
    // its write: it must wait for the shared credential locks.
    const writer = await pool!.connect();
    let writerDone = false;
    const writerPromise = writer
      .query(`UPDATE users SET profile_revision = profile_revision + 1 WHERE account_id = $1`, [account.accountId])
      .then(() => {
        writerDone = true;
      });
    const raced = await Promise.race([
      writerPromise.then(() => "writer"),
      new Promise<string>((resolve) => setTimeout(() => resolve("blocked"), 500)),
    ]);
    expect(raced).toBe("blocked");
    expect(writerDone).toBe(false);

    // Release the alias lock: the paused consume resolves FIRST (consume-first
    // commit order), then the writer's revision bump applies on top.
    await blocker.query("ROLLBACK");
    blocker.release();
    const response = await consumePromise;
    expect(response.statusCode, response.body).toBe(200);
    await writerPromise;
    writer.release();

    const rows = await pool!.query<{ status: string; result_committed_at: Date | null }>(
      `SELECT status, result_committed_at FROM desktop_auth_attempts WHERE id = $1`,

    const targetVerifier = newVerifier();
    const targetState = newState();
    const target = await prepare({
      ...roundPrepare({
        scope,
        intent: "link_provider",
        targetProvider: "google",
        role: "target",
        provider: "google",
        challengeId: grant.continuation.provider_challenge.challenge_id,
        credentialAttemptId: grant.continuation.credential_attempt_id,
        flowRef,
        roundRef: `round-target-${randomUUID()}`,
      }),
      verifier: targetVerifier,
      state: targetState,
      token,
    });
    await providerRound({
      attempt: target,
      state: targetState,
      provider: "google",
      subject: `google-sub-race-target-${randomUUID()}`,
      email: `race-target-${randomUUID()}@example.test`,
      token,
    });
    const targetCode = await codeFor(target, targetState, token);

    const { completeCredentialChange } = await import("./accountLoginMethods.js");
    const desktop = await import("./desktopAuth.js");
    const auth = {
      accountId: account.accountId,
      accountSlug: "race",
      userId: account.userId,
      userEmail: "race@example.test",
      userKind: "apple_human",
      sessionId,
    } as never;
    const caller = {
      webOrigin: ORIGIN,
      auth: auth as never,
      authFingerprint: sha256(token),
    } as Parameters<typeof desktop.consumeDesktopAuthAttempt>[3];

    // Force the critical interleaving: both consumers are admitted and then
    // block on the SAME grant row before either commits.
    const barrier = await pool!.connect();
    await barrier.query("BEGIN");
    await barrier.query(`SELECT id FROM credential_change_attempts WHERE id = $1 FOR UPDATE`, [
      grant.continuation.credential_attempt_id,
    ]);
    const ordinary = completeCredentialChange(
      pool!,
      config,
      auth,
      {
        attempt_id: grant.continuation.credential_attempt_id,
        attempt_secret: grant.continuation.attempt_secret,
        origin: "client:talent-signal-desktop",
        identity_token: googleToken(`google-sub-ordinary-${randomUUID()}`, `race-ordinary-${randomUUID()}@example.test`, "stale-nonce"),
      } as never,
      "client:talent-signal-desktop",
      deps,
      false,
    );
    const targetConsume = desktop.consumeDesktopAuthAttempt(
      pool!,
      config,
      {
        attempt_id: target.attempt_id,
        code: targetCode,
        verifier: targetVerifier,
        pairing_secret: target.pairing_secret,
        attempt_secret: grant.continuation.attempt_secret,
      },
      caller,
      deps,
      { attemptSecret: grant.continuation.attempt_secret },
    );
    await new Promise((resolve) => setTimeout(resolve, 300));
    await barrier.query("COMMIT");
    barrier.release();

    const settled = await Promise.allSettled([ordinary, targetConsume]);
    const outcomes = settled.map((entry) =>
      entry.status === "fulfilled"
        ? "committed"
        : String((entry.reason as { code?: string }).code ?? "error"),
    );
    // Exactly one authorized winner; the loser is an exact domain refusal and
    // never a deadlock victim or generic conflict wrapper.
    expect(outcomes.filter((outcome) => outcome === "committed"), outcomes.join("|")).toHaveLength(1);
    const loser = outcomes.find((outcome) => outcome !== "committed")!;
    expect(["CREDENTIAL_ATTEMPT_INVALID", "GOOGLE_CHALLENGE_INVALID", "APPLE_CHALLENGE_INVALID"]).toContain(loser);
    expect(loser).not.toBe("DESKTOP_AUTH_CONFLICT");
    expect(outcomes.join(",")).not.toContain("40P01");

    // The grant is single-use with exactly one audit and one revision
    // increment from this operation.
    const credential = await pool!.query<{ consumed_at: Date | null }>(
      `SELECT consumed_at FROM credential_change_attempts WHERE id = $1`,
      [grant.continuation.credential_attempt_id],
    );
    expect(credential.rows[0]!.consumed_at).not.toBeNull();
    const audits = await pool!.query<{ count: string }>(
      `SELECT count(*)::text AS count FROM account_access_events WHERE id = $1`,
      [grant.continuation.credential_attempt_id],
    );
    expect(Number(audits.rows[0]!.count)).toBe(1);
    const revisions = await pool!.query<{ settings_revision: number }>(
      `SELECT settings_revision FROM accounts WHERE id = $1`,
      [account.accountId],
    );
    expect(Number(revisions.rows[0]!.settings_revision)).toBe(
      scope.expected_account_revision + 1,
    );
  });

describeDB("durable completion fact across every consumer (F2/F3)", () => {
  async function relayedGrant(input: {
    intent: "set_password" | "change_password" | "unlink_provider" | "link_provider";
    targetProvider: "apple" | "google" | "password";
    provider?: "google" | "apple";
  }) {
    const provider = input.provider ?? "google";
    const subject = `fact-sub-${randomUUID()}`;
    const account = await createAccount({
      email: `fact-${randomUUID()}@example.test`,
      kind: provider === "google" ? "google_human" : "apple_human",
      password: "quiet-fact-1",
    });
    await linkIdentity(account.accountId, account.userId, provider, subject, `fact-${randomUUID()}@example.test`);
    const { token, sessionId } = await sessionTokenFor(account.accountId, account.userId, "fact-test");
    const challengeId = await reauthChallenge(provider);
    const scope = await currentScope(account.accountId, account.userId);
    const flowRef = `flow-${randomUUID()}`;
    const run = await runRound({
      provider,
      subject,
      email: `fact-${randomUUID()}@example.test`,
      token,
      prepare: roundPrepare({
        scope,
        intent: input.intent,
        targetProvider: input.targetProvider,
        provider,
        challengeId,
        flowRef,
      }),
    });
    const body = run.response.json() as {
      status: string;
      result?: { intent: string; status: string };
      continuation?: {
        credential_attempt_id: string;
        attempt_secret: string;
        provider_challenge?: { challenge_id: string; oidc_nonce: string };
      };
    };
    return { account, token, sessionId, scope, flowRef, run, body };
  }

  async function pairedRead(attemptId: string, pairingSecret: string, verifier: string, token?: string | undefined) {
    return inject("POST", "/v1/desktop-auth/result", {
      token,
      payload: { attempt_id: attemptId, pairing_secret: pairingSecret, verifier },
    });
  }

  it("F2: an ordinary first-password completion is recoverable as the same operation's committed receipt", async () => {
    const grant = await relayedGrant({ intent: "set_password", targetProvider: "password" });
    expect(grant.body.status).toBe("awaiting_password");
    const credentialAttemptId = grant.body.continuation!.credential_attempt_id;
    const attemptSecret = grant.body.continuation!.attempt_secret;

    // The current round is acknowledged exactly as the WK consumer does.
    const ack = await inject("POST", "/v1/desktop-auth/ack", {
      token: grant.token,
      payload: {
        attempt_id: grant.run.attempt.attempt_id,
        pairing_secret: grant.run.attempt.pairing_secret,
        verifier: grant.run.verifier,
        credential_attempt_id: credentialAttemptId,
      },
    });
    expect(ack.statusCode, ack.body).toBe(200);

    // The ordinary WK password Action completes this exact grant.
    const { completeCredentialChange } = await import("./accountLoginMethods.js");
    const result = await completeCredentialChange(
      pool!,
      config,
      {
        accountId: grant.account.accountId,
        accountSlug: "fact",
        userId: grant.account.userId,
        userEmail: "fact@example.test",
        userKind: "google_human",
        sessionId: grant.sessionId,
      } as never,
      {
        attempt_id: credentialAttemptId,
        attempt_secret: attemptSecret,
        origin: "client:talent-signal-desktop",
        password: "brand-new-fact-password-1",
      },
      "client:talent-signal-desktop",
      deps,
      false,
    );
    expect(result.status).toBe("password_set");

    // The response is "lost": the paired result read must recover the SAME
    // operation's committed receipt despite its own revision increment.
    const recovered = await pairedRead(
      grant.run.attempt.attempt_id,
      grant.run.attempt.pairing_secret,
      grant.run.verifier,
      grant.token,
    );
    expect(recovered.statusCode, recovered.body).toBe(200);
    const receipt = recovered.json() as {
      status: string;
      result: { intent: string; status: string };
      settings: { workspace: { id: string }; user: { id: string } };
    };
    expect(receipt.status).toBe("committed");
    expect(receipt.result).toEqual({ intent: "set_password", status: "password_set" });
    expect(receipt.settings.workspace.id).toBe(grant.account.accountId);
    expect(receipt.settings.user.id).toBe(grant.account.userId);

    // The acknowledgment of the committed effect stays operation-specific.
    const committedAck = await inject("POST", "/v1/desktop-auth/ack", {
      token: grant.token,
      payload: {
        attempt_id: grant.run.attempt.attempt_id,
        pairing_secret: grant.run.attempt.pairing_secret,
        verifier: grant.run.verifier,
        credential_attempt_id: credentialAttemptId,
      },
    });
    expect(committedAck.statusCode).toBe(200);
    expect(committedAck.json()).toMatchObject({ acknowledged: true, status: "committed" });

    // One audit and one revision increment; the original session is intact.
    const audits = await pool!.query<{ count: string }>(
      `SELECT count(*)::text AS count FROM account_access_events WHERE id = $1`,
      [credentialAttemptId],
    );
    expect(Number(audits.rows[0]!.count)).toBe(1);
    const revisions = await pool!.query<{ settings_revision: number }>(
      `SELECT settings_revision FROM accounts WHERE id = $1`,
      [grant.account.accountId],
    );
    expect(Number(revisions.rows[0]!.settings_revision)).toBe(grant.scope.expected_account_revision + 1);
    const live = await pool!.query<{ count: string }>(
      `SELECT count(*)::text AS count FROM sessions WHERE id = $1 AND revoked_at IS NULL`,
      [grant.sessionId],
    );
    expect(Number(live.rows[0]!.count)).toBe(1);

    // Actor isolation: anonymous, another account, a sibling session and a
    // revoked original never receive the receipt or its acknowledgment.
    const other = await createAccount({ email: `fact-other-${randomUUID()}@example.test` });
    const otherToken = (await sessionTokenFor(other.accountId, other.userId, "fact-other")).token;
    const sibling = (await sessionTokenFor(grant.account.accountId, grant.account.userId, "fact-sibling")).token;
    for (const token of [undefined, otherToken, sibling]) {
      const denied = await pairedRead(
        grant.run.attempt.attempt_id,
        grant.run.attempt.pairing_secret,
        grant.run.verifier,
        token,
      );
      expect([401, 409]).toContain(denied.statusCode);
      expect(["DESKTOP_AUTH_FINGERPRINT_CHANGED", "SESSION_INVALID", "AUTHENTICATION_REQUIRED"]).toContain(
        (denied.json() as { error: { code: string } }).error.code,
      );
    }
    await pool!.query(`UPDATE sessions SET revoked_at = now() WHERE id = $1`, [grant.sessionId]);
    const revoked = await pairedRead(
      grant.run.attempt.attempt_id,
      grant.run.attempt.pairing_secret,
      grant.run.verifier,
      grant.token,
    );
    expect([401, 409]).toContain(revoked.statusCode);
    const revokedAck = await inject("POST", "/v1/desktop-auth/ack", {
      token: grant.token,
      payload: {
        attempt_id: grant.run.attempt.attempt_id,
        pairing_secret: grant.run.attempt.pairing_secret,
        verifier: grant.run.verifier,
        credential_attempt_id: credentialAttemptId,
      },
    });
    expect([401, 409]).toContain(revokedAck.statusCode);
  });

  it("F3: a committed effect is never relabelled cancelled and its result is recovered", async () => {
    const grant = await relayedGrant({ intent: "link_provider", targetProvider: "google", provider: "apple" });
    const credentialAttemptId = grant.body.continuation!.credential_attempt_id;
    const targetChallenge = grant.body.continuation!.provider_challenge!;
    await inject("POST", "/v1/desktop-auth/ack", {
      token: grant.token,
      payload: {
        attempt_id: grant.run.attempt.attempt_id,
        pairing_secret: grant.run.attempt.pairing_secret,
        verifier: grant.run.verifier,
        credential_attempt_id: credentialAttemptId,
      },
    });

    // The target round is prepared and authorized, but the legitimate target

    const targetState = newState();

        flowRef: grant.flowRef,
        roundRef: `round-target-${randomUUID()}`,
      }),
      verifier: targetVerifier,
      state: targetState,
      token: grant.token,
    });
    await authorize(target, targetState, grant.token);

    const { completeCredentialChange } = await import("./accountLoginMethods.js");
    const linkedSubject = `google-sub-fact-target-${randomUUID()}`;

    const ordinary = await completeCredentialChange(
      pool!,
      config,
      {
        accountId: grant.account.accountId,
        accountSlug: "fact",
        userId: grant.account.userId,
        userEmail: "fact@example.test",
        userKind: "apple_human",
        sessionId: grant.sessionId,
      } as never,
      {
        attempt_id: credentialAttemptId,
        attempt_secret: grant.body.continuation!.attempt_secret,
        origin: "client:talent-signal-desktop",
        identity_token: googleToken(linkedSubject, linkedEmail, targetChallenge.oidc_nonce),
      },
      "client:talent-signal-desktop",
      deps,
      false,
    );
    expect(ordinary.status).toBe("linked");

    // Now cancel the still-open desktop target with BOTH authorities: the
    // committed effect is truthful history and can never be cancelled.
    for (const payload of [
      { attempt_id: target.attempt_id, mode: "wk", pairing_secret: target.pairing_secret },
      { attempt_id: target.attempt_id, mode: "system", state: targetState },

      expect((cancelled.json() as { status: string }).status).toBe("consumed");

    // The paired result reports the ACTUAL credential outcome for this grant.
    const recovered = await pairedRead(
      target.attempt_id,
      target.pairing_secret,
      targetVerifier,
      grant.token,
    );
    expect(recovered.statusCode, recovered.body).toBe(200);
    const receipt = recovered.json() as {
      status: string;
      result: { intent: string; status: string } | null;
    };
    expect(receipt.status).toBe("committed");
    expect(receipt.result).toEqual({ intent: "link_provider", status: "linked" });

    // The provider is linked, the grant is consumed, one audit exists and no
    // revocation audit was fabricated.
    const identities = await pool!.query<{ count: string }>(
      `SELECT count(*)::text AS count FROM auth_identities
       WHERE account_id = $1 AND provider = 'google'`,
      [grant.account.accountId],
    );

    const audits = await pool!.query<{ count: string }>(
      `SELECT count(*)::text AS count FROM account_access_events WHERE id = $1`,

    );

      [grant.account.accountId],

      [grant.account.accountId],
    );

  });
