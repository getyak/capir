import { CONTRACT_VERSION, type SessionResponse } from "@talent-signal/contracts";
import { afterEach, beforeEach, expect, it, vi } from "vitest";

import { POST } from "./route";

const secret = "s".repeat(43);

function labSession(): SessionResponse {
  return {
    contract_version: CONTRACT_VERSION,
    access_token: "k".repeat(43),
    expires_at: new Date(Date.now() + 3_600_000).toISOString(),
    account: {
      id: "11111111-1111-4111-8111-111111111111",
      slug: "lab-22222222-2222-4222-8222-222222222222",
      name: "Test workspace · 22222222",
    },
    user: {
      id: "33333333-3333-4333-8333-333333333333",
      email: "test-22222222@lab.invalid",
      display_name: "Test user",
      kind: "lab_human",
      role: "member",
      username: null,
    },
  };
}

function handoffRequest(
  body: unknown,
  headers: Record<string, string> = {},
): Request {
  return new Request("http://127.0.0.1:3000/capir/handoff", {
    method: "POST",
    headers: { "content-type": "application/json", ...headers },
    body: JSON.stringify(body),
  });
}

beforeEach(() => {
  vi.stubEnv("CAPIR_WEB_HANDOFF_KEY", "test-web-handoff-key-0123456789abcdef");
});

afterEach(() => {
  vi.unstubAllEnvs();
  vi.unstubAllGlobals();
});

it("exchanges the private secret and redirects only to /workspace", async () => {
  const calls: Array<{ url: string; init: RequestInit }> = [];
  vi.stubGlobal(
    "fetch",
    vi.fn(async (url: string, init: RequestInit) => {
      calls.push({ url: String(url), init });
      return new Response(
        JSON.stringify({
          schema_version: "capir.v1",
          session: labSession(),
          entry_path: "/capir/sandboxes/22222222-2222-4222-8222-222222222222",
        }),
        { status: 200, headers: { "content-type": "application/json" } },
      );
    }),
  );
  const response = await POST(handoffRequest({ handoff_secret: secret }));
  expect(response.status).toBe(303);
  expect(response.headers.get("location")).toBe("/workspace");
  const setCookie = response.headers.get("set-cookie") ?? "";
  expect(setCookie).toContain("talent-signal.session-v2=");
  expect(setCookie).toContain("HttpOnly");
  expect(setCookie).toContain("SameSite=Lax");
  expect(setCookie).toContain("Max-Age=");
  // The backend exchange is a private POST: the secret is in the body only.
  const backend = calls[0]!;
  expect(backend.url).toBe("http://127.0.0.1:4317/v1/capir/handoffs/exchange");
  expect(backend.url).not.toContain(secret);
  expect(String(backend.init.body)).toContain(secret);
});

it("refuses cross-site handoff posts", async () => {
  const fetchSpy = vi.fn();
  vi.stubGlobal("fetch", fetchSpy);
  for (const headers of [
    { origin: "https://evil.example" },
    { "sec-fetch-site": "cross-site" },
  ] as Array<Record<string, string>>) {
    const response = await POST(
      handoffRequest({ handoff_secret: secret }, headers),
    );
    expect(response.status).toBe(403);
    const body = (await response.json()) as { error: { code: string } };
    expect(body.error.code).toBe("CAPIR_HANDOFF_CSRF");
  }
  expect(fetchSpy).not.toHaveBeenCalled();
});

it("requires a private JSON body with a well-formed secret", async () => {
  const fetchSpy = vi.fn();
  vi.stubGlobal("fetch", fetchSpy);
  const wrongType = await POST(
    new Request("http://127.0.0.1:3000/capir/handoff", {
      method: "POST",
      headers: { "content-type": "text/plain" },
      body: secret,
    }),
  );
  expect(wrongType.status).toBe(415);
  const malformed = await POST(handoffRequest({ handoff_secret: "nope" }));
  expect(malformed.status).toBe(400);
  expect(fetchSpy).not.toHaveBeenCalled();
});

it("reports a consumed (replayed) secret as a denial without minting a cookie", async () => {
  // Test adapter behavior: the stub simulates the backend single-use fence.
  let consumed = false;
  vi.stubGlobal(
    "fetch",
    vi.fn(async () => {
      if (consumed) {
        return new Response(
          JSON.stringify({
            error: { code: "CAPIR_AUTH_DENIED", message: "denied", request_id: "r" },
          }),
          { status: 401, headers: { "content-type": "application/json" } },
        );
      }
      consumed = true;
      return new Response(
        JSON.stringify({
          schema_version: "capir.v1",
          session: labSession(),
          entry_path: "/capir/sandboxes/22222222-2222-4222-8222-222222222222",
        }),
        { status: 200, headers: { "content-type": "application/json" } },
      );
    }),
  );
  const first = await POST(handoffRequest({ handoff_secret: secret }));
  expect(first.status).toBe(303);
  const replay = await POST(handoffRequest({ handoff_secret: secret }));
  expect(replay.status).toBe(401);
  expect(replay.headers.get("set-cookie")).toBeNull();
  const body = (await replay.json()) as { error: { code: string } };
  expect(body.error.code).toBe("CAPIR_AUTH_DENIED");
});

it("sends the exact addressed origin even when Next normalizes the URL host", async () => {
  // Regression: next dev normalizes request.url to localhost while grants and
  // handoffs bind the literal loopback origin from the configuration.
  const calls: Array<{ init: RequestInit }> = [];
  vi.stubGlobal(
    "fetch",
    vi.fn(async (_url: string, init: RequestInit) => {
      calls.push({ init });
      return new Response(
        JSON.stringify({
          schema_version: "capir.v1",
          session: labSession(),
          entry_path: "/capir/sandboxes/22222222-2222-4222-8222-222222222222",
        }),
        { status: 200, headers: { "content-type": "application/json" } },
      );
    }),
  );
  const response = await POST(
    new Request("http://localhost:3000/capir/handoff", {
      method: "POST",
      headers: {
        "content-type": "application/json",
        host: "127.0.0.1:3000",
        origin: "http://127.0.0.1:3000",
      },
      body: JSON.stringify({ handoff_secret: secret }),
    }),
  );
  expect(response.status).toBe(303);
  expect(JSON.parse(String(calls[0]!.init.body))).toEqual({
    handoff_secret: secret,
    web_origin: "http://127.0.0.1:3000",
  });
  // A page origin that differs from the addressed host is still refused.
  const cross = await POST(
    new Request("http://localhost:3000/capir/handoff", {
      method: "POST",
      headers: {
        "content-type": "application/json",
        host: "127.0.0.1:3000",
        origin: "http://localhost:3000",
      },
      body: JSON.stringify({ handoff_secret: secret }),
    }),
  );
  expect(cross.status).toBe(403);
});

it("fails closed when the trusted consumer key is not configured", async () => {
  vi.stubEnv("CAPIR_WEB_HANDOFF_KEY", "");
  const fetchSpy = vi.fn();
  vi.stubGlobal("fetch", fetchSpy);
  const response = await POST(handoffRequest({ handoff_secret: secret }));
  expect(response.status).toBe(503);
  const body = (await response.json()) as { error: { code: string } };
  expect(body.error.code).toBe("CAPIR_WEB_HANDOFF_UNAVAILABLE");
  expect(fetchSpy).not.toHaveBeenCalled();
});
