    const f = await fixture();

    const tickGate = new Promise<void>(resolve => { releaseTick = resolve; });
    const finalGate = new Promise<void>(resolve => { releaseFinal = resolve; });
    let watch = false, tickHeld = false, tickReleased = false, finalEntered = false, postProducerProbes = 0, probesDuringFinal = 0;

        const normalized = sql.replace(/\s+/gu," ").trim();
        const probe = normalized.startsWith("SELECT original_authority FROM legacy_chat_task_attempts");
        if (probe && finalEntered) probesDuringFinal += 1;
        const result = await client.query(sql,values);
        if (watch && probe && !tickHeld) { tickHeld = true;heldTick = true;await tickGate; }
        else if (probe && h?.producer.returned && !heldTick && !finalEntered) postProducerProbes += 1;
        if (normalized.startsWith("UPDATE context_manifests SET status='active'")) {
          expect(tickReleased,"actual heartbeat client release precedes our final mutation").toBe(true);
          finalEntered = true;await finalGate;
        }
        return result;
      }};

      await vi.waitFor(() => { expect(h!.producer.returned).toBe(true);expect(postProducerProbes).toBeGreaterThanOrEqual(2); }, {timeout:10000});
