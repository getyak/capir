    func testRevokedAuthorityAdmitsNoNewDispatchAndCannotReinstallAnOldLogin() async throws {
        ControlledURLProtocol.reset()
        defer { ControlledURLProtocol.reset() }
        let service = try Self.makeControlledService()

        let firstLoad = Task { try await service.loadWorkspace() }
        for _ in 0..<1_000 where ControlledURLProtocol.recordedCount(matching: "simulated-login") == 0 {
            await Task.yield()
        }
        XCTAssertEqual(ControlledURLProtocol.recordedCount(matching: "simulated-login"), 1)

        service.invalidateAuthority(.authentication)

        // The login response lands only after the revocation boundary.
        ControlledURLProtocol.answer(matching: "simulated-login", with: .ok(Self.loginBody(token: "token-old")))
        do {
            _ = try await firstLoad.value
            XCTFail("A response that lands after revocation must not install authority.")
        } catch {
            XCTAssertTrue(error.localizedDescription.contains("revoked"))
        }

        XCTAssertEqual(
            ControlledURLProtocol.recordedCount(matching: "v1/pursuits"),
            0,
            "No scope readback may be dispatched after revocation."
        )
        XCTAssertEqual(
            ControlledURLProtocol.recordedCount(matching: "v1/"),
            1,
            "Only the already admitted login ran; no further request left this process."
        )

        // The discarded login cannot be reinstalled or reused: the next load
        // must authenticate again and only the fresh credential may travel.
        let secondLoad = Task { try await service.loadWorkspace() }
        for _ in 0..<1_000 where ControlledURLProtocol.recordedCount(matching: "simulated-login") < 2 {
            await Task.yield()
        }
        XCTAssertEqual(ControlledURLProtocol.recordedCount(matching: "simulated-login"), 2)
        ControlledURLProtocol.answer(matching: "simulated-login", with: .ok(Self.loginBody(token: "token-new")))
        ControlledURLProtocol.answer(matching: "v1/pursuits", with: .failure)
        ControlledURLProtocol.answer(matching: "v1/people", with: .failure)
        ControlledURLProtocol.answer(matching: "v1/pursuit-proposals", with: .failure)
        _ = try? await secondLoad.value

        let dispatchedTokens = ControlledURLProtocol.authorizations(matching: "v1/")
        XCTAssertFalse(dispatchedTokens.contains("Bearer token-old"), "Revoked credentials are never dispatched again.")
        XCTAssertTrue(dispatchedTokens.contains("Bearer token-new"))
    }

    func testLogoutCleanupComparesOperationIdentityAndPreservesANewerOperationRecord() async throws {
        ControlledURLProtocol.reset()
        defer { ControlledURLProtocol.reset() }
        let directory = FileManager.default.temporaryDirectory
            .appending(path: "talent-signal-unknown-resolution-\(UUID().uuidString)", directoryHint: .isDirectory)
        defer { try? FileManager.default.removeItem(at: directory) }
        let store = SecureUnknownResolutionStore(
            directory: directory,
            keyProvider: FixedUnknownResolutionKeyProvider()
        )
        try store.save(Self.durableResolution(operationID: "operation-old"), scopeKey: Self.controlledScopeKey)
        let service = try Self.makeControlledService(store: store)
        // A REAL original login is warmed before retirement; the zero-login
        // case is covered separately and stays truthfully local-only.
        try await Self.completeWorkspaceLoad(service: service, token: "token-logout")
        XCTAssertEqual(ControlledURLProtocol.recordedCount(matching: "simulated-login"), 1)

        let retirement = service.withdrawAuthentication()
        let signOut = Task { try await service.signOut(ticket: retirement) }
        for _ in 0..<1_000 where ControlledURLProtocol.recordedCount(matching: "logout") == 0 {
            await Task.yield()
        }
        XCTAssertEqual(
            ControlledURLProtocol.recordedCount(matching: "logout"),
            1,
            "The captured credential is used only for this exact logout."
        )

        // A newer operation's record lands while the logout is still in
        // flight. The static scope key alone must not delete it.
        try store.save(Self.durableResolution(operationID: "operation-new"), scopeKey: Self.controlledScopeKey)

        ControlledURLProtocol.answer(matching: "logout", with: .ok(Self.logoutBody))
        let outcome = try await signOut.value

        guard case .revoked = outcome.remote else {
            return XCTFail("Expected a verified remote revocation of the original session")
        }
        XCTAssertEqual(
            try store.load(scopeKey: Self.controlledScopeKey)?.operationID,
            "operation-new",
            "Logout cleanup compares the captured operation identity and never deletes a newer operation's record."
        )
    }

    func testDispatchedDecisionWriteStaysUnknownAndNeverRecreatesAClearedRecoveryFile() async throws {
        ControlledURLProtocol.reset()
        defer { ControlledURLProtocol.reset() }
        let directory = FileManager.default.temporaryDirectory
            .appending(path: "talent-signal-unknown-resolution-\(UUID().uuidString)", directoryHint: .isDirectory)
        defer { try? FileManager.default.removeItem(at: directory) }
        let store = SecureUnknownResolutionStore(
            directory: directory,
            keyProvider: FixedUnknownResolutionKeyProvider()
        )
        try store.save(Self.durableResolution(operationID: "operation-seed"), scopeKey: Self.controlledScopeKey)
        let service = try Self.makeControlledService(store: store)

        let load = Task { try await service.loadWorkspace() }
        for _ in 0..<1_000 where ControlledURLProtocol.recordedCount(matching: "simulated-login") == 0 {
            await Task.yield()
        }
        ControlledURLProtocol.answer(matching: "simulated-login", with: .ok(Self.loginBody(token: "token-live")))
        for _ in 0..<1_000 where ControlledURLProtocol.recordedCount(matching: "v1/pursuits") == 0 {
            await Task.yield()
        }
        ControlledURLProtocol.answer(matching: "v1/pursuits", with: .ok(Self.pursuitsBody))
        ControlledURLProtocol.answer(matching: "v1/people", with: .ok(Self.peopleBody))
        ControlledURLProtocol.answer(matching: "v1/pursuit-proposals", with: .ok(Self.proposalsBody))
        for _ in 0..<1_000 where ControlledURLProtocol.recordedCount(matching: "contexts/") == 0 {
            await Task.yield()
        }
        ControlledURLProtocol.answer(matching: "contexts/", with: .ok(Self.relationshipBody))
        let loaded = try await load.value
        guard case .canonical(let restored) = loaded, restored.displayMode == .outcomeUnknown else {
            return XCTFail("Expected the durable outcome-unknown operation to be restored without a new write.")
        }

        let request = CanonicalDecisionRequest(
            bundleID: "bundle-1",
            taskID: "task-1",
            taskRevision: 2,
            bundleRevision: 1,
            proposalID: "proposal-1",
            baseRevision: 7,
            reason: "Controlled dispatch.",
            decisions: []
        )
        let resolve = Task { try await service.resolveDecision(request) }
        for _ in 0..<1_000 where ControlledURLProtocol.recordedCount(matching: "resources/") == 0 {
            await Task.yield()
        }
        ControlledURLProtocol.answer(matching: "resources/", with: .ok(Self.resourceBody))
        for _ in 0..<1_000 where ControlledURLProtocol.recordedCount(matching: "resolve") == 0 {
            await Task.yield()
        }
        XCTAssertEqual(ControlledURLProtocol.recordedCount(matching: "resolve"), 1)
        XCTAssertEqual(
            try store.load(scopeKey: Self.controlledScopeKey)?.operationID != "operation-seed",
            true,
            "The exact dispatched operation identity is persisted before the write leaves this process."
        )

        // Sign-out cleanup clears the durable file before the lost response
        // lands. The late catch must never recreate it.
        try store.clear(scopeKey: Self.controlledScopeKey)
        for _ in 0..<4 { ControlledURLProtocol.answer(matching: "v1/operations/", with: .failure) }
        ControlledURLProtocol.answer(matching: "resolve", with: .failure)

        let resolved = try await resolve.value
        guard case .canonical(let unknown) = resolved, unknown.displayMode == .outcomeUnknown else {
            return XCTFail("An already dispatched write must stay outcome-unknown, never be reported as canceled.")
        }
        XCTAssertNil(
            try store.load(scopeKey: Self.controlledScopeKey),
            "A late decision catch must never recreate the cleared durable recovery file."
        )
        XCTAssertEqual(
            ControlledURLProtocol.recordedCount(matching: "resolve"),
            1,
            "Reconciliation and recovery must never manufacture a second decision write."
        )
    }

    func testRetirementAndMarkerCleanupReturnPromptlyWithoutLockDeadlock() throws {
        let directory = FileManager.default.temporaryDirectory
            .appending(path: "talent-signal-unknown-resolution-\(UUID().uuidString)", directoryHint: .isDirectory)
        defer { try? FileManager.default.removeItem(at: directory) }
        let store = SecureUnknownResolutionStore(
            directory: directory,
            keyProvider: FixedUnknownResolutionKeyProvider()
        )
        try store.save(Self.durableResolution(operationID: "operation-old"), scopeKey: Self.controlledScopeKey)
        let operationFile = Self.operationFile(directory: directory)
        let interruptedCorrelation = try Data(contentsOf: operationFile)

        // Normal retirement of an EXISTING correlation must return promptly:
        // the nested clear runs inside the shared namespace lock without
        // deadlocking on the nonrecursive lock.
        let box = RetiredBox()
        Thread.detachNewThread {
            box.result = try? store.retireOperation(
                scopeKey: Self.controlledScopeKey,
                retirement: Self.retirement(operationID: "operation-old"),
                now: Date()
            )
        }
        let deadline = Date().addingTimeInterval(5)
        while box.result == nil && Date() < deadline {
            Thread.sleep(forTimeInterval: 0.02)
        }
        XCTAssertNotNil(box.result, "retirement of an existing correlation must return promptly (no nested-lock deadlock)")
        XCTAssertEqual(box.result, .established(correlationRemoved: true, newerOperationPreserved: false))

        // Fresh-store marker-backed cleanup on restart must also return
        // promptly and never restore the retired operation.
        let restarted = SecureUnknownResolutionStore(
            directory: directory,
            keyProvider: FixedUnknownResolutionKeyProvider()
        )
        XCTAssertThrowsError(try restarted.save(Self.durableResolution(operationID: "operation-old"), scopeKey: Self.controlledScopeKey))
        // Owned synthetic crash fixture: leave the original encrypted bytes as
        // if marker persistence completed but correlation unlink did not.
        try interruptedCorrelation.write(to: operationFile)
        let reloadBox = RetiredBox()
        Thread.detachNewThread {
            reloadBox.value = try? restarted.load(scopeKey: Self.controlledScopeKey)
        }
        let reloadDeadline = Date().addingTimeInterval(5)
        while reloadBox.finishedAfterRead == false && Date() < reloadDeadline {
            Thread.sleep(forTimeInterval: 0.02)
        }
        XCTAssertTrue(reloadBox.finishedAfterRead, "marker-backed restart cleanup must return promptly (no nested-lock deadlock)")
        XCTAssertNil(reloadBox.value, "A retired operation never restores across restart.")
    }

private final class ControlledURLProtocol: URLProtocol {
    struct Reply: Sendable {
        let statusCode: Int
        let body: Data

        static func ok(_ body: Data) -> Reply { Reply(statusCode: 200, body: body) }
        static let failure = Reply(statusCode: 500, body: Data("{}".utf8))
    }

    /// One held request. A single struct keeps the tuple-order question moot.
    fileprivate struct HeldRequest {
        let key: String
        let requestProtocol: URLProtocol
    }

    private final class State: @unchecked Sendable {
        let lock = NSLock()
        var held: [HeldRequest] = []
        var pendingAnswers: [(fragment: String, reply: Reply)] = []
        var recorded: [(key: String, authorization: String?)] = []
        var requestURLs: [URL] = []
        var deliveredStatuses: [String] = []
    }

    private static let state = State()
    static let session: URLSession = {
        let configuration = URLSessionConfiguration.ephemeral
        configuration.protocolClasses = [ControlledURLProtocol.self]
        return URLSession(configuration: configuration)
    }()

    private static func key(for request: URLRequest) -> String {
        "\(request.httpMethod ?? "GET") \(request.url?.path ?? "")"
    }

    static func reset() {
        state.lock.lock()
        defer { state.lock.unlock() }
        state.held = []
        state.pendingAnswers = []
        state.recorded = []
        state.requestURLs = []
        state.deliveredStatuses = []
    }

    static func recordedCount(matching fragment: String) -> Int {
        state.lock.lock()
        defer { state.lock.unlock() }
        return state.recorded.filter { $0.key.contains(fragment) }.count
    }

    static func authorizations(matching fragment: String) -> [String] {
        state.lock.lock()
        defer { state.lock.unlock() }
        return state.recorded.filter { $0.key.contains(fragment) }.compactMap(\.authorization)
    }

    static func deliveredTaskStatuses() -> [String] {
        state.lock.lock(); defer { state.lock.unlock() }
        return state.deliveredStatuses
    }

    static func queryValues(name: String) -> [String] {
        state.lock.lock(); defer { state.lock.unlock() }
        return state.requestURLs.compactMap { url in
            guard let query = URLComponents(url: url, resolvingAgainstBaseURL: false)?.percentEncodedQuery else { return nil }
            for part in query.split(separator: "&") {
                let fields = part.split(separator: "=", maxSplits: 1, omittingEmptySubsequences: false)
                if fields.count == 2 && String(fields[0]).removingPercentEncoding == name {
                    return String(fields[1]).replacingOccurrences(of: "+", with: " ").removingPercentEncoding
                }
            }
            return nil
        }
    }

    /// Answers the first held request matching the fragment, or stages the
    /// reply for the next matching request. Held requests stay in flight until
    /// answered, which is the seam for revocation races.
    static func answer(matching fragment: String, with reply: Reply) {
        var waiter: HeldRequest?
        state.lock.lock()
        if let index = state.held.firstIndex(where: { $0.key.contains(fragment) }) {
            waiter = state.held.remove(at: index)
        } else {
            state.pendingAnswers.append((fragment, reply))
        }
        state.lock.unlock()
        (waiter?.requestProtocol as? ControlledURLProtocol)?.deliver(reply)
    }

    /// Releases every currently held request, e.g. to finish a test cleanly.
    static func answerEveryHeld(with reply: Reply) {
        var waiters: [HeldRequest] = []
        state.lock.lock()
        waiters = state.held
        state.held = []
        state.lock.unlock()
        for waiter in waiters {
            (waiter.requestProtocol as? ControlledURLProtocol)?.deliver(reply)
        }
    }

    override class func canInit(with request: URLRequest) -> Bool { true }
    override class func canonicalRequest(for request: URLRequest) -> URLRequest { request }

    override func startLoading() {
        let request = self.request
        let key = Self.key(for: request)
        var reply: Reply?
        Self.state.lock.lock()
        Self.state.recorded.append((key, request.value(forHTTPHeaderField: "authorization")))
        if let url = request.url { Self.state.requestURLs.append(url) }
        if let index = Self.state.pendingAnswers.firstIndex(where: { key.contains($0.fragment) }) {
            reply = Self.state.pendingAnswers.remove(at: index).reply
        } else {
            Self.state.held.append(HeldRequest(key: key, requestProtocol: self))
        }
        Self.state.lock.unlock()
        if let reply { deliver(reply) }
    }

    override func stopLoading() {
        Self.state.lock.lock()
        Self.state.held.removeAll { $0.requestProtocol === self }
        Self.state.lock.unlock()
    }

    private func deliver(_ reply: Reply) {
        if request.httpMethod == "GET", request.url?.path.contains("/v1/agent-tasks/") == true,
           let envelope = try? JSONSerialization.jsonObject(with: reply.body) as? [String: Any],
           let task = envelope["task"] as? [String: Any], let status = task["status"] as? String {
            Self.state.lock.lock()
            Self.state.deliveredStatuses.append(status)
            Self.state.lock.unlock()
        }
        guard let url = request.url, let client else { return }
        let response = HTTPURLResponse(
            url: url,
            statusCode: reply.statusCode,
            httpVersion: "HTTP/1.1",
            headerFields: ["content-type": "application/json"]
        )!
        client.urlProtocol(self, didReceive: response, cacheStoragePolicy: .notAllowed)
        client.urlProtocol(self, didLoad: reply.body)
        client.urlProtocolDidFinishLoading(self)
    }
}

/// Wraps the actual encrypted retirement home and injects exact failure seams
/// for load, clear, and marker writes.
private final class ControlledRetirementFailureStore: UnknownResolutionPersisting, @unchecked Sendable {
    let backing: SecureUnknownResolutionStore

    init() {
        let directory = FileManager.default.temporaryDirectory
            .appending(path: "talent-signal-retirement-failure-\(UUID().uuidString)", directoryHint: .isDirectory)
        backing = SecureUnknownResolutionStore(
            directory: directory,
            keyProvider: FixedUnknownResolutionKeyProvider()
        )
    }
    var loadResult: Result<Void, Error> = .success(())
    var retireResult: Result<UnknownRetirementEstablishment, Error> = .success(
        .established(correlationRemoved: true, newerOperationPreserved: false)
    )

    func load(scopeKey: String) throws -> DurableUnknownResolution? {
        try loadResult.get()
        return try backing.load(scopeKey: scopeKey)
    }

    func save(_ resolution: DurableUnknownResolution, scopeKey: String) throws {
        try backing.save(resolution, scopeKey: scopeKey)
    }

    func clear(scopeKey: String) throws {
        try backing.clear(scopeKey: scopeKey)
    }

    func retireOperation(
        scopeKey: String,
        retirement: DurableUnknownRetirement,
        now: Date
    ) throws -> UnknownRetirementEstablishment {
        try retireResult.get()
    }

    func retirementState(scopeKey: String, operationID: String) throws -> DurableUnknownRetirement? {
        try backing.retirementState(scopeKey: scopeKey, operationID: operationID)
    }
}


/// Tiny thread-safe result box for deadlock-regression timing checks.
private final class RetiredBox: @unchecked Sendable {
    private let lock = NSLock()
    private var storedResult: UnknownRetirementEstablishment?
    private var storedValue: DurableUnknownResolution?
    private var didReadValue = false
    private var done = false

    var result: UnknownRetirementEstablishment? {
        get { lock.lock(); defer { lock.unlock() }; return storedResult }
        set { lock.lock(); defer { lock.unlock() }; storedResult = newValue; done = true }
    }

    var value: DurableUnknownResolution? {
        get { lock.lock(); defer { lock.unlock() }; return storedValue }
        set { lock.lock(); defer { lock.unlock() }; storedValue = newValue; didReadValue = true; done = true }
    }

    var finishedAfterRead: Bool {
        lock.lock(); defer { lock.unlock() }; return done && didReadValue
    }

    var finished: Bool {
        lock.lock(); defer { lock.unlock() }; return done
    }
}

/// Creates an actual ENOTDIR atomic-write failure, preserving the original
/// encrypted file in an owned renamed directory. No stubs report a save pass.
private final class PhysicalTaskCapsuleFailureStore: LocalCapsulePersisting, @unchecked Sendable {
    enum Point { case canonicalAssociation, terminalCleanup, taskPreparation }
    let backing: SecureLocalCapsuleStore
    private let directory: URL
    private let held: URL
    private let failAt: Point
    private var didFail = false
    init(directory: URL, failAt: Point) {
        self.directory = directory
        held = directory.appendingPathExtension("held")
        self.failAt = failAt
        backing = SecureLocalCapsuleStore(directory: directory, keyProvider: TaskAdmissionTestKeyProvider())
    }
    func load(accountID: String, now: Date) throws -> CapsuleRecoveryResult { try backing.load(accountID: accountID, now: now) }
    func clear(accountID: String, deleteKey: Bool) throws -> LocalCapsuleDeletionReceipt { try backing.clear(accountID: accountID, deleteKey: deleteKey) }
    func save(_ draft: ContextCapsuleDraft, accountID: String, now: Date) throws {
        let matches: Bool
        switch failAt {
        case .taskPreparation: matches = draft.workflowOwnerships?.contains { $0.phase == .taskOutcomeUnknown && $0.taskAdmission != nil } == true
        case .canonicalAssociation: matches = draft.workflowOwnerships?.contains { $0.phase == .canonicalTask } == true
        case .terminalCleanup: matches = draft.workflowOwnerships?.isEmpty != false && draft.items.count == 1
        }
        if !didFail && matches {
            try FileManager.default.moveItem(at: directory, to: held)
            try Data("owned physical failure seam".utf8).write(to: directory)
            didFail = true
        }
        try backing.save(draft, accountID: accountID, now: now)
    }
    func repairPhysicalFault() throws {
        guard FileManager.default.fileExists(atPath: held.path) else { return }
        try FileManager.default.removeItem(at: directory)
        try FileManager.default.moveItem(at: held, to: directory)
    }
}
