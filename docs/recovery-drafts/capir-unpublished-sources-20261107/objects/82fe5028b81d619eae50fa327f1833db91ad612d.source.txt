async function fixture(options: { execution?: "past"|"future"; canonical?: "past"|"future";
  operation?: "create_chat_task"|"create_unscoped_chat_task"; original?: boolean; completed?: boolean; dispatched?: boolean } = {}) {
  const accountSlug = `maintenance-${randomUUID()}`, userEmail = `${randomUUID()}@synthetic.invalid`;
  const account = randomUUID(), actor = randomUUID(), session = randomUUID(), intent = randomUUID(), attempt = randomUUID(), task = randomUUID(), run = randomUUID();
  const now = (await pool!.query("SELECT clock_timestamp() AS now")).rows[0]!.now as Date;
  const execution = new Date(now.valueOf() + (options.execution === "future" ? 3600000 : -3600000));
  const canonical = new Date(now.valueOf() + (options.canonical === "past" ? -7200000 : 86400000));
  const state = options.completed ? "completed" : "running";
  const dispatch = options.dispatched === false ? "not_dispatched" : "dispatched";
  const original = {version:1,kind:"ordinary",session:{accountID:account,userID:actor,id:session,actorKind:"simulated_human",expiresAt:execution.valueOf()}};
  await inTransaction(pool!,async client=>{
    await client.query("INSERT INTO accounts(id,slug,name) VALUES($1,$2,'Synthetic metadata')",[account,accountSlug]);
    await client.query("INSERT INTO users(id,account_id,email,display_name,kind) VALUES($1,$2,$3,'Synthetic owner','simulated_human')",[actor,account,userEmail]);
    await client.query("INSERT INTO sessions(id,account_id,user_id,token_hash,client_label,expires_at) VALUES($1,$2,$3,$4,'synthetic',$5)",[session,account,actor,randomUUID(),execution]);
    await client.query(`INSERT INTO legacy_chat_task_intents(id,account_id,actor_user_id,operation_scope,idempotency_key,request_hash,task_id,state,
      current_attempt_id,fencing_generation,auth_session_id,execution_authority_expires_at,canonical_retention_expires_at)
      VALUES($1,$2,$3,$4,$5,repeat('a',64),$6,$7,$8,1,$9,$10,$11)`,[intent,account,actor,options.operation??"create_chat_task",randomUUID(),task,state,attempt,session,execution,canonical]);
    await client.query(`INSERT INTO legacy_chat_task_attempts(id,account_id,intent_id,actor_user_id,task_id,run_id,fencing_generation,state,auth_session_id,
      auth_actor_kind,execution_authority_expires_at,original_authority,dispatch_state,lease_until)
      VALUES($1,$2,$3,$4,$5,$6,1,$7,$8,'simulated_human',$9,$10::jsonb,$11,clock_timestamp()+interval '1 hour')`,
      [attempt,account,intent,actor,task,run,state,session,execution,options.original===false?null:JSON.stringify(original),dispatch]);
  });
  const identity:LifecycleIdentity = {intentID:intent,attemptID:attempt,taskID:task,runID:run,fencingGeneration:1,revision:1,
    state,dispatchState:dispatch,executionAuthorityExpiresAt:execution,canonicalRetentionExpiresAt:canonical};
  const owner: LifecycleMaintenanceOwner = {accountID:account,actorUserID:actor,operationScope:"create_chat_task",
    intentID:intent,attemptID:attempt,taskID:task,runID:run,fencingGeneration:1};
  const settlement=gate(), controller = new AbortController(), failures:ApiError[]=[];
  const register = (events?:string[]) => registerLifecycleRun(identity,controller,settlement.pending,{accountID:account,actorUserID:actor,
    operationScope:"create_chat_task",onFailure:error=>{events?.push("hook");failures.push(error);}});
  const cleanup = () => {settlement.release();releaseLifecycleRun(attempt);};
  const read = async () => (await pool!.query(`SELECT i.state AS intent_state,i.terminal_reason AS intent_reason,
    i.current_attempt_id,i.fencing_generation,i.canonical_retention_expires_at AS canonical,
    a.state AS attempt_state,a.terminal_reason AS attempt_reason,a.usage_uncertain,a.dispatch_state,
    a.execution_authority_expires_at AS execution FROM legacy_chat_task_intents i JOIN legacy_chat_task_attempts a
    ON a.account_id=i.account_id AND a.id=$2 WHERE i.account_id=$1 AND i.id=$3`,[account,attempt,intent])).rows[0]!;
  return {account,accountSlug,userEmail,actor,session,intent,attempt,task,run,execution,canonical,identity,owner,controller,failures,register,cleanup,read};
}
