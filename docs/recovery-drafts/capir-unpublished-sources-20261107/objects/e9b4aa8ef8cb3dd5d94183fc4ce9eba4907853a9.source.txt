// @vitest-environment happy-dom
import { CONTRACT_VERSION, type SessionResponse } from "@talent-signal/contracts";
import type { ReactElement } from "react";
import { renderToStaticMarkup } from "react-dom/server";
import { expect, it } from "vitest";

import { capirSessionClaims } from "@/lib/capirHandoff";
import { CapirDemoBanner } from "@/lib/capirSandboxBanner";

/**
 * The workspace chrome for a direct CLI-created Lab primary session derives
 * from the verified Lab account only: sandbox identity and expiry come from
 * the handoff exchange session, never from parent test-workspace cookies.
 */
function labSession(): SessionResponse {
  return {
    contract_version: CONTRACT_VERSION,
    access_token: "k".repeat(43),
    expires_at: "2026-10-01T12:00:00.000Z",
    account: {
      id: "11111111-1111-4111-8111-111111111111",
      slug: "lab-22222222-2222-4222-8222-222222222222",
      name: "Test workspace · 22222222",
    },
    user: {
      id: "33333333-3333-4333-8333-333333333333",
      email: "test-22222222@lab.invalid",
      display_name: "Test user",
      kind: "lab_human",
      role: "member",
      username: null,
    },
  };
}

it("binds the rendered demo banner to the verified Lab session claims", () => {
  const claims = capirSessionClaims(labSession());
  const banner = CapirDemoBanner({
    sandboxId: claims.capir.sandbox_id,
    sandboxExpiresAt: "2026-10-01T15:00:00.000Z",
    sessionExpiresAt: claims.backendExpiresAt,
  });
  const html = renderToStaticMarkup(banner as ReactElement);
  expect(html).toContain('data-capir-session-state="active"');
  expect(html).toContain("22222222-2222-4222-8222-222222222222");
  expect(html).toContain('data-capir-demo-expiry="2026-10-01T15:00:00.000Z"');
  expect(html).toContain("浏览器会话有效期至 2026-10-01 12:00 UTC");
});
