"""Real POSIX terminal acceptance driver; owns and closes only its child and PTYs."""
import errno, json, os, pty, select, subprocess, sys, time, fcntl, termios, struct
from pathlib import Path
spec=json.load(sys.stdin)
in_master,in_slave=pty.openpty()
out_master,out_slave=pty.openpty()
err_master,err_slave=pty.openpty() if spec.get('stderr_tty') else (None,None)
for fd in [out_slave,err_slave]:
    if fd is not None: fcntl.ioctl(fd,termios.TIOCSWINSZ,struct.pack('HHHH',24,120,0,0))
child=subprocess.Popen([spec['node'],spec['binary'],'chat',*spec.get('args',[])],stdin=in_slave,stdout=out_slave,stderr=err_slave if err_slave is not None else subprocess.PIPE,env=os.environ.copy(),start_new_session=True)
if err_slave is not None: os.close(err_slave)
err_fd=err_master if err_master is not None else child.stderr.fileno()
os.close(in_slave);os.close(out_slave)
stdout=bytearray();stderr=bytearray();transcript=bytearray();search_cursor=0;active={out_master:stdout,err_fd:stderr}
pause_stderr=False;pause_stdout=False
def pump(timeout=0.05):
    readable=[fd for fd in active if not (child.poll() is None and ((pause_stderr and fd==err_fd) or (pause_stdout and fd==out_master)))]
    for fd in select.select(readable,[],[],timeout)[0]:
        try: data=os.read(fd,65536)
        except OSError as e:
            if e.errno!=errno.EIO: raise
            data=b''
        if data: active[fd].extend(data);transcript.extend(data)
        else: active.pop(fd,None)
def wait_for(text):
    global search_cursor
    needle=text.encode('utf8');deadline=time.monotonic()+10
    while True:
        found=transcript.find(needle,search_cursor)
        if found>=0:
            search_cursor=found+len(needle);return
        if time.monotonic()>deadline or child.poll() is not None:
            raise RuntimeError(f'PTY did not reach expected fresh interaction (child={child.poll()}, output_bytes={len(stdout)}, diagnostic_bytes={len(stderr)}, synthetic_diagnostic_tail={bytes(stderr[-200:])!r})')
        pump()
os.set_blocking(in_master,False)
try:
    for action in spec['actions']:
        if 'pause_stderr' in action: pause_stderr=action['pause_stderr']
        if 'pause_stdout' in action: pause_stdout=action['pause_stdout']
        if 'wait' in action: wait_for(action['wait'])
        if 'wait_file' in action:
            ready_deadline=time.monotonic()+10
            while not Path(action['wait_file']).exists():
                if time.monotonic()>ready_deadline or child.poll() is not None: raise RuntimeError('Synthetic provider did not acknowledge expected request')
                pump()

        if 'write' in action or 'bytes' in action:
            remaining=memoryview(action['write'].encode() if 'write' in action else bytes(action['bytes']))
            write_deadline=time.monotonic()+15
            while remaining:
                if child.poll() is not None and action.get('allow_input_close'): break
                if time.monotonic()>write_deadline or child.poll() is not None: raise RuntimeError('PTY input did not finish writing')
                pump(0)
                if not select.select([], [in_master], [], 0.05)[1]: continue
                try: written=os.write(in_master,remaining[:16384])
                except BlockingIOError: continue
                except OSError as error:
                    if error.errno==errno.EIO and action.get('allow_input_close'): break
                    raise
                remaining=remaining[written:]
        if 'signal' in action: child.send_signal(action['signal'])
    deadline=time.monotonic()+12
    while child.poll() is None or active:
        if time.monotonic()>deadline: raise RuntimeError(f'PTY session exceeded deadline (child={child.poll()}, output_bytes={len(stdout)}, diagnostic_bytes={len(stderr)}, last_action_keys={list(spec["actions"][-1])!r})')
        pump()
    print(json.dumps({'code':child.returncode,'stdout':stdout.decode('utf8','replace').replace('\r\n','\n'),'stderr':stderr.decode('utf8','replace').replace('\r\n','\n')}))
finally:
    if child.poll() is None: child.kill(); child.wait()
    if child.stderr is not None: child.stderr.close()
    if err_master is not None: os.close(err_master)
    os.close(in_master);os.close(out_master)
