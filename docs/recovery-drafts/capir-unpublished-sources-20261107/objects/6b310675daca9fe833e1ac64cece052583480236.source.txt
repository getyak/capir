const respond=(res,value)=>{res.writeHead(200,{'content-type':'application/json'});res.end(JSON.stringify(value));};
const metadata=(id='pinned-model',extra={})=>({type:'model',id,max_input_tokens:1000000,max_tokens:128000,...extra});
function answer(res,body,model=body.model) {
  let text=anthropic('answer','end_turn',body.stream).replaceAll('"model":"fixture"',JSON.stringify({model}).slice(1,-1));
  res.writeHead(200,{'content-type':body.stream?'text/event-stream':'application/json'});res.end(text);
}
async function configured(t,handle) {
  const f=await fixture(t),p=await provider(t,(req,res,b)=>{
    if(req.url === '/prefix/v1/messages/count_tokens'){respond(res,{input_tokens:100});return;}
    return handle(req,res,b);
  });
  const added=await cli(['models','add','work','--provider','anthropic','--base-url',p.url,'--model','alias','--api-key-env','CAPIR_SYNTHETIC_KEY','--default'],f.env);
  assert.equal(added.code,0);
  return {...f,...p,env:{...f.env,CAPIR_SYNTHETIC_KEY:'synthetic-only-secret-12345'}};
}

test('SIGINT during metadata cancels process and never advances to generation',{timeout:15000},async t=>{
  let entered;const ready=new Promise(r=>entered=r);const f=await configured(t,(_,res)=>{res.writeHead(200,{'content-type':'application/json'});res.flushHeaders();entered();});
  const child=spawn(process.execPath,[binary,'ask','x','--json'],{env:f.env,stdio:['pipe','pipe','pipe']});t.after(()=>child.kill());child.stdin.end();let out='';child.stdout.on('data',x=>out+=x);child.stderr.resume();const closed=new Promise(r=>child.once('close',r));await ready;child.kill('SIGINT');assert.equal(await closed,130);assert.equal(JSON.parse(out).error.code,'CAPIR_MODEL_CANCELLED');assert.equal(generations(f).length,0);
});
function pty(env,actions,args=[]){return new Promise((resolve,reject)=>{const child=spawn('python3',[fileURLToPath(new URL('./model-pty.py',import.meta.url))],{env,stdio:['pipe','pipe','pipe']});let out='',err='';child.stdout.on('data',x=>out+=x);child.stderr.on('data',x=>err+=x);child.once('error',reject);child.once('close',code=>code!==0?reject(new Error(err)):resolve(JSON.parse(out)));child.stdin.end(JSON.stringify({node:process.execPath,binary,actions,args}));});}
test('actual PTY resolves selected model once and keeps its pin across completed turns',async t=>{
  const f=await configured(t,(req,res,b)=>req.method==='GET'?respond(res,metadata()):answer(res,b));
  const r=await pty(f.env,[{wait:'capir> '},{write:'first\nsecond\n'},{wait:'capir> '},{wait:'capir> '},{write:'/model\n'},{wait:'capir> '},{write:'/exit\n'}]);assert.equal(r.code,0);assert.equal(operations(f).length,3);assert.equal(operations(f).filter(x=>x.method==='GET').length,1);assert.equal(generations(f).length,2);assert.equal(generations(f).every(x=>x.body.model==='pinned-model'&&x.body.max_tokens===128000),true);assert.deepEqual(generations(f)[1].body.messages.map(x=>x.content),['first','answer','second']);assert.match(r.stderr,/resolved model pinned-model/);
});
