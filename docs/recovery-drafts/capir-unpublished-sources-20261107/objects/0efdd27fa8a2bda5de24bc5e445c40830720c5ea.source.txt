/**
 * Durable conversation queue.
 *
 * Split by responsibility:
 * - conversationQueueState: storage, claims, leases, fencing, snapshots.
 * - conversationQueueAdmission: admission and client queue mutations.
 * - conversationQueueCompletion: governed lineage + canonical Session write.
 * - conversationQueueRunner: bounded worker and recovery.
 * - conversationQueueSweep: retention and revocation cascade.
 *
 * This barrel also owns the incomplete-sync queue fence boundary: while a
 * Session is visibly incomplete, admission, claim, continue, retry and
 * prioritize are refused so restored pending work can never run early. Stop,
 * withdrawal, edit, source revocation and deletion stay usable.
 */

async function assertQueueWorkAdmissible(
  pool: Pool,
  accountId: string,
  sessionId: string,
): Promise<void> {
  await inTransaction(pool, async (client: PoolClient) => {
    await lockConversationQueueSession(client, sessionId);
    if (!(await conversationQueueSyncComplete(client, accountId, sessionId))) {
      throw new ApiError(
        409,
        "CONVERSATION_QUEUE_SYNC_INCOMPLETE",
        "This conversation is still synchronizing its history. Queued work resumes when the sync completes.",
      );
    }
  });
}

/** Admission is fenced while a Session is visibly incomplete: a partial
 * canonical history accepts no new queued work until its transfer finalizes. */
export async function admitConversationQueueEntry(
  pool: Pool,
  auth: AuthContext,
  request: ConversationQueueAdmitRequest,
): Promise<{ response: ConversationQueueAdmitResponse; replayed: boolean }> {
  await assertQueueWorkAdmissible(pool, auth.accountId, request.session_id);
  return admitConversationQueueEntryImpl(pool, auth, request);
}

/** Continue, retry and prioritize move work toward running; they share the
 * incomplete-sync fence. Stop, edit and withdrawal remain usable. */
export async function mutateConversationQueueEntry(
  pool: Pool,
  auth: AuthContext,
  sessionId: string,
  request: ConversationQueueMutationRequest,
): Promise<Awaited<ReturnType<typeof mutateConversationQueueEntryImpl>>> {
  if (["continue", "retry", "prioritize"].includes(request.kind)) {
    await assertQueueWorkAdmissible(pool, auth.accountId, sessionId);
  }
  return mutateConversationQueueEntryImpl(pool, auth, sessionId, request);
}
