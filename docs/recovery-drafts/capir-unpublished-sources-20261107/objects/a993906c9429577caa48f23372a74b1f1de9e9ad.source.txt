async function owner(): Promise<{ auth: AuthContext; token: string }> {
  const token = secret();
  const auth: AuthContext = { accountId: randomUUID(), accountSlug: `legacy-b-${randomUUID()}`, userId: randomUUID(),
    userEmail: `${randomUUID()}@synthetic.invalid`, userKind: "simulated_human", sessionId: randomUUID() };
  await inTransaction(pool!, async client => {
    await client.query("INSERT INTO accounts(id,slug,name) VALUES($1,$2,'Synthetic B owner')", [auth.accountId, auth.accountSlug]);
    await client.query("INSERT INTO users(id,account_id,email,display_name,kind) VALUES($1,$2,$3,'Synthetic B owner','simulated_human')", [auth.userId, auth.accountId, auth.userEmail]);
    await client.query("INSERT INTO sessions(id,account_id,user_id,token_hash,client_label,expires_at) VALUES($1,$2,$3,$4,'synthetic B',clock_timestamp()+interval '2 hours')",
      [auth.sessionId, auth.accountId, auth.userId, hash(token)]);
  });
  return { auth, token };
}
