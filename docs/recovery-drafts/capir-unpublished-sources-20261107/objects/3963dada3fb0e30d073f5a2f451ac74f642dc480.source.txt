async function makeAuth(label: string): Promise<AuthContext> {
  const accountId = randomUUID();
  const userId = randomUUID();
  const accountSlug = `lineage-${label}-${accountId.slice(0, 8)}`;
  await pool!.query(
    "INSERT INTO accounts(id,slug,name) VALUES($1,$2,$3)",
    [accountId, accountSlug, `Lineage ${label}`],
  );
  await pool!.query(
    "INSERT INTO users(id,account_id,email,display_name,kind) VALUES($1,$2,$3,$4,'simulated_human')",
    [userId, accountId, `${accountSlug}@synthetic.local`, "Synthetic Owner"],
  );
  return {
    accountId,
    accountSlug,
    userId,
    userEmail: `${accountSlug}@synthetic.local`,
    userKind: "simulated_human",
    sessionId: randomUUID(),
  };
}

async function removeAccount(accountId: string): Promise<void> {
  const tables = (
    await pool!.query<{ table_name: string }>(
      "SELECT table_name FROM lab_test_workspace_table_manifest WHERE scope='account' ORDER BY table_name",
    )
  ).rows.map((row) => row.table_name);
  const targets = [...tables, "sessions"];
  await pool!.query(
    `WITH ${targets.map((table, index) => `d${index} AS (DELETE FROM "${table}" WHERE account_id=$1 RETURNING 1)`).join(",")}
     SELECT ${targets.map((_, index) => `(SELECT count(*) FROM d${index})` ).join("+")} AS removed`,
    [accountId],
  );
  // Deleting accepted Memory retires its harness working context once more;
  // remove that final generation row before the account can leave.
  await pool!.query("DELETE FROM harness_source_generations WHERE account_id=$1", [accountId]);
  await pool!.query("DELETE FROM users WHERE account_id=$1", [accountId]);
  await pool!.query("DELETE FROM accounts WHERE id=$1", [accountId]);
}

/** Accept one self-scope statement through the real review/commit machinery. */
async function saveAcceptedMemory(
  auth: AuthContext,
  text: string,
): Promise<{ sourceSessionId: string; sourceMessageId: string; memoryItemID: string; version: number }> {
  const sourceSessionId = randomUUID();
  const sourceMessageId = randomUUID();
  await insertSourceSession(auth, sourceSessionId, sourceMessageId, text);
  const authority = await resolveSessionSourceAuthority(
    pool!, auth, sourceSessionId, sourceMessageId,
  );
  const request: MemoryProposalStageRequest = {
    idempotency_key: randomUUID(),
    surface: "chat",
    session_id: sourceSessionId,
    source_task_id: randomUUID(),
    source_message_id: sourceMessageId,
    person_id: null,
    relationship_context_id: null,
    contact_decision: "none",
    identity_clue: null,
    new_contact: null,
    proposer: { kind: "human", name: "lineage-test", version: "1" },
    items: [candidate({ display_text: text, source_excerpt: text })],
  };
  const staged = await inTransaction(pool!, (client) =>
    stageMemoryProposal(client, auth, request, authority),
  );
  const opened = await openMemoryReview(pool!, auth, staged!.proposal.proposal_id, {
    purpose: "chat",
    person_id: null,
    relationship_context_id: null,
  });
  const body: MemoryCommitRequest = {
    idempotency_key: randomUUID(),
    expected_proposal_revision: opened.review.proposal_revision,
    contact_decision: "none",
    selected_item_ids: opened.review.items.map((item) => item.id),
    edited_text: {},
    item_decisions: {},
    expected_item_versions: {},
    new_contact: null,
    reason: "lineage test commit",
  };
  await commitMemoryReview(pool!, auth, opened.review.review_scope_id, opened.review_credential!, body);
  const saved = await searchSavedMemories(pool!, auth, { terms: [] });
  const item = saved.items.find((entry) => entry.display_text === text);
  expect(item, "the accepted statement must be discoverable").toBeDefined();
  return { sourceSessionId, sourceMessageId, memoryItemID: item!.id, version: item!.version };
}

/** Harness-like scripted provider: it executes the real supplemental tool. */
class SavedMemoryScriptedProvider implements AgentProvider {
  readonly id = "deterministic";
  readonly model = "talent-signal-scripted-v1";
  readonly sdkVersion = "deterministic-provider.v1";
  readonly inputCapabilities = {
    text: true,
    image: true,
    imageUnderstanding: false,
  } as const;

  constructor(
    private readonly body: (found: string) => string,
  ) {}

  async run(
    request: AgentProviderRequest,
    _invokeTool: unknown,
    signal: AbortSignal,
  ): Promise<AgentProviderResult> {
    const tool = request.supplementalTools?.find(
      (entry) => entry.name === "search_saved_memory",
    );
    expect(tool, "the run must expose the saved-Memory search tool").toBeDefined();
    const receipt = await tool!.execute({ terms: ["harborsentinel9"] }, signal);
    const content = receipt.content[0]!;
    const page = JSON.parse(
      content.type === "text" ? content.text : "{}",
    ) as { items: Array<{ display_text: string }> };
    return {
      structuredOutput: {
        outcome: "reply",
        title: "Saved record",
        body: this.body(page.items[0]?.display_text ?? "nothing"),
      },
      inputTokens: 1,
      outputTokens: 1,
      estimatedUsd: 0,
      turns: 1,
      permissionDenials: [],
    };
  }
}

function searchProvider(): AgentProvider {
  return new SavedMemoryScriptedProvider(
    (found) => `Derived answer quoting ${found}`,
  );
}

function plainProvider(): ScriptedAgentProvider {
  return new ScriptedAgentProvider([], {
    outcome: "reply",
    title: "Plain",
    body: "Follow-up answer that only reuses earlier context.",
  });
}

async function dependencyRows(accountId: string, taskID: string) {
  return (
    await pool!.query<{
      memory_item_id: string;
      memory_item_version: number;
      owner_user_id: string;
    }>(
      `SELECT memory_item_id, memory_item_version, owner_user_id
       FROM agent_task_memory_dependencies
       WHERE account_id=$1 AND task_id=lower($2::text) ORDER BY memory_item_id`,
      [accountId, taskID],
    )
  ).rows;
}

async function retractedTasks(accountId: string): Promise<string[]> {
  return (
    await pool!.query<{ task_id: string }>(
      "SELECT task_id FROM agent_session_retracted_tasks WHERE account_id=$1",
      [accountId],
    )
  ).rows.map((row) => row.task_id);
}

function sleep(ms: number): Promise<void> {
  return new Promise((resolve) => setTimeout(resolve, ms));
}

function deferred<T = void>(): { promise: Promise<T>; resolve: (value: T) => void } {
  let resolve!: (value: T) => void;
  const promise = new Promise<T>((settle) => { resolve = settle; });
  return { promise, resolve };
}

/** A test-only barrier key whose advisory lock pauses one exact statement. */
async function newBarrierKey(): Promise<string> {
  return String(
    (
      await pool!.query<{ key: string }>(
        "SELECT ('x'||substr(md5($1),1,16))::bit(64)::bigint AS key",
        [randomUUID()],
      )
    ).rows[0]!.key,
  );
}

/**
 * Pause the real final Session mutation exactly at its Session UPDATE, i.e.
 * after its redaction/currentness reads, while its transaction holds the
 * saved-Memory lineage fence. Scoped to one Session and dropped afterwards.
 */
async function installSessionWritePause(sessionId: string, barrierKey: string): Promise<() => Promise<void>> {
  const name = `lineage_test_pause_${randomUUID().replace(/-/g, "")}`;
  await pool!.query(
    `CREATE FUNCTION ${name}() RETURNS trigger LANGUAGE plpgsql AS $$
     BEGIN
       -- The pause must outlive the mutation's own bounded lock waits.
       PERFORM set_config('lock_timeout', '0', true);
       PERFORM pg_advisory_xact_lock(${barrierKey});
       RETURN NEW;
     END $$`,
  );
  await pool!.query(
    `CREATE TRIGGER ${name} BEFORE UPDATE ON agent_sessions
     FOR EACH ROW WHEN (NEW.id = '${sessionId}'::uuid) EXECUTE FUNCTION ${name}()`,
  );
  return async () => {
    await pool!.query(`DROP TRIGGER IF EXISTS ${name} ON agent_sessions`);
    await pool!.query(`DROP FUNCTION IF EXISTS ${name}()`);
  };
}

describe.skipIf(!pool)("Saved Memory chat lineage", () => {
  it("saves the queue result, then deleting the source scrubs the replay body and refuses replay", async () => {
    const auth = await makeAuth("replay");
    try {
      const memory = await saveAcceptedMemory(
        auth, "Mira is preparing HarborSentinel9 documentation.",
      );
      const { sessionId } = await seedConversation(auth);
      const claimed = (await admitAndClaim(auth, sessionId, "What is the current documentation work?"))!;
      expect(claimed).not.toBeNull();

      // The run discovers saved Memory and persists durable dependencies
      // before the discovered text can escape to the model.
      const execution = await runTask({
        auth, sessionId, taskID: claimed.runId,
        objective: "What is the current documentation work?", provider: searchProvider(),
      });
      expect(execution.body.blocks[0]!.body).toContain("HarborSentinel9");
      const deps = await dependencyRows(auth.accountId, claimed.runId);
      expect(deps).toEqual([
        {
          memory_item_id: memory.memoryItemID,
          memory_item_version: memory.version,
          owner_user_id: auth.userId,
        },
      ]);
      // Lineage is identity and version only, never raw source content.
      expect(JSON.stringify(deps)).not.toContain("HarborSentinel9");

      const result = executionResult(execution);
      await recordConversationQueueResult(pool!, claimed, result);

      // Explicit source deletion before the crash replay.
      await mutateAgentSession(
        pool!, auth, memory.sourceSessionId,
        { expected_revision: 1, idempotency_key: randomUUID() }, true,
      );

      // The stored replay body is scrubbed and the task is retracted.
      const stored = await readStoredConversationQueueResult(pool!, auth.accountId, claimed.entryId);
      expect(stored).toMatchObject({ state: "withdrawn", taskID: claimed.runId.toLowerCase() });
      const row = await pool!.query<{ result: Record<string, unknown> }>(
        "SELECT result FROM conversation_queue_entries WHERE account_id=$1 AND id=$2",
        [auth.accountId, claimed.entryId],
      );
      expect(row.rows[0]!.result).toMatchObject({ source_context_unavailable: true });
      expect(JSON.stringify(row.rows[0]!.result)).not.toContain("HarborSentinel9");
      expect(await retractedTasks(auth.accountId)).toContain(claimed.runId.toLowerCase());

      // Even a caller still holding the original result cannot replay it: the
      // durable rows, not a process-local map, gate the canonical write.
      await expect(
        persistConversationQueueCompletion(pool!, auth, {
          fence: claimed, sessionId,
          messageId: claimed.messageId, objective: claimed.objective,
          acceptedAt: claimed.acceptedAt, images: [], result,
        }),
      ).rejects.toMatchObject({ statusCode: 410 });
      expect((await getAgentSession(pool!, auth, sessionId)).payload!.turns).toEqual([]);
    } finally {
      await removeAccount(auth.accountId);
    }
  });

  it("retracts a completed derived answer and future Session history when its source is withdrawn", async () => {
    const auth = await makeAuth("history");
    try {
      const memory = await saveAcceptedMemory(
        auth, "Mira is preparing HarborSentinel9 documentation.",
      );
      const { sessionId } = await seedConversation(auth);
      const claimed = (await admitAndClaim(auth, sessionId, "What is the current documentation work?"))!;
      const execution = await runTask({
        auth, sessionId, taskID: claimed.runId,
        objective: "What is the current documentation work?", provider: searchProvider(),
      });
      const result = executionResult(execution);
      await recordConversationQueueResult(pool!, claimed, result);
      await persistConversationQueueCompletion(pool!, auth, {
        fence: claimed, sessionId,
        messageId: claimed.messageId, objective: claimed.objective,
        acceptedAt: claimed.acceptedAt, images: [], result,
      });

      const saved = (await getAgentSession(pool!, auth, sessionId)).payload!;
      expect(saved.turns).toHaveLength(1);
      expect(JSON.stringify(saved.turns[0]!.response)).toContain("HarborSentinel9");
      const history = await readAgentSessionConversation(pool!, auth, sessionId, {
        personId: null, relationshipContextId: null,
      });
      expect(
        history.messages.some((message) => message.text.includes("HarborSentinel9")),
      ).toBe(true);

      // The owner explicitly withdraws the accepted statement.
      await mutateMemoryItem(pool!, auth, memory.memoryItemID, {
        operation: "delete",
        idempotency_key: randomUUID(),
        expected_version: memory.version,
        reason: "withdrawn for lineage test",
      });

      // Completed derived text is retracted from canonical history.
      const after = (await getAgentSession(pool!, auth, sessionId)).payload!;
      expect(JSON.stringify(after)).not.toContain("HarborSentinel9");
      const historyAfter = await readAgentSessionConversation(pool!, auth, sessionId, {
        personId: null, relationshipContextId: null,
      });
      expect(
        historyAfter.messages.some((message) => message.text.includes("HarborSentinel9")),
      ).toBe(false);
      await expect(
        assertSessionChatSourcesAvailable(pool!, auth, claimed.runId),
      ).rejects.toMatchObject({ statusCode: 410 });
      // A later request cannot rebuild Session history from the withdrawn text.
      await expect(
        runTask({
          auth, sessionId, taskID: randomUUID(),
          objective: "Continue the documentation discussion.", provider: plainProvider(),
        }),
      ).rejects.toMatchObject({ code: "AGENT_SESSION_CONTEXT_UNAVAILABLE" });
    } finally {
      await removeAccount(auth.accountId);
    }
  });

  it("fails closed on a changed version before any final use or replay", async () => {
    const auth = await makeAuth("version");
    try {
      const memory = await saveAcceptedMemory(
        auth, "Mira is preparing HarborSentinel9 documentation.",
      );
      const { sessionId } = await seedConversation(auth);
      const claimed = (await admitAndClaim(auth, sessionId, "What is the current documentation work?"))!;
      await runTask({
        auth, sessionId, taskID: claimed.runId,
        objective: "What is the current documentation work?", provider: searchProvider(),
      });

      await pool!.query(
        "UPDATE memory_items SET version = version + 1 WHERE account_id = $1 AND id = $2",
        [auth.accountId, memory.memoryItemID],
      );

      await expect(
        assertSavedMemoriesCurrent(pool!, auth, new Map([[memory.memoryItemID, memory.version]])),
      ).rejects.toMatchObject({ code: "MEMORY_SEARCH_SOURCE_NOT_CURRENT" });
      await expect(
        assertSavedMemoryDependenciesCurrent(pool!, auth, claimed.runId),
      ).rejects.toMatchObject({ statusCode: 410 });
      await expect(
        assertSessionChatSourcesAvailable(pool!, auth, claimed.runId),
      ).rejects.toMatchObject({ statusCode: 410 });
      // A re-read at the new version cannot silently overwrite the exposed one.
      await expect(
        recordSavedMemoryDependencies(pool!, auth, claimed.runId, [
          { memoryItemID: memory.memoryItemID, version: memory.version + 1 },
        ]),
      ).rejects.toMatchObject({ code: "MEMORY_SEARCH_SOURCE_NOT_CURRENT" });
    } finally {
      await removeAccount(auth.accountId);
    }
  });

  it("denies a same-account other user the derived answer and its lineage", async () => {
    const auth = await makeAuth("owner");
    try {
      await saveAcceptedMemory(auth, "Mira is preparing HarborSentinel9 documentation.");
      const { sessionId } = await seedConversation(auth);
      const claimed = (await admitAndClaim(auth, sessionId, "What is the current documentation work?"))!;
      await runTask({
        auth, sessionId, taskID: claimed.runId,
        objective: "What is the current documentation work?", provider: searchProvider(),
      });

      const otherUserId = randomUUID();
      await pool!.query(
        "INSERT INTO users(id,account_id,email,display_name,kind) VALUES($1,$2,$3,'Other','simulated_human')",
        [otherUserId, auth.accountId, `${otherUserId}@synthetic.local`],
      );
      const other: AuthContext = { ...auth, userId: otherUserId, userEmail: `${otherUserId}@synthetic.local` };

      expect((await searchSavedMemories(pool!, other, { terms: ["harborsentinel9"] })).items).toEqual([]);
      await expect(
        assertSavedMemoryDependenciesCurrent(pool!, other, claimed.runId),
      ).rejects.toMatchObject({ statusCode: 404 });
      await expect(
        assertSessionChatSourcesAvailable(pool!, other, claimed.runId),
      ).rejects.toMatchObject({ statusCode: 404 });
    } finally {
      await removeAccount(auth.accountId);
    }
  });

  it("keeps a safe no-Memory task across unrelated saved-Memory withdrawal", async () => {
    const auth = await makeAuth("safe");
    try {
      const { sessionId } = await seedConversation(auth);
      const claimed = (await admitAndClaim(auth, sessionId, "Say hello."))!;
      const execution = await runTask({
        auth, sessionId, taskID: claimed.runId,
        objective: "Say hello.", provider: plainProvider(),
      });
      const result = executionResult(execution);
      await recordConversationQueueResult(pool!, claimed, result);
      await persistConversationQueueCompletion(pool!, auth, {
        fence: claimed, sessionId,
        messageId: claimed.messageId, objective: claimed.objective,
        acceptedAt: claimed.acceptedAt, images: [], result,
      });
      expect(await dependencyRows(auth.accountId, claimed.runId)).toEqual([]);

      // An unrelated accepted statement is created and withdrawn afterwards.
      const memory = await saveAcceptedMemory(
        auth, "Mira is preparing HarborSentinel9 documentation.",
      );
      await mutateMemoryItem(pool!, auth, memory.memoryItemID, {
        operation: "delete",
        idempotency_key: randomUUID(),
        expected_version: memory.version,
        reason: "unrelated withdrawal",
      });

      await expect(
        assertSessionChatSourcesAvailable(pool!, auth, claimed.runId),
      ).resolves.toBeUndefined();
      const history = await readAgentSessionConversation(pool!, auth, sessionId, {
        personId: null, relationshipContextId: null,
      });
      expect(
        history.messages.some((message) => message.text.includes("Follow-up answer")),
      ).toBe(true);

      await recordConversationQueueResult(pool!, first, firstResult);
      await persistConversationQueueCompletion(pool!, auth, {
        fence: first, sessionId,
        messageId: first.messageId, objective: first.objective,
        acceptedAt: first.acceptedAt, images: [], result: firstResult,
      });
      await finalizeConversationQueueEntry(pool!, { fence: first, status: "completed" });

      const second = (await admitAndClaim(auth, sessionId, "Continue the documentation discussion."))!;
      const started = deferred();
      const gate = deferred();
      const executionPromise = executeUnscopedChatTask({
        request: {
          idempotency_key: randomUUID(),
          objective: "Continue the documentation discussion.",
          session_id: sessionId,
          message_id: randomUUID(),
        },
        taskID: second.runId,
        provider: legacyProvider(async (request) => {
          started.resolve();
          // The reused derived answer reached the ordinary answer branch, and
          // its inherited dependencies were durable before it did.
          expect(JSON.stringify(request.conversation_history)).toContain("HarborSentinel9");
          expect(await dependencyRows(auth.accountId, second.runId)).toEqual(
            await dependencyRows(auth.accountId, first.runId),
          );
          await gate.promise;
          return legacyAnswer("Legacy answer quoting HarborSentinel9");
        }),
        database: pool!,
        probePool: pool!,
        auth,
      });
      await started.promise;

      // The source withdraws while the legacy answer is still being produced.
      await mutateMemoryItem(pool!, auth, memory.memoryItemID, {
        operation: "delete",
        idempotency_key: randomUUID(),
        expected_version: memory.version,
        reason: "withdrawn during legacy answer",
      });
      gate.resolve();
      const execution = await executionPromise;
      expect(execution.body.blocks[0]!.body).toContain("HarborSentinel9");

      // The result write keeps only the truthful withdrawal marker: the
      // inherited rows make the body inadmissible even though this provider
      // never called a memory tool itself.
      await recordConversationQueueResult(pool!, second, executionResult(execution));
      const row = await pool!.query<{ result: Record<string, unknown> }>(
        "SELECT result FROM conversation_queue_entries WHERE account_id=$1 AND id=$2",
        [auth.accountId, second.entryId],
      );
      expect(row.rows[0]!.result).toMatchObject({ source_context_unavailable: true });
      expect(JSON.stringify(row.rows[0]!.result)).not.toContain("HarborSentinel9");
      expect(
        await readStoredConversationQueueResult(pool!, auth.accountId, second.entryId),
      ).toMatchObject({ state: "withdrawn" });
      await expect(
        persistConversationQueueCompletion(pool!, auth, {
          fence: second, sessionId,
          messageId: second.messageId, objective: second.objective,
          acceptedAt: second.acceptedAt, images: [], result: executionResult(execution),
        }),
      ).rejects.toMatchObject({ statusCode: 410 });
      const after = (await getAgentSession(pool!, auth, sessionId)).payload!;
      expect(after.turns).toHaveLength(1);
      expect(JSON.stringify(after)).not.toContain("HarborSentinel9");
    } finally {
      await removeAccount(auth.accountId);
    }
  }, 90_000);

  it("commits sync-path lineage durably before exposure and never holds the account fence across provider calls", async () => {
    const auth = await makeAuth("sync-durable");
    try {
      const memory = await saveAcceptedMemory(
        auth, "Mira is preparing HarborSentinel9 documentation.",
      );
      const { sessionId } = await seedConversation(auth);
      const first = (await admitAndClaim(auth, sessionId, "What is the current documentation work?"))!;
      const firstExecution = await runTask({
        auth, sessionId, taskID: first.runId,
        objective: "What is the current documentation work?", provider: searchProvider(),
      });
      const firstResult = executionResult(firstExecution);
      await recordConversationQueueResult(pool!, first, firstResult);
      await persistConversationQueueCompletion(pool!, auth, {
        fence: first, sessionId,
        messageId: first.messageId, objective: first.objective,
        acceptedAt: first.acceptedAt, images: [], result: firstResult,
      });
      await finalizeConversationQueueEntry(pool!, { fence: first, status: "completed" });

      // Exact sync wiring: the request transaction hands its PoolClient to the
      // adapter while an independent pool owns durable lineage writes.
      const syncTaskID = randomUUID();
      await inTransaction(pool!, (client) => executeUnscopedChatTask({
        request: {
          idempotency_key: randomUUID(),
          objective: "Continue the documentation discussion.",
          session_id: sessionId,
          message_id: randomUUID(),
        },
        taskID: syncTaskID,
        provider: legacyProvider(async (request) => {
          // From an independent connection the inherited rows are already
          // committed while the request transaction is still open.
          expect(JSON.stringify(request.conversation_history)).toContain("HarborSentinel9");
          expect(await dependencyRows(auth.accountId, syncTaskID)).toEqual(
            await dependencyRows(auth.accountId, first.runId),
          );
          // No account fence may stay held across a provider call.
          const probe = await pool!.connect();
          try {
            await probe.query("BEGIN");
            const free = (
              await probe.query<{ ok: boolean }>(
                "SELECT pg_try_advisory_xact_lock(hashtextextended('talent_signal.saved_memory_lineage:' || $1, 0)) AS ok",
                [auth.accountId],
              )
            ).rows[0]!.ok;
            await probe.query("COMMIT");
            expect(free, "no account fence may stay held across provider calls").toBe(true);
          } finally {
            probe.release();
          }
          return legacyAnswer("Legacy answer quoting HarborSentinel9");
        }),
        database: client,
        probePool: pool!,
        auth,
      }));

      // A source withdrawal during a sync provider call fails the request
      // closed: no idempotent body, no Session turn, no stored raw text.
      const gate = deferred();
      const started = deferred();
      const request = {
        idempotency_key: randomUUID(),
        objective: "Continue the documentation discussion.",
        session_id: sessionId,
        message_id: randomUUID(),
      };
      const run = createUnscopedChatTask(pool!, auth, request, legacyProvider(async () => {
        started.resolve();
        await gate.promise;
        return legacyAnswer("Withdrawn answer quoting HarborSentinel9");
      }));
      await started.promise;
      await mutateMemoryItem(pool!, auth, memory.memoryItemID, {
        operation: "delete",
        idempotency_key: randomUUID(),
        expected_version: memory.version,
        reason: "withdrawn during sync answer",
      });
      gate.resolve();
      await expect(run).rejects.toMatchObject({ statusCode: 410 });
      const idempotent = await pool!.query<{ n: string }>(
        "SELECT count(*) AS n FROM idempotency_records WHERE account_id=$1 AND idempotency_key=$2",
        [auth.accountId, request.idempotency_key],
      );
      expect(Number(idempotent.rows[0]!.n), "no idempotent body may survive the withdrawal").toBe(0);
      const after = (await getAgentSession(pool!, auth, sessionId)).payload!;
      expect(JSON.stringify(after)).not.toContain("HarborSentinel9");
    } finally {
      await removeAccount(auth.accountId);
    }
  }, 90_000);
