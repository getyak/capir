import type { Metadata } from "next";
import Link from "next/link";

import { SiteFooter } from "@/components/site-footer";
import { SiteHeader } from "@/components/site-header";
import { getMarketingLocale } from "@/lib/server/marketing-locale";
import { siteConfig } from "@/lib/site";
import { SupportEmailEntry } from "@/components/support-email-entry";

import {
  MACOS_DISTRIBUTION_DOC_HREF,
  downloadSurfaces,
} from "./release";

export const metadata: Metadata = {
  title: "下载 capri",
  description:
    "下载 capri 桌面与移动端：macOS 公开下载，iOS 通过受邀预览提供，可申请使用。",
  alternates: {
    canonical: "/download",
  },
};

export default async function DownloadPage() {
  const locale = await getMarketingLocale();
  const en = locale === "en";
  const surfaces = downloadSurfaces(locale);
  return (
    <>
      <SiteHeader />
      <main id="main-content" className="prose-page" lang={locale} tabIndex={-1}>
        <article className="shell prose-page__inner">
          <header>
            <p className="eyebrow">{en ? "Download" : "下载"}</p>
            <h1>{en ? "capri, on your devices." : "在你的设备上使用 capri。"}</h1>
            <p>
              {en ? "Each platform releases independently. Downloads point to the existing releases, which may still use the Talent Signal name. The website name does not mean a renamed app has shipped. Update notices appear only after a verified check." : "各端独立发布。安装入口指向现有发布位置；已有版本可能仍显示 Talent Signal，官网更名不代表新品牌安装包已经发布。只有取得可靠更新结果时，应用才会提示可更新。"}
            </p>
          </header>

          {surfaces.map((surface) => (
            <section key={surface.platform}>
              <h2>{surface.title}</h2>
              <p>
                <strong>{surface.status}</strong>
              </p>
              <p>{surface.detail}</p>
              <p>
                {surface.installHref ? (
                  <a href={surface.installHref}>
                    {en ? `${surface.title} public download (GitHub Releases) ↗` : `${surface.title} 公开下载（GitHub Releases）↗`}
                  </a>
                ) : (
                  <SupportEmailEntry subject={en ? "Request capri access" : "申请使用 capri"}>
                    {en ? `Request ${surface.title} access by email ↗` : `申请使用 ${surface.title} 版（邮件）↗`}
                  </SupportEmailEntry>
                )}
              </p>
              {surface.platform === "macos" ? (
                <p>
                  {en ? "Installation, checksum and signing instructions: " : "安装步骤、校验与签名说明见 "}
                  <a href={MACOS_DISTRIBUTION_DOC_HREF}>{en ? "macOS distribution guide ↗" : "macOS 下载与分发文档 ↗"}</a>{en ? ". " : "。"}
                  {en ? "Installed apps can check for updates in This Mac settings." : "已安装的应用可在「此 Mac 设置」中检查更新。"}
                </p>
              ) : null}
              {surface.platform === "ios" ? (
                <p>
                  {en ? "Install through TestFlight after receiving an invitation. The invitation supplies the install link; there is no public iOS install link here." : "收到邀请后通过 TestFlight 安装；安装地址由邀请提供，这里不发布公开安装链接。"}
                </p>
              ) : null}
            </section>
          ))}

          <section>
            <h2>{en ? "Help and support" : "帮助与支持"}</h2>
            <p>
              {en ? "For installation or usage help, open Account → Help and support in the workspace to check service status, or email " : "遇到安装或使用问题时，可在工作区「账号 → 帮助与支持」查看系统检测，或直接来信 "}
              <SupportEmailEntry>{siteConfig.email}</SupportEmailEntry>
              {en ? "; or " : "；也可以 "}<Link href="/login">{en ? "sign in to the workspace" : "登录工作区"}</Link>{en ? " to view settings and diagnostics." : " 后查看设置与诊断。"}
            </p>
          </section>
        </article>
      </main>
      <SiteFooter />
    </>
  );
}
