import { describe, expect, it } from "vitest";
import {
  type AgentSessionPayload,
  AgentSessionDeltaOperationError,
  agentSessionDeltaFrameRequestBytes,
  agentSessionIntentDigest,
  applyAgentSessionDeltaOperations,
  canonicalSessionIntent,
  decodeJsonPointer,
  frameSessionIntent,
  planAgentSessionDelta,
  planAgentSessionDeltaFrames,
  sessionIntentDigest,
  sessionIntentFramingFixtures,
  sha256Hex,
  verifyScreenshotInheritanceStep,
  SCREENSHOT_ADMISSION_PATHS,
  type AgentSessionDeltaOperation,
} from "@talent-signal/contracts";

const UUID_A = "7BA9A4B2-8E31-4C0A-9D2E-55AA7C4E1B10";
const UUID_B = "0D90B077-2BD1-4C13-82CA-8F4DD366DB1B";

function turn(id: string, objective: string): AgentSessionPayload["turns"][number] {
  return {
    id,
    objective,
    createdAt: "2026-10-08T03:07:24.123Z",
    response: {
      contractVersion: "talent-signal.v1",
      taskID: id,
      contextManifestID: "none-unbound-conversation",
      knowledgeSnapshotID: "none-unbound-conversation",
      disposition: "answered",
      createdAt: "2026-10-08T03:07:24.200Z",
      savedBlocks: [
        {
          id: `block-${id}`,
          kind: "answer",
          title: "Answer",
          body: "synthetic",
          status: "informational",
          citation_dependency_ids: [],
          requires_user_decision: false,
        },
      ],
    },
  };
}

function payload(id: string = UUID_A): AgentSessionPayload {
  return {
    id,
    scopeKind: "unresolved_intent",
    personDisplayLabel: "New session",
    contextDisplayLabel: "Conversation",
    title: "Synthetic conversation",
    turns: [turn(UUID_B, "first")],
    updatedAt: "2026-10-08T03:07:24.000Z",
    isUnread: false,
  };
}

describe("session intent framing", () => {
  it("matches node crypto for the sha256 primitive", () => {
    expect(sha256Hex(new Uint8Array([]))).toBe(
      "e3b0c44298fc1c149afbf4c8996fb92427ae41e4649b934ca495991b7852b855",
    );
    expect(sha256Hex(new TextEncoder().encode("abc"))).toBe(
      "ba7816bf8f01cfea414140de5dae2223b00361a396177a9cb410ff61f20015ad",
    );
    // Block-boundary lengths exercise the padding of the dependency-free hash.
    expect(sha256Hex(new Uint8Array(55).fill(0x61))).toHaveLength(64);
    expect(sha256Hex(new Uint8Array(64).fill(0x61))).toHaveLength(64);
    expect(sha256Hex(new Uint8Array(1000).fill(0x61))).toHaveLength(64);
  });

  it("reproduces the shared cross-language fixtures exactly", () => {
    for (const fixture of sessionIntentFramingFixtures) {
      expect(
        sessionIntentDigest(fixture.value),
        fixture.name,
      ).toBe(fixture.digest);
    }
  });

  it("frames -0 as zero and rejects unframeable values", () => {
    expect(sessionIntentDigest({ a: -0 })).toBe(sessionIntentDigest({ a: 0 }));
    expect(() => sessionIntentDigest({ a: Number.NaN })).toThrow(
      /finite JSON numbers/,
    );
    expect(() => sessionIntentDigest({ a: Number.POSITIVE_INFINITY })).toThrow();
    expect(() => sessionIntentDigest("lone \ud800 surrogate")).toThrow(
      /unpaired surrogate/,
    );
    expect(() => sessionIntentDigest({ "ob\u0000ject": 1 })).not.toThrow();
    expect(() => sessionIntentDigest(undefined)).toThrow(/Only JSON values/);
  });

  it("orders object keys by UTF-16 code units regardless of insertion order", () => {
    const left = { z: 1, "\u00e9": 2, a: 3, "\uffff": 4 };
    const right = { "\uffff": 4, a: 3, "\u00e9": 2, z: 1 };
    expect(frameSessionIntent(left)).toEqual(frameSessionIntent(right));
  });
});

describe("restricted delta operations", () => {
  it("decodes JSON pointers with strict escapes", () => {
    expect(decodeJsonPointer("/turns/0~1x~0y")).toEqual(["turns", "0/x~y"]);
    expect(() => decodeJsonPointer("turns/0")).toThrow(/bounded JSON pointer/);
    expect(() => decodeJsonPointer("/turns/~2")).toThrow(/invalid escape/);
    expect(() => decodeJsonPointer(`/${"x".repeat(600)}`)).toThrow();
  });

  it("appends new turns in order and never reorders history", () => {
    const before = payload();
    const after = structuredClone(before);
    after.turns.push(turn(UUID_A, "second"));
    const applied = applyAgentSessionDeltaOperations(before, [
      { op: "add", path: "/turns/1", value: after.turns[1] },
    ]) as AgentSessionPayload;
    expect(applied.turns.map((item) => item.id)).toEqual([
      UUID_B,
      UUID_A,
    ]);
    expect(
      () =>
        applyAgentSessionDeltaOperations(before, [
          { op: "add", path: "/turns/0", value: turn(UUID_A, "insert") },
        ]),
    ).toThrow(/append in order/);
  });

  it("rejects prototype keys in paths and values", () => {
    const before = payload();
    expect(() =>
      applyAgentSessionDeltaOperations(before, [
        { op: "replace", path: "/__proto__", value: {} },
      ]),
    ).toThrow(/prototype keys/);
    expect(() =>
      applyAgentSessionDeltaOperations(before, [
        {
          op: "add",
          path: "/turns/1",
          value: { ...turn(UUID_A, "x"), extra: JSON.parse('{"__proto__": {"polluted": true}}') },
        },
      ]),
    ).toThrow(/prototype keys/);
    expect(() =>
      applyAgentSessionDeltaOperations(before, [
        {
          op: "replace",
          path: "/title",
          value: JSON.parse('{"constructor": null}'),
        },
      ]),
    ).toThrow(/prototype keys/);
    expect(({} as Record<string, unknown>).polluted).toBeUndefined();
  });

  it("rejects absent parents, invalid indexes and non-JSON values", () => {
    const before = payload();
    expect(() =>
      applyAgentSessionDeltaOperations(before, [
        { op: "add", path: "/turns/0/feedback", value: "helpful" },
      ]),
    ).not.toThrow();
    const withFeedback = applyAgentSessionDeltaOperations(before, [
      { op: "add", path: "/turns/0/feedback", value: "helpful" },
    ]);
    expect(() =>
      applyAgentSessionDeltaOperations(withFeedback, [
        { op: "replace", path: "/turns/0/feedback", value: "unhelpful" },
      ]),
    ).not.toThrow();
    expect(() =>
      applyAgentSessionDeltaOperations(before, [
        { op: "replace", path: "/turns/0/feedbackUpdatedAt", value: "2026-10-08T03:07:24.000Z" },
      ]),
    ).toThrow(/existing member/);
    expect(() =>
      applyAgentSessionDeltaOperations(before, [
        { op: "replace", path: "/turns/9/feedback", value: "helpful" },
      ]),
    ).toThrow(/invalid index|absent parent/);
    expect(() =>
      applyAgentSessionDeltaOperations(before, [
        { op: "remove", path: "/composerDraft" },
      ]),
    ).toThrow(/absent parent|cannot be removed/);
    expect(() =>
      applyAgentSessionDeltaOperations(before, [
        { op: "replace", path: "/title", value: undefined },
      ]),
    ).toThrow(/plain JSON/);
    expect(() =>
      applyAgentSessionDeltaOperations(before, [
        { op: "replace", path: "/turns/0/objective", value: "rewrite" },
      ]),
    ).toThrow(/feedback fields/);
  });

  it("rejects executable authority and fabricated execution provenance", () => {
    const before = payload();
    expect(() =>
      applyAgentSessionDeltaOperations(before, [
        { op: "replace", path: "/originSessionID", value: UUID_A },
      ]),
    ).toThrow(/not part of the mutable Session surface/);
    expect(() =>
      applyAgentSessionDeltaOperations(before, [
        {
          op: "add",
          path: "/turns/1",
          value: {
            ...turn(UUID_A, "forged"),
            response: {
              ...turn(UUID_A, "forged").response,
              execution: { started_at: "2026-10-08T03:07:24.000Z", tools: [] },
            },
          },
        },
      ]),
    ).toThrow(/execution provenance/);
    expect(() =>
      applyAgentSessionDeltaOperations(before, [
        {
          op: "add",
          path: "/turns/1",
          value: {
            ...turn(UUID_A, "forged"),
            response: {
              ...turn(UUID_A, "forged").response,
              savedBlocks: [
                {
                  id: "b",
                  kind: "answer",
                  title: "t",
                  body: "b",
                  status: "informational",
                  citation_dependency_ids: [],
                  requires_user_decision: true,
                },
              ],
            },
          },
        },
      ]),
    ).toThrow(/executable authority|user decision/);
    expect(() =>
      applyAgentSessionDeltaOperations(before, [
        {
          op: "add",
          path: "/contactReceipts/0",
          value: { id: UUID_A, operationKey: "k", target: "execute" },
        },
      ]),
    ).not.toThrow();
  });

  it("keeps metadata arrays append-only", () => {
    const before = payload();
    before.screenshotTaskIDs = [UUID_B];
    expect(() =>
      applyAgentSessionDeltaOperations(before, [
        { op: "replace", path: "/screenshotTaskIDs/0", value: UUID_A },
      ]),
    ).toThrow(/only append/);
    expect(() =>
      applyAgentSessionDeltaOperations(before, [
        { op: "add", path: "/screenshotTaskIDs/0", value: UUID_A },
      ]),
    ).toThrow(/append in accepted order/);
  });
});

describe("delta planning", () => {
  it("emits only new turns and changed fields, never a full list rewrite", () => {
    const before = payload();
    const after = structuredClone(before);
    after.title = "Renamed";
    after.turns.push(turn(UUID_A, "second"));
    const operations = planAgentSessionDelta(before, after);
    expect(operations.map((operation) => operation.path)).toEqual([
      "/title",
      "/turns/1",
    ]);
    const applied = applyAgentSessionDeltaOperations(before, operations);
    expect(agentSessionIntentDigest(applied)).toBe(
      agentSessionIntentDigest(after),
    );
  });

  it("orders pending execution metadata last", () => {
    const before = payload();
    const after = structuredClone(before);
    after.turns.push(turn(UUID_A, "second"));
    after.pendingObjective = "resume the ask";
    after.pendingScreenshotRequestIdentity = "a".repeat(64);
    after.pendingScreenshotIdempotencyKey = "key";
    after.pendingScreenshotCapturedAt = "2026-10-08T03:07:24.000Z";
    after.title = "Renamed";
    const operations = planAgentSessionDelta(before, after);
    const lastNonPending = operations
      .map((operation, index) => ({ operation, index }))
      .filter(({ operation }) => !operation.path.startsWith("/pending"))
      .map(({ index }) => index)
      .reduce((max, index) => Math.max(max, index), -1);
    const firstPending = operations.findIndex((operation) =>
      operation.path.startsWith("/pending"),
    );
    expect(firstPending).toBeGreaterThan(lastNonPending);
  });

  it("splits frames while keeping pending metadata in the final frames", () => {
    const before = payload();
    const after = structuredClone(before);
    for (let index = 0; index < 12; index += 1) {
      after.turns.push(turn(UUID_A, `message ${index}`));
    }
    after.pendingObjective = "last";
    const plan = planAgentSessionDeltaFrames(before, after, {
      transferId: UUID_B,
      maxOperationsPerFrame: 3,
      maxFrameBytes: 1024 * 1024,
    });
    expect(plan.frames.length).toBeGreaterThan(1);
    expect(plan.frames.at(-1)!.at(-1)!.path).toBe("/pendingObjective");
    expect(plan.target_payload_digest).toBe(agentSessionIntentDigest(after));
    let cursor: unknown = before;
    for (const [index, frame] of plan.frames.entries()) {
      const merged = applyAgentSessionDeltaOperations(cursor, frame);
      if (index < plan.frames.length - 1) {
        expect(agentSessionIntentDigest(merged)).not.toBe(
          agentSessionIntentDigest(after),
        );
      }
      cursor = merged as AgentSessionPayload;
    }
    expect(agentSessionIntentDigest(cursor)).toBe(
      agentSessionIntentDigest(after),
    );
  });

  it("honors the per-frame byte estimate", () => {
    const before = payload();
    const after = structuredClone(before);
    for (let index = 0; index < 6; index += 1) {
      after.turns.push(turn(UUID_A, `message ${index} ${"x".repeat(2000)}`));
    }
    const plan = planAgentSessionDeltaFrames(before, after, {
      transferId: UUID_B,
      maxOperationsPerFrame: 256,
      maxFrameBytes: 4096,
    });
    for (const frame of plan.frames) {
      expect(JSON.stringify(frame).length).toBeLessThan(6000);
    }
  });

  it("plans frames by actual UTF-8 request bytes for non-ASCII and emoji turns", () => {
    // Four Han-character turns (~105KiB each) are individually admissible but
    // must never merge into one over-guard frame; byte planning counts the
    // complete request envelope, not a character proxy.
    const base = payload();
    const after = structuredClone(base);
    for (let index = 0; index < 4; index += 1) {
      after.turns.push(turn(UUID_A, `段落${"漢".repeat(35000)}${index}`));
    }
    after.title = "emoji 🎉🎉🎉 与汉字标题";
    const plan = planAgentSessionDeltaFrames(base, after, {
      transferId: UUID_B,
      skeleton: base,
    });
    expect(plan.frames.length).toBeGreaterThan(1);
    for (const [index, frame] of plan.frames.entries()) {
      const bytes = agentSessionDeltaFrameRequestBytes({
        transferId: plan.transfer_id,
        targetPayloadDigest: plan.target_payload_digest,
        operations: frame,
        ...(index === 0 ? { skeleton: base } : {}),
      });
      expect(bytes).toBeLessThanOrEqual(250 * 1024);
      // UTF-8 request bytes count the real wire size; a character proxy
      // (JSON.stringify length) under-counts CJK turns threefold.
      if (frame.some((operation) => operation.path.startsWith("/turns/"))) {
        expect(bytes).toBeGreaterThan(JSON.stringify(frame).length);
      }
    }
    let cursor: unknown = base;
    for (const frame of plan.frames) {
      cursor = applyAgentSessionDeltaOperations(cursor, frame);
    }
    expect(agentSessionIntentDigest(cursor)).toBe(agentSessionIntentDigest(after));
    expect(() =>
      planAgentSessionDeltaFrames(base, after, {
        transferId: UUID_B,
        skeleton: base,
        // A single 105KiB turn can never fit a tiny frame: explicit failure,
        // never a silent oversized frame.
        maxFrameBytes: 64 * 1024,
      }),
    ).toThrow(/does not fit a single request frame/);
  });

  it("pairs source-associated metadata so every frame keeps inheritance authorization", () => {
    const before = payload() as unknown as Record<string, unknown>;
    before.screenshotTaskIDs = [UUID_A];
    const after = structuredClone(before) as Record<string, unknown>;
    const live = Array.from({ length: 33 }, (_, index) =>
      `10000000-0000-4000-8000-${String(index).padStart(12, "0")}`,
    );
    after.screenshotTaskIDs = [...(before.screenshotTaskIDs as string[]), ...live];
    after.inheritedScreenshotTaskIDs = [UUID_A, ...live];
    for (let index = 0; index < 5; index += 1) {
      (after.turns as unknown[]).push(turn(UUID_A, `screenshot turn ${index}`));
    }
    const plan = planAgentSessionDeltaFrames(before, after, {
      transferId: UUID_B,
      maxOperationsPerFrame: 6,
    });
    expect(plan.frames.length).toBeGreaterThan(3);
    // The planning verification walks every frame boundary through the exact
    // inheritance rules; a plan that would break them is refused up front.
    let cursor = before;
    for (const frame of plan.frames) {
      const next = applyAgentSessionDeltaOperations(cursor, frame) as Record<
        string,
        unknown
      >;
      expect(verifyScreenshotInheritanceStep(cursor, next)).toBe(true);
      cursor = next;
    }
    expect((cursor.inheritedScreenshotTaskIDs as string[]).length).toBe(34);

    // Marking inherited later must still cover every previously accepted task
    // reference; an uncovered live reference stays unexpressible.
    const uncovered = structuredClone(before) as Record<string, unknown>;
    uncovered.inheritedScreenshotTaskIDs = [UUID_B];
    expect(() => planAgentSessionDelta(before, uncovered)).toThrow(
      /cover every previously accepted task reference/,
    );
  });

  it("keeps the screenshot admission identity inside one frame", () => {
    const before = payload();
    const after = structuredClone(before);
    after.pendingObjective = "resume the screenshot ask";
    after.pendingScreenshotIdempotencyKey = "screenshot-key";
    after.pendingScreenshotRequestIdentity = "a".repeat(64);
    after.pendingScreenshotCapturedAt = "2026-10-08T03:07:24.000Z";
    for (let index = 0; index < 12; index += 1) {
      after.turns.push(turn(UUID_A, `bulk ${index}`));
    }
    const plan = planAgentSessionDeltaFrames(before, after, {
      transferId: UUID_B,
      maxOperationsPerFrame: 1,
    });
    const admissionFrame = plan.frames.find((frame) =>
      frame.some((operation) =>
        SCREENSHOT_ADMISSION_PATHS.has(operation.path),
      ),
    );
    expect(
      admissionFrame!.filter((operation) =>
        SCREENSHOT_ADMISSION_PATHS.has(operation.path),
      ),
    ).toHaveLength(4);
  });

  it("refuses unexpressible changes instead of rewriting history", () => {
    const before = payload();
    const removed = structuredClone(before);
    removed.turns = [];
    expect(() => planAgentSessionDelta(before, removed)).toThrow(
      /never removes accepted history/,
    );
    const rewritten = structuredClone(before);
    rewritten.turns[0]!.objective = "different message";
    expect(() => planAgentSessionDelta(before, rewritten)).toThrow(
      /identity and order/,
    );
    const moved = structuredClone(before);
    moved.originSessionID = UUID_B;
    expect(() => planAgentSessionDelta(before, moved)).toThrow(
      /immutable/,
    );
  });

  it("preserves unknown server fields through raw overlay planning", () => {
    // Raw canonical GET JSON carries fields a local DTO may not model. They
    // ride in the base record untouched while only the changed field emits an
    // operation.
    const before = payload() as unknown as Record<string, unknown>;
    const firstTurn = before.turns as Array<Record<string, unknown>>;
    (firstTurn[0]!.response as Record<string, unknown>).execution = {
      started_at: "2026-10-08T03:07:24.000Z",
      completed_at: "2026-10-08T03:07:25.000Z",
      tools: [],
    };
    const after = structuredClone(before);
    after.isUnread = true;
    const operations = planAgentSessionDelta(before, after);
    expect(operations).toEqual([
      { op: "replace", path: "/isUnread", value: true },
    ]);
    const applied = applyAgentSessionDeltaOperations(before, operations) as Record<
      string,
      unknown
    >;
    expect(
      (applied.turns as Array<Record<string, unknown>>)[0]!.response,
    ).toEqual(firstTurn[0]!.response);
    expect(agentSessionIntentDigest(applied)).toBe(
      agentSessionIntentDigest(after),
    );
  });

  it("canonicalizes identifiers on both sides of the digest", () => {
    const lower = canonicalSessionIntent(payload(UUID_A));
    expect((lower as { id: string }).id).toBe(UUID_A.toLowerCase());
    expect(agentSessionIntentDigest(payload(UUID_A))).toBe(
      agentSessionIntentDigest(payload(UUID_A.toLowerCase())),
    );
    expect(
      agentSessionIntentDigest(payload(UUID_A)) ===
        agentSessionIntentDigest(payload(UUID_B)),
    ).toBe(false);
  });

  it("applies every planned operation type through the shared validator", () => {
    const before = payload() as unknown as Record<string, unknown>;
    before.composerDraft = "old draft";
    before.contactProposal = { idempotencyKey: "k", updatedAt: "2026-10-08T03:07:24.000Z", draft: {} };
    const after = structuredClone(before) as Record<string, unknown>;
    delete after.composerDraft;
    delete after.contactProposal;
    after.contextWasTrimmed = true;
    (after.turns as unknown[]).push(turn(UUID_A, "second"));
    const operations = planAgentSessionDelta(before, after);
    const applied = applyAgentSessionDeltaOperations(before, operations);
    expect(agentSessionIntentDigest(applied)).toBe(
      agentSessionIntentDigest(after),
    );
    for (const operation of operations as AgentSessionDeltaOperation[]) {
      expect(["add", "remove", "replace"]).toContain(operation.op);
    }
  });

  it("rejects planning across non-payload values", () => {
    expect(() => planAgentSessionDelta([], {})).toThrow(
      AgentSessionDeltaOperationError,
    );
  });
});
