import { CONTRACT_VERSION, type SessionResponse } from "@talent-signal/contracts";
import { afterEach, beforeEach, expect, it, vi } from "vitest";

import {
  buildCapirSessionCookie,
  CAPIR_WEB_HANDOFF_KEY_HEADER,
  exchangeCapirHandoff,
} from "./capirHandoff";

const secret = "s".repeat(43);

function labSession(): SessionResponse {
  return {
    contract_version: CONTRACT_VERSION,
    access_token: "k".repeat(43),
    expires_at: new Date(Date.now() + 3_600_000).toISOString(),
    account: {
      id: "11111111-1111-4111-8111-111111111111",
      slug: "lab-22222222-2222-4222-8222-222222222222",
      name: "Test workspace · 22222222",
    },
    user: {
      id: "33333333-3333-4333-8333-333333333333",
      email: "test-22222222@lab.invalid",
      display_name: "Test user",
      kind: "lab_human",
      role: "member",
      username: null,
    },
  };
}

beforeEach(() => {
  vi.stubEnv("CAPIR_WEB_HANDOFF_KEY", "test-web-handoff-key-0123456789abcdef");
});

afterEach(() => {
  vi.unstubAllEnvs();
  vi.unstubAllGlobals();
});

it("authenticates to the backend with the server-only consumer key", async () => {
  const calls: Array<{ url: string; init: RequestInit }> = [];
  vi.stubGlobal(
    "fetch",
    vi.fn(async (url: string, init: RequestInit) => {
      calls.push({ url: String(url), init });
      return new Response(JSON.stringify({ schema_version: "capir.v1", session: labSession(), entry_path: "/capir/sandboxes/22222222-2222-4222-8222-222222222222" }), {
        status: 200,
        headers: { "content-type": "application/json" },
      });
    }),
  );
  const session = await exchangeCapirHandoff(secret, "http://127.0.0.1:3000");
  expect(session.user.kind).toBe("lab_human");
  expect(calls).toHaveLength(1);
  const { url, init } = calls[0]!;
  expect(url).toBe("http://127.0.0.1:4317/v1/capir/handoffs/exchange");
  // The secret travels in the private POST body, never in the URL.
  expect(url).not.toContain(secret);
  const headers = init.headers as Record<string, string>;
  expect(init.redirect).toBe("error");
  expect(headers[CAPIR_WEB_HANDOFF_KEY_HEADER]).toBe(
    "test-web-handoff-key-0123456789abcdef",
  );
  expect(JSON.parse(String(init.body))).toEqual({
    handoff_secret: secret,
    web_origin: "http://127.0.0.1:3000",
  });
});

it("fails closed when a secret exchange attempts to redirect", async () => {
  const fetchImpl = vi.fn(async (_url: string, init: RequestInit) => {
    expect(init.redirect).toBe("error");
    throw new TypeError("redirect mode is set to error");
  }) as unknown as typeof fetch;
  await expect(exchangeCapirHandoff(secret, "http://127.0.0.1:3000", fetchImpl))
    .rejects.toMatchObject({ status: 502, code: "CAPIR_HANDOFF_EXCHANGE_FAILED" });
  expect(fetchImpl).toHaveBeenCalledOnce();
});

it("fails closed when the trusted consumer key is not configured", async () => {
  vi.stubEnv("CAPIR_WEB_HANDOFF_KEY", "");
  const fetchSpy = vi.fn();
  vi.stubGlobal("fetch", fetchSpy);
  await expect(
    exchangeCapirHandoff(secret, "http://127.0.0.1:3000"),
  ).rejects.toMatchObject({ status: 503, code: "CAPIR_WEB_HANDOFF_UNAVAILABLE" });
  expect(fetchSpy).not.toHaveBeenCalled();
});

it("maps consumed, expired and wrong-origin secrets to a denial", async () => {
  vi.stubGlobal(
    "fetch",
    vi.fn(async () =>
      new Response(
        JSON.stringify({
          error: { code: "CAPIR_AUTH_DENIED", message: "denied", request_id: "r" },
        }),
        { status: 401, headers: { "content-type": "application/json" } },
      ),
    ),
  );
  await expect(
    exchangeCapirHandoff(secret, "http://127.0.0.1:3000"),
  ).rejects.toMatchObject({ status: 401, code: "CAPIR_AUTH_DENIED" });
});

it("builds the existing encrypted Auth.js HttpOnly cookie bound to the entry expiry", async () => {
  const session = labSession();
  const cookie = await buildCapirSessionCookie(session, Date.now());
  expect(cookie.name).toBe("talent-signal.session-v2");
  expect(cookie.options.httpOnly).toBe(true);
  expect(cookie.options.sameSite).toBe("lax");
  expect(cookie.options.path).toBe("/");
  expect(cookie.options.maxAge).toBeGreaterThan(0);
  expect(cookie.options.maxAge).toBeLessThanOrEqual(3600);
  expect(cookie.claims.capir.sandbox_id).toBe(
    "22222222-2222-4222-8222-222222222222",
  );
  const { decode } = await import("next-auth/jwt");
  const decoded = await decode({
    token: cookie.value,
    secret: "talent-signal-local-development-secret-v2",
    salt: "talent-signal.session-v2",
  });
  expect(decoded?.backendAccountSlug).toBe(session.account.slug);
  expect((decoded as { capir?: { sandbox_id?: string } } | null)?.capir?.sandbox_id).toBe(
    "22222222-2222-4222-8222-222222222222",
  );
  expect(decoded?.backendAccessToken).toBe(session.access_token);
});

it("refuses to mint a cookie for a non-Lab account", async () => {
  const session = labSession();
  const main = {
    ...session,
    account: { ...session.account, slug: "fixture-alpha" },
  };
  await expect(buildCapirSessionCookie(main, Date.now())).rejects.toThrow(
    /scoped Lab account/,
  );
});
