function control(res,b,{content=block,reason='compaction',iterations=[{type:'compaction',input_tokens:10,output_tokens:3}],mutation=x=>x,keepOpen=false,error=false}={}) {
  const usage={input_tokens:0,output_tokens:0,iterations};
  if(!b.stream){json(res,{type:'message',role:'assistant',model:'pin',content:content?[content]:[],stop_reason:reason,usage,...(error?{error:{message:'do not expose'}}:{})});return;}
  res.writeHead(200,{'content-type':'text/event-stream'});
  let text=event('message_start',{message:{model:'pin',usage:{input_tokens:0,output_tokens:0}}});
  if(content)text+=event('content_block_start',{index:0,content_block:content})+event('content_block_stop',{index:0});
  text+=event('message_delta',{delta:{stop_reason:reason},usage,...(error?{error:{message:'do not expose'}}:{})})+event('message_stop',{});
  if(keepOpen)res.write(mutation(text));else res.end(mutation(text));
}

}

      return json(res,handlers.metadata??metadata());

  });

}
