import { afterEach, describe, expect, it, vi } from "vitest";
import { executionDeadline } from "./executionDeadline.js";
afterEach(() => vi.useRealTimers());
describe("execution deadlines", () => {
  it("does not create a timer without a selected bound", () => {
    expect(executionDeadline(vi.fn(), undefined)).toBeUndefined();
  });
  it("waits across the Node timer boundary instead of overflowing to one millisecond", async () => {
    vi.useFakeTimers();
    const callback = vi.fn();
    const cancel = executionDeadline(callback, 2_147_483_648);
    await vi.advanceTimersByTimeAsync(2_147_483_647);
    expect(callback).not.toHaveBeenCalled();
    await vi.advanceTimersByTimeAsync(1);
    expect(callback).toHaveBeenCalledOnce();
    cancel!();
  });
  it("cancels a later timer chunk", async () => {
    vi.useFakeTimers();
    const callback = vi.fn();
    const cancel = executionDeadline(callback, 3_000_000_000);
    await vi.advanceTimersByTimeAsync(2_147_483_647);
    cancel!();
    await vi.advanceTimersByTimeAsync(1_000_000_000);
    expect(callback).not.toHaveBeenCalled();
  });
});
