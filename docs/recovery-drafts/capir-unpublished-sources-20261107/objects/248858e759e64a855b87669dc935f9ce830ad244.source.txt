    func testLateWindowResultCannotRepopulateOrCrossChangedIntakeBoundary() async throws {
        for boundary in ["clear", "stop", "pause-resume", "sign-out", "account-change", "cancel"] {
            let capture = LateWindowCapture()
            let store = AccountFixtureCapsuleStore()
            let service = SwitchingAccountService(accountIDs: ["account-a", "account-b"])
            let model = AppModel(service: service, accountID: "account-a", capsuleStore: store, windowCapture: capture)
            await model.load()
            let task = Task { await model.addSystemSelectedWindow() }
            while !capture.started { await Task.yield() }
            switch boundary {
            case "clear": model.clearLocalContext()
            case "stop": model.stopContextIntake()
            case "pause-resume": model.togglePause(); model.togglePause()
            case "sign-out": await model.signOutAndClearLocalData()
            case "account-change": await model.load()
            default: model.cancelSystemSelectedWindow()
            }
            capture.completeLate()
            await task.value
            XCTAssertTrue(model.capsule.items.isEmpty, boundary)
            XCTAssertFalse(model.isSelectingWindow, boundary)
            for account in ["account-a", "account-b", "signed-out"] {
                XCTAssertTrue(try store.load(accountID: account, now: Date()).draft.items.isEmpty, "\(boundary)/\(account)")
            }
        }
    }

    func testClearingContextInvalidatesLateFileExtractionResult() async {
        let extraction = LocalFileTextExtraction(
            displayName: "late-screenshot.png",
            recognizedText: "Candidate needs the remote policy before September 3, 2026.",
            rawData: Data([9, 8, 7]),
            mediaType: "image/png",
            sourceFingerprint: "late-file-fingerprint",
            method: .localImageOCR
        )
        let extractor = SuspendedLocalFileTextExtractor()
        let model = AppModel(
            service: FixtureRelationshipService(initialMode: .ready),
            localFileTextExtractor: extractor
        )

        let importTask = Task {
            await model.addFiles([URL(fileURLWithPath: "/tmp/late-screenshot.png")])
        }
        for _ in 0..<100 {
            if await extractor.didStart() { break }
            await Task.yield()
        }
        XCTAssertTrue(model.isImportingFiles)

        model.clearLocalContext()
        await extractor.finish(with: extraction)
        await importTask.value

        XCTAssertTrue(model.capsule.items.isEmpty)
        XCTAssertNil(model.provisionalInsight)
        XCTAssertFalse(model.isImportingFiles)
        XCTAssertNil(model.fileIngestReceipt)
        XCTAssertTrue(model.deletionReceipt?.contains("No external record was changed") == true)
    }

    private func waitForCall(
        _ service: ControlledRelationshipService,
        method: String,

    }

    func draftCount(for accountID: String) -> Int {
        drafts[accountID] == nil ? 0 : 1
    }

    func drafts(for accountID: String) -> ContextCapsuleDraft? {
        drafts[accountID]
    }
}

private final class AccountFixtureReminderRecoveryStore: ReminderOperationRecoveryPersisting, @unchecked Sendable {

    var clearedAccountIDs: [String] = []

            recoveries.removeValue(forKey: accountID)

        return recovery

        recoveries[accountID] = recovery

        return recoveries.removeValue(forKey: accountID) != nil
    }
}

/// A relationship service whose every async call suspends until the test
/// releases it. Controlled continuations are what make late success, late
/// error, replacement, and superseding orderings observable.
private actor ControlledRelationshipService: MacRelationshipServing {
    enum Outcome: Sendable {
        case accepted
        case response(MacRelationshipServiceResponse)
        case failure(String)
        /// Positive host proof that no Agent Task write ever left the service.
        case taskNotDispatched(String)
    }

    enum SignOutOutcome: Sendable {
        case revoked(SessionSignOutReceipt)
        case localOnly(String)
        case remoteUnknown(String)
        case failure(String)
    }

    /// Local retirement status surfaced by the controlled service; the real
    /// encrypted retirement-store proofs live in the boundary tests.
    var signOutLocalRetirement: LocalRetirementStatus = .notRequired(
        "The controlled service manages no durable unknown-operation correlation."
    )

    private var queued: [String: [Outcome]] = [:]
    private var waiters: [String: [CheckedContinuation<Outcome, Error>]] = [:]
    private var queuedSignOut: [SignOutOutcome] = []
    private var signOutWaiters: [CheckedContinuation<SessionSignOutOutcome, Error>] = []
    private var callCounts: [String: Int] = [:]

    func callCount(_ method: String) -> Int { callCounts[method] ?? 0 }

    func finish(_ method: String, with outcome: Outcome) {
        var waiter: CheckedContinuation<Outcome, Error>?
        if var pending = waiters[method], !pending.isEmpty {
            waiter = pending.removeFirst()
            waiters[method] = pending
        } else {
            queued[method, default: []].append(outcome)
        }
        if let waiter {
            switch outcome {
            case .accepted, .response:
                waiter.resume(returning: outcome)
            case .failure(let message):
                waiter.resume(throwing: ControlledServiceError(message))
            case .taskNotDispatched(let message):
                waiter.resume(throwing: RelationshipServiceError.sourceWorkflowStopped(message))
            }
        }
    }

    func finishSignOut(with outcome: SignOutOutcome) {
        var waiter: CheckedContinuation<SessionSignOutOutcome, Error>?
        if !signOutWaiters.isEmpty {
            waiter = signOutWaiters.removeFirst()
        } else {
            queuedSignOut.append(outcome)
        }
        guard let waiter else { return }
        switch outcome {
        case .revoked, .localOnly, .remoteUnknown:
            waiter.resume(returning: outcome.sessionOutcome(localRetirement: signOutLocalRetirement))
        case .failure(let message):
            waiter.resume(throwing: ControlledServiceError(message))
        }
    }

    private func perform(_ method: String) async throws -> Outcome {
        callCounts[method, default: 0] += 1
        if var pending = queued[method], !pending.isEmpty {
            let outcome = pending.removeFirst()
            queued[method] = pending
            return outcome
        }
        return try await withCheckedThrowingContinuation { continuation in
            waiters[method, default: []].append(continuation)
        }
    }

    private static func response(_ method: String, from outcome: Outcome) throws -> MacRelationshipServiceResponse {
        switch outcome {
        case .response(let response):
            return response
        case .accepted:
            return .syntheticFixture(FixtureRelationshipService.fixture(mode: .ready))
        case .failure(let message):
            throw ControlledServiceError(message)
        case .taskNotDispatched(let message):
            throw RelationshipServiceError.sourceWorkflowStopped(message)
        }
    }

    func loadWorkspace() async throws -> MacRelationshipServiceResponse {
        try Self.response("loadWorkspace", from: try await perform("loadWorkspace"))
    }

    func confirmScope(_ selection: RelationshipScopeSelection) async throws {
        guard case .accepted = try await perform("confirmScope") else {
            throw ControlledServiceError("confirmScope expected an accepted outcome.")
        }
    }

    func submit(manifest: SubmittedContextManifest) async throws -> MacRelationshipServiceResponse {
        try Self.response("submit", from: try await perform("submit"))
    }

    func resolveDecision(_ request: CanonicalDecisionRequest) async throws -> MacRelationshipServiceResponse {
        try Self.response("resolveDecision", from: try await perform("resolveDecision"))
    }

    func reconcileDecisionOutcome() async throws -> MacRelationshipServiceResponse {
        try Self.response("reconcileDecisionOutcome", from: try await perform("reconcileDecisionOutcome"))
    }

    func openProjection(_ projection: CanonicalProjectionRequest) async throws -> MacRelationshipServiceResponse {
        try Self.response("openProjection", from: try await perform("openProjection"))
    }

    func signOut() async throws -> SessionSignOutOutcome {
        callCounts["signOut", default: 0] += 1
        if queuedSignOut.isEmpty {
            return try await withCheckedThrowingContinuation { continuation in
                signOutWaiters.append(continuation)
            }
        }
        let outcome = queuedSignOut.removeFirst()
        switch outcome {
        case .revoked, .localOnly, .remoteUnknown:
            return outcome.sessionOutcome(localRetirement: signOutLocalRetirement)
        case .failure(let message):
            throw ControlledServiceError(message)
        }
    }
}

private extension ControlledRelationshipService.SignOutOutcome {
    func sessionOutcome(localRetirement: LocalRetirementStatus) -> SessionSignOutOutcome {
        let remote: SessionSignOutOutcome.Remote = switch self {
        case .revoked(let receipt):
            .revoked(receipt)
        case .localOnly(let detail):
            .localOnly(detail)
        case .remoteUnknown(let detail):
            .remoteUnknown(detail)
        case .failure(let message):
            .remoteUnknown(message)
        }
        return SessionSignOutOutcome(remote: remote, localRetirement: localRetirement)
    }
}

private struct ControlledServiceError: LocalizedError {
    let message: String
    init(_ message: String) { self.message = message }
    var errorDescription: String? { message }
}

/// A reminder service that can hold one executed write or removal in flight so
/// delayed landing behavior is testable across account, sign-out, and scope
/// changes.
@MainActor
private final class ControlledFollowUpReminderService: FollowUpReminderServing {
    var holdsExecute = false
    var holdsRemove = false
    private(set) var executeCallCount = 0
    private(set) var removeCallCount = 0
    private var executedProposal: FollowUpReminderProposal?
    private var executeContinuation: CheckedContinuation<Result<FollowUpReminderReceipt, FollowUpReminderFailure>, Never>?
    private var removeContinuation: CheckedContinuation<Result<FollowUpReminderRemovalReceipt, FollowUpReminderFailure>, Never>?

    func setLocalRecoveryAccount(_ accountID: String) { }
    func clearLocalRecovery() { }

    func previewDestination() async -> Result<FollowUpReminderDestination, FollowUpReminderFailure> {
        .success(.init(identifier: "follow-up-list", title: "Follow Up"))
    }

    func execute(_ proposal: FollowUpReminderProposal) async -> Result<FollowUpReminderReceipt, FollowUpReminderFailure> {
        executeCallCount += 1
        executedProposal = proposal
        if !holdsExecute { return .success(verifiedReceipt(for: proposal)) }
        return await withCheckedContinuation { executeContinuation = $0 }
    }

    func completeExecuteWithVerifiedReceipt() {
        guard let proposal = executedProposal else { return }
        executeContinuation?.resume(returning: .success(verifiedReceipt(for: proposal)))
        executeContinuation = nil
    }

    func reconcile(_ proposal: FollowUpReminderProposal) async -> Result<FollowUpReminderReceipt, FollowUpReminderFailure> {
        .success(verifiedReceipt(for: proposal))
    }

    func remove(_ receipt: FollowUpReminderReceipt) async -> Result<FollowUpReminderRemovalReceipt, FollowUpReminderFailure> {
        removeCallCount += 1
        if !holdsRemove { return .success(removalReceipt(for: receipt)) }
        return await withCheckedContinuation { removeContinuation = $0 }
    }

    func completeRemoveWithSuccess(idempotencyKey: String, reminderIdentifier: String) {
        removeContinuation?.resume(returning: .success(FollowUpReminderRemovalReceipt(
            idempotencyKey: idempotencyKey,
            reminderIdentifier: reminderIdentifier,
            destinationTitle: "Follow Up",
            removedAt: Date(),
            wasAlreadyAbsent: false
        )))
        removeContinuation = nil
    }

    func reconcileRemoval(_ receipt: FollowUpReminderReceipt) async -> Result<FollowUpReminderRemovalReceipt, FollowUpReminderFailure> {
        .success(removalReceipt(for: receipt))
    }

    private func verifiedReceipt(for proposal: FollowUpReminderProposal) -> FollowUpReminderReceipt {
        FollowUpReminderReceipt(
            idempotencyKey: proposal.idempotencyKey,
            reminderIdentifier: "controlled-reminder",
            title: proposal.title,
            dueAt: proposal.dueAt,
            destinationIdentifier: proposal.destination.identifier,
            destinationTitle: proposal.destination.title,
            verifiedAt: Date()
        )
    }

    private func removalReceipt(for receipt: FollowUpReminderReceipt) -> FollowUpReminderRemovalReceipt {
        FollowUpReminderRemovalReceipt(
            idempotencyKey: receipt.idempotencyKey,
            reminderIdentifier: receipt.reminderIdentifier,
            destinationTitle: receipt.destinationTitle,
            removedAt: Date(),
            wasAlreadyAbsent: false
        )
    }
}


/// Capsule store whose save seam can fail so real persistence barriers are
/// testable; a plain successful roundtrip is not a failure test.
private final class FailingSaveCapsuleStore: LocalCapsulePersisting, @unchecked Sendable {
    var failSaves = false
    private var drafts: [String: ContextCapsuleDraft] = [:]

    func load(accountID: String, now: Date) throws -> CapsuleRecoveryResult {
        .init(draft: drafts[accountID] ?? ContextCapsuleDraft(), expiredItemCount: 0)
    }

    func save(_ draft: ContextCapsuleDraft, accountID: String, now: Date) throws {
        if failSaves {
            throw LocalCapsuleStoreError.encryptionFailed
        }
        drafts[accountID] = draft
    }

    func clear(accountID: String, deleteKey: Bool) throws -> LocalCapsuleDeletionReceipt {
        .init(deletedFile: drafts.removeValue(forKey: accountID) != nil, deletedKey: deleteKey)
    }
}
