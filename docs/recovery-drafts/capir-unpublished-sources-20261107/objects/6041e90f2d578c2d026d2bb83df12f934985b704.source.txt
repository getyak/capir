    static func bootstrap() -> AppModel {
        let arguments = ProcessInfo.processInfo.arguments
        let fixtureState = value(after: "--fixture-state", in: arguments)
            .flatMap(WorkspaceMode.argumentValue) ?? .ready

        if let rawBaseURL = value(after: "--live-backend", in: arguments),
           let baseURL = URL(string: rawBaseURL),
           let service = try? URLMacRelationshipService(
               configuration: .init(
                   baseURL: baseURL,
                   accountSlug: value(after: "--account-slug", in: arguments) ?? "fixture-alpha",
                   userEmail: value(after: "--user-email", in: arguments) ?? "recruiter@alpha.local"
               ),
               unknownResolutionStore: SecureUnknownResolutionStore.shared
           ) {
            let model = AppModel(
                service: service,
                initialMode: .working,
                accountID: "pending-live-account",
                pursuitID: nil,
                personID: nil,
                relationshipContextID: nil,
                capsuleStore: SecureLocalCapsuleStore.shared,
                followUpReminderService: EventKitFollowUpReminderService(),
                reminderRecoveryStore: SecureReminderOperationRecoveryStore.shared
            )
            model.isSyntheticFixture = false
            if arguments.contains("--open-action-center") {
                model.selectedNavigation = .actionCenter
            }
            return model
        }

        // Fixture mode is an explicit, separate service result and is always
        // labeled in the UI. It is never treated as backend proof.
        let model = AppModel(
            service: FixtureRelationshipService(initialMode: fixtureState),
            initialMode: fixtureState,
            followUpReminderService: PreviewOnlyFollowUpReminderService()
        )
        if arguments.contains("--ui-testing"), !arguments.contains("--today-preview") {
            model.selectedNavigation = .workspace
        }
        if arguments.contains("--quick-panel-preview") || arguments.contains("--today-preview") {
            model.addSelectedText("I need clarity on the remote policy before Friday because the other process has accelerated.")
        }
        if arguments.contains("--today-preview") {
            model.todayAttention = Self.syntheticTodayAttention(for: fixtureState)
        }
        if arguments.contains("--open-action-center") {
            model.selectedNavigation = .actionCenter
        }
        return model
    }

    private static func value(after flag: String, in arguments: [String]) -> String? {
        guard let index = arguments.firstIndex(of: flag), arguments.indices.contains(index + 1) else { return nil }
        return arguments[index + 1]
    }

    private static func scopeIdentityTerms(_ displayLabel: String) -> [String] {
        let name = displayLabel
            .components(separatedBy: " — ").first?
            .components(separatedBy: " – ").first?
            .trimmingCharacters(in: .whitespacesAndNewlines) ?? displayLabel
        let normalizedName = normalizedIdentityText(name)
        let words = normalizedName.split(separator: " ").map(String.init)
        let distinctiveWords = words.filter { word in
            word.unicodeScalars.contains(where: { !$0.isASCII }) ? word.count >= 2 : word.count >= 4
        }
        return Array(Set(([normalizedName] + distinctiveWords).filter { !$0.isEmpty }))
    }

    private static func source(_ source: String, containsIdentityTerm term: String) -> Bool {
        let normalizedSource = normalizedIdentityText(source)
        if term.contains(" ") {
            return normalizedSource.contains(term)
        }
        return Set(normalizedSource.split(separator: " ").map(String.init)).contains(term)
    }

    private static func normalizedIdentityText(_ text: String) -> String {
        text.folding(options: [.caseInsensitive, .diacriticInsensitive], locale: .current)
            .lowercased()
            .unicodeScalars
            .map { CharacterSet.alphanumerics.contains($0) ? Character(String($0)) : " " }
            .reduce(into: "") { partial, character in
                if character == " ", partial.last == " " { return }
                partial.append(character)
            }
            .trimmingCharacters(in: .whitespacesAndNewlines)
    }

    private var initializationTask: Task<Bool, Never>?
    private var didInitialize = false

    /// Native intake cannot save under the pre-login placeholder account.
    func ensureInitialized() async -> Bool {
        if didInitialize { return true }
        if let initializationTask { return await initializationTask.value }
        let initializationID = UUID()
        let ticket = authority.ticket("initialize-workspace")
        activeInitializationID = initializationID
        let task = Task { @MainActor in
            do {
                let response = try await self.service.loadWorkspace()
                // An old initialization can neither reload over a later login
                // nor clear its memory or requests.
                guard self.authority.preservesAuthentication(ticket) else { return false }
                self.apply(response)
                let landingTicket = self.authority.ticket("initialize-admission-recovery")
                let landingRuntimeID = self.sessionRuntimeID
                await self.recoverExactTaskAdmissions()
                guard self.authority.preservesAuthentication(landingTicket),
                      self.sessionRuntimeID == landingRuntimeID, !self.isSignedOut,
                      self.activeInitializationID == initializationID else { return false }
                self.didInitialize = true
                return true
            } catch {
                guard self.authority.preservesAuthentication(ticket) else { return false }
                self.fail(error)
                return false
            }
        }
        initializationTask = task
        let ready = await task.value
        // Initialization cleanup only clears the matching initialization ID.
        if activeInitializationID == initializationID {
            initializationTask = nil
            activeInitializationID = nil
        }
        return ready
    }

    func load() async {
        // An explicit new load is a deliberate authority transition: every
        // older ticket stops being able to land over the workspace it creates.
        authority.invalidate(.authentication)
        let ticket = authority.ticket("load-workspace")
        do {
            let response = try await service.loadWorkspace()
            guard authority.preservesAuthentication(ticket) else { return }
            apply(response)
            await recoverExactTaskAdmissions()
        } catch {
            guard authority.preservesAuthentication(ticket) else { return }
            fail(error)
        }
    }

    func addSelectedText(_ text: String) {
        guard !isSignedOut else {
            errorMessage = "Sign in before creating a new Context Capsule."
            return
        }
        guard !isPaused else {
            errorMessage = "Context intake is paused. Resume before adding anything."
            return
        }
        invalidateActiveLocalIntake()
        let acceptedAt = Date()
        let countBefore = capsule.items.count
        capsule.addSelectedText(text, now: acceptedAt)
        guard capsule.items.count > countBefore else { return }
        intakeControlReceipt = nil
        fileIngestReceipt = nil
        localRecoveryNotice = nil
        companionTrialMetrics = CompanionTrialMetrics(inputAcceptedAt: acceptedAt)
        companionTrialExportReceipt = nil
        persistCapsule()
        refreshProvisionalInsight()
        companionTrialMetrics.firstValueMilliseconds = Date().timeIntervalSince(acceptedAt) * 1_000
        companionTrialMetrics.actionWasProposed = provisionalInsight?.suggestedAction != nil
        mode = capsule.items.isEmpty ? .empty : .ready
        errorMessage = nil
    }

    @discardableResult
    func addServiceSelectedText(_ text: String, requestID: UUID) -> Bool {
        guard !handledServiceRequestIDs.contains(requestID) else { return false }
        handledServiceRequestIDs.insert(requestID)
        let countBefore = capsule.items.count
        addSelectedText(text)
        if capsule.items.count > countBefore {
            intakeControlReceipt = "Added text from an explicitly invoked macOS selection service. No clipboard polling or background capture occurred."
            return true
        }
        return false
    }

    func addFiles(_ urls: [URL]) async {
        guard !isSignedOut else {
            errorMessage = "Sign in before creating a new Context Capsule."
            return
        }
        guard !isPaused else {
            errorMessage = "Context intake is paused. Resume before adding anything."
            return
        }
        guard !isImportingFiles, !isSelectingWindow, !urls.isEmpty else { return }

        let acceptedAt = Date()
        let importID = UUID()
        let importAccountID = accountID
        let selectedURLs = Array(urls.prefix(8))
        var extractedFiles: [LocalFileTextExtraction] = []
        var failureCount = max(0, urls.count - selectedURLs.count)
        var firstFailure: Error?
        activeFileImportID = importID
        isImportingFiles = true
        fileIngestReceipt = "Reading \(selectedURLs.count) explicitly chosen file\(selectedURLs.count == 1 ? "" : "s") on this Mac. Nothing has left the device."
        defer {
            if activeFileImportID == importID {
                activeFileImportID = nil
                isImportingFiles = false
            }
        }

        for url in selectedURLs {
            let didAccess = url.startAccessingSecurityScopedResource()
            defer { if didAccess { url.stopAccessingSecurityScopedResource() } }
            do {
                let extracted = try await localFileTextExtractor.extract(url: url)
                guard activeFileImportID == importID, accountID == importAccountID, !isSignedOut, !isPaused else { return }
                extractedFiles.append(extracted)
            } catch {
                guard activeFileImportID == importID, accountID == importAccountID, !isSignedOut, !isPaused else { return }
                failureCount += 1
                if firstFailure == nil { firstFailure = error }
            }
        }

        guard activeFileImportID == importID, accountID == importAccountID, !isSignedOut, !isPaused else { return }
        guard !extractedFiles.isEmpty else {
            fileIngestReceipt = "No file was retained or added to the Capsule."
            errorMessage = firstFailure?.localizedDescription ?? "No supported file was selected."
            return
        }

        for extracted in extractedFiles {
            capsule.addProcessedFile(
                displayName: extracted.displayName,
                reviewedText: extracted.recognizedText,
                rawData: extracted.rawData,
                mediaType: extracted.mediaType,
                acquisition: extracted.method.acquisition,
                sourceFingerprint: extracted.sourceFingerprint,
                now: acceptedAt
            )
        }
        let importedCount = extractedFiles.count
        let reviewableCount = extractedFiles.lazy.filter { !$0.recognizedText.isEmpty }.count

        intakeControlReceipt = nil
        windowCaptureReceipt = nil
        localRecoveryNotice = nil
        companionTrialMetrics = CompanionTrialMetrics(inputAcceptedAt: acceptedAt)
        companionTrialExportReceipt = nil
        persistCapsule()
        refreshProvisionalInsight()
        companionTrialMetrics.firstValueMilliseconds = provisionalInsight == nil
            ? nil
            : Date().timeIntervalSince(acceptedAt) * 1_000
        companionTrialMetrics.actionWasProposed = provisionalInsight?.suggestedAction != nil
        mode = capsule.items.isEmpty ? .empty : .ready
        let reviewSummary = reviewableCount > 0
            ? "\(reviewableCount) produced reviewable local text."
            : "No reviewable text was found; the raw file remains encrypted and local-only."
        let failureSummary = failureCount > 0
            ? " \(failureCount) other selection\(failureCount == 1 ? "" : "s") was not retained."
            : ""
        fileIngestReceipt = "Added \(importedCount) explicitly chosen file\(importedCount == 1 ? "" : "s"). \(reviewSummary) Raw bytes and text remain on this Mac until you change the visible boundary.\(failureSummary)"
        errorMessage = failureCount > 0
            ? "Some selected files could not be reviewed locally. \(firstFailure?.localizedDescription ?? "Unsupported selections were not retained.")"
            : nil
    }

    func addSystemSelectedWindow() async {
        guard !isSignedOut else {
            errorMessage = "Sign in before creating a new Context Capsule."
            return
        }
        guard !isPaused else {
            errorMessage = "Context intake is paused. Resume before adding anything."
            return
        }
        guard !isSelectingWindow, !isImportingFiles else { return }

        let captureID = UUID()
        let captureAccountID = accountID
        activeWindowCaptureID = captureID
        isSelectingWindow = true
        fileIngestReceipt = nil
        windowCaptureReceipt = "Waiting for the macOS single-window picker. Nothing has been captured yet."
        defer {
            if activeWindowCaptureID == captureID {
                activeWindowCaptureID = nil
                isSelectingWindow = false
            }
        }
        do {
            let captured = try await windowCapture.captureOneWindow()
            guard activeWindowCaptureID == captureID, accountID == captureAccountID,
                  !isSignedOut, !isPaused, !Task.isCancelled else { return }
            let acceptedAt = Date()
            capsule.addWindowCapture(
                recognizedText: captured.recognizedText,
                imagePNG: captured.imagePNG,
                pixelWidth: captured.pixelWidth,
                pixelHeight: captured.pixelHeight,
                sourceFingerprint: captured.sourceFingerprint,
                now: acceptedAt
            )
            intakeControlReceipt = nil
            localRecoveryNotice = nil
            companionTrialMetrics = CompanionTrialMetrics(inputAcceptedAt: acceptedAt)
            companionTrialExportReceipt = nil
            persistCapsule()
            refreshProvisionalInsight()
            companionTrialMetrics.firstValueMilliseconds = provisionalInsight == nil
                ? nil
                : Date().timeIntervalSince(acceptedAt) * 1_000
            companionTrialMetrics.actionWasProposed = provisionalInsight?.suggestedAction != nil
            mode = .ready
            let derivative: String
            if captured.localTextRecognition == .unavailable {
                derivative = "Local text recognition was unavailable. The encrypted image remains on this Mac; no cloud fallback ran."
            } else if captured.recognizedText.trimmingCharacters(in: .whitespacesAndNewlines).isEmpty {
                derivative = "No text derivative was recognized; the image cannot leave this Mac."
            } else {
                derivative = "Local OCR produced a reviewable text derivative; it remains local-only until you change the visible boundary."
            }
            windowCaptureReceipt = "Captured exactly one frame from the system-selected window. No cursor or audio was captured. \(derivative)"
            errorMessage = nil
        } catch WindowCaptureError.cancelled {
            guard activeWindowCaptureID == captureID, accountID == captureAccountID,
                  !isSignedOut, !isPaused else { return }
            windowCaptureReceipt = "Window selection cancelled. Nothing was captured or retained."
            errorMessage = nil
        } catch {
            guard activeWindowCaptureID == captureID, accountID == captureAccountID,
                  !isSignedOut, !isPaused else { return }
            windowCaptureReceipt = "Window capture failed before a Capsule item was created."
            fail(error)
        }
    }

    func cancelSystemSelectedWindow() {
        guard isSelectingWindow else { return }
        invalidateActiveLocalIntake()
        windowCaptureReceipt = "Window selection cancelled. Nothing was captured or retained."
        errorMessage = nil
    }

    func removeCapsuleItem(id: UUID) {
        // Exact selected-source withdrawal: only workflows frozen over this
        // source stop in the real service. Unrelated local-only additions and
        // unaffected approved operations are never invalidated.
        service.invalidateSelectedSource(id.uuidString)
        authority.invalidateSelectedSource(id.uuidString)
        capsule.remove(id: id)
        persistCapsule()
        refreshProvisionalInsight()
        mode = capsule.items.isEmpty ? .empty : .ready
    }

    func setLocalOnly(id: UUID, value: Bool) {
        service.invalidateSelectedSource(id.uuidString)
        authority.invalidateSelectedSource(id.uuidString)
        capsule.setLocalOnly(id: id, value: value)
        persistCapsule()
        refreshProvisionalInsight()
    }

    func setRetention(id: UUID, value: CapsuleRetention) {
        service.invalidateSelectedSource(id.uuidString)
        authority.invalidateSelectedSource(id.uuidString)
        capsule.setRetention(id: id, value: value)
        persistPreparedDraftRecoveryIfNeeded()
        refreshProvisionalInsight()
    }

    func setAttribution(id: UUID, actorKind: CapsuleActorKind?) {
        service.invalidateSelectedSource(id.uuidString)
        authority.invalidateSelectedSource(id.uuidString)
        capsule.setActorKind(id: id, value: actorKind)
        persistCapsule()
        refreshProvisionalInsight()
    }

    func confirmAttribution(id: UUID) {
        service.invalidateSelectedSource(id.uuidString)
        authority.invalidateSelectedSource(id.uuidString)
        capsule.confirmAttribution(id: id)
        persistCapsule()
        refreshProvisionalInsight()
    }

    func attributeFirstUnresolvedItemToCandidate() {
        guard let item = capsule.items.first(where: { !$0.hasConfirmedAttribution }) else { return }
        setAttribution(id: item.id, actorKind: .candidate)
    }

    func confirmFirstPendingAttribution() {
        guard let item = capsule.items.first(where: {
            $0.actorKind != nil && !$0.hasConfirmedAttribution
        }) else { return }
        confirmAttribution(id: item.id)
    }

    func redactCapsuleItem(id: UUID, exactTerms: [String]) {
        service.invalidateSelectedSource(id.uuidString)
        authority.invalidateSelectedSource(id.uuidString)
        let count = capsule.redact(id: id, exactTerms: exactTerms)

            persistCapsule()
            refreshProvisionalInsight()
        }
    }

    private func submitCapsule(sourceItemID: UUID?) async {
        // A superseding submission invalidates the older submission's ticket
        // without touching authentication or an unrelated approved operation.
        // The real service stops the older source workflow's next dispatch too.
        var ticket = authority.ticket("submit-capsule")
        var boundTicket: MacRelationshipAuthorityTicket?
        var frozenManifest: SubmittedContextManifest?
        var capturedRegistration: CapsuleWorkflowOwnership?
        do {
            guard scopeReviewStatus == .confirmed else {
                throw RelationshipServiceError.invalidResponse("Confirm the Pursuit, Person, and relationship context, or keep identity unresolved, before submission.")
            }
            let manifest = try capsule.freeze(
                accountID: accountID,
                pursuitID: pursuitID,
                personID: personID,
                sourceItemID: sourceItemID
            )
            // One local owner per original external key. A repeated button,
            // hotkey or reload never overwrites an in-flight/unknown owner or
            // generates a fresh key to replay paid work.
            if let existing = workflowOwnerships.first(where: {
                $0.operationID == manifest.idempotencyKey && $0.accountID == accountID
            }) {
                let currentScope = CapsuleWorkflowSubmissionScope(
                    pursuitID: manifest.pursuitID, personID: manifest.personID,
                    relationshipContextID: relationshipContextID
                )
                guard existing.record.originalSubmissionScope == currentScope else {
                    localRecoveryNotice = "This original submission belongs to a different or unverified relationship scope. Its exact owner was retained; no new capture, task or lookup was dispatched. Return to the original scope to review it."
                    return
                }
                if existing.record.phase == .admittedPreDispatch {
                    localRecoveryNotice = "The original submission is still in progress; its exact ownership was retained and nothing was dispatched again."
                } else {
                    await recoverExactTaskAdmissions()
                }
                return
            }
            authority.invalidate(.submittedIntake)
            service.invalidateAuthority(.submittedIntake)
            ticket = authority.ticket("submit-capsule")
            // The operation is bound to its exact frozen selected sources: a
            // later withdrawal of one of them stops this workflow only.
            let bound = authority.ticket(
                "submit-capsule",
                boundSources: manifest.selectedItems.map(\.sourceID)
            )
            let capturedAccountID = accountID
            // Throwing durable pre-dispatch barrier: ownership is recorded
            // before the first await and before ANY dispatch. A failed
            // encrypted write stops the submission completely.
            let registered = try registerWorkflowOwnership(for: manifest)
            capturedRegistration = registered
            guard let attemptID = registered.record.localAttemptID else {
                throw RelationshipServiceError.sourceWorkflowStopped("No Task POST was admitted without an exact local attempt identity.")
            }
            let runtimeID = sessionRuntimeID
            let observer = MacTaskLifecycleObserver(
                runtimeID: runtimeID, attemptID: attemptID, expiresAt: registered.record.expiresAt,
                observe: { [weak self] event in
                    guard let self else { throw RelationshipServiceError.staleAuthority("The original submission owner is unavailable.") }
                    try self.persistTaskLifecycle(event, manifest: manifest, ticket: bound, runtimeID: runtimeID, attemptID: attemptID)
                }
            )
            boundTicket = bound
            frozenManifest = manifest
            lastSubmittedManifest = manifest
            mode = .working
            errorMessage = nil
            let response = try await service.submit(manifest: manifest, taskLifecycle: observer)
            // Every async return checks its own ticket before landing state.
            // The registration lifecycle is settled even for an obsolete
            // landing; positive host proof is the only basis to retire it.
            settleWorkflowRegistration(registered, outcome: .success(()))
            guard authority.preservesGenerations(bound), accountID == capturedAccountID else { return }
            guard authority.preservesSelectedSources(bound),
                  capsule.matchesSubmittedSnapshot(manifest), registered.record.expiresAt > Date() else {
                fail(Self.obsoleteSubmissionError)
                return
            }
            if case .canonical(let readback) = response {
                attachWorkflowID(readback.taskID, toAttempt: registered)
            }
            apply(response)
        } catch {
            if let capturedRegistration {
                // The registration lifecycle is settled even when the UI
                // landing is obsolete: only positive host proof retires a
                // pre-dispatch registration.
                settleWorkflowRegistration(
                    capturedRegistration,
                    outcome: .failure(error)
                )
            }
            if let bound = boundTicket, let frozenManifest {
                guard authority.preservesGenerations(bound),
                      accountID == frozenManifest.accountID else { return }
                guard authority.preservesSelectedSources(bound),
                      capsule.matchesSubmittedSnapshot(frozenManifest) else {
                    fail(Self.obsoleteSubmissionError)
                    return
                }
            } else {
                guard authority.preserves(ticket) else { return }
            }
            fail(error)
        }
    }

    func confirmRelationshipScope() async {
        guard let selectedScopeOptionID,
              let option = relationshipScopeOptions.first(where: { $0.id == selectedScopeOptionID }) else {
            errorMessage = "Select one exact Pursuit, Person, and relationship context, or keep identity unresolved."
            return
        }
        guard pursuitID == option.pursuitID,
              personID == option.personID,
              relationshipContextID == option.relationshipContextID else {
            errorMessage = "The proposed relationship scope is incomplete. Keep it unresolved instead of guessing."
            return
        }
        let ticket = authority.ticket("confirm-scope")
        let capturedOptionID = option.id
        do {
            try await service.confirmScope(option.selection)
            // An old confirmation can never land after scope re-selection,
            // keep-unresolved, clear/stop, a superseding submission, sign-out,

                  scopeReviewStatus == .proposed,

                companionTrialMetrics.scopeConfirmedMilliseconds = elapsedTrialMilliseconds()
            }
            errorMessage = nil
        } catch {
            guard authority.preserves(ticket),
                  scopeReviewStatus == .proposed,
                  selectedScopeOptionID == capturedOptionID else { return }
            fail(error)
        }
    }

    func signOutAndClearLocalData() async {
        invalidateActiveLocalIntake()
        // Revoke local intake before the remote sign-out await can yield.
        isPaused = true
        focusedTodayAttentionItem = nil
        let priorAccountID = accountID
        // The same-account new-login race is closed by ordering: the captured
        // old local Capsule, reminder recovery, and key are cleared before the
        // first logout await, so a newer login's keys and receipts can only be
        // created after this cleanup has already run and no late logout
        // cleanup can blanket-delete them. Local cleanup failure stays visible
        // below instead of being hidden by the remote result.
        var reminderRecoveryDeleted = false
        var reminderRecoveryFailure: Error?
        followUpReminderService.clearLocalRecovery()
        do {
            reminderRecoveryDeleted = try reminderRecoveryStore.clear(accountID: priorAccountID)
        } catch {
            reminderRecoveryFailure = error
        }
        var localReceipt: LocalCapsuleDeletionReceipt?
        var localFailure: Error?
        do {
            localReceipt = try capsuleStore.clear(accountID: priorAccountID, deleteKey: true)
        } catch {
            localFailure = error
        }
        // An authentication boundary invalidates all old authority here and in
        // the service before the first suspension. Captured old credentials, if
        // any, are used only for this exact logout and never a new action.
        // The exact retiring credential is frozen at the synchronous
        // authentication-withdrawal boundary, before any await. The UI only
        // ever sees this opaque ticket; the bearer stays inside the service's
        // private synchronized state and is used only for this exact logout.
        let retirement = service.withdrawAuthentication()
        authority.invalidate(.authentication)
        isSignOutInProgress = true
        let ticket = authority.ticket("sign-out")

        // The retiring in-memory capsule, drafts, decisions, and receipts are
        // detached synchronously, before the first suspension: an explicit
        // sign-out clear must never be bypassed by a racing same-account load,
        // and only the captured logout/recovery identity stays outside the
        // current workspace for the tail to report on.
        let localSummary = Self.signOutLocalSummary(
            localReceipt: localReceipt,
            localFailure: localFailure,
            reminderRecoveryDeleted: reminderRecoveryDeleted,
            reminderRecoveryFailure: reminderRecoveryFailure
        )
        detachRetiringWorkspaceForSignOut(capturedAccountID: priorAccountID)
        deletionReceipt = "\(localSummary) Remote session revocation is still in flight."
        if let reminderRecoveryFailure {
            errorMessage = "Signed out locally, but reminder recovery deletion needs attention: \(reminderRecoveryFailure.localizedDescription)"
        } else if let localFailure {
            errorMessage = "Signed out locally, but local recovery deletion needs attention: \(localFailure.localizedDescription)"
        } else {
            errorMessage = nil
        }

        let outcome: SessionSignOutOutcome
        do {
            outcome = try await service.signOut(ticket: retirement)
        } catch {
            outcome = SessionSignOutOutcome(
                remote: .remoteUnknown(error.localizedDescription),
                localRetirement: .pending("The service did not report its durable operation retirement status.")
            )
        }

        // Every async return checks its own ticket. If a newer login landed
        // during the logout await it owns this workspace: the old sign-out
        // already cleared only its captured account and its own in-memory
        // retiring state, and it must not clear the later login's memory,
        // requests, keys, receipts, or its own earlier cleanup notices.
        guard authority.preservesAuthentication(ticket) else { return }

        // The remote outcome and the local durable retirement status are
        // reported independently: a verified remote revocation survives a
        // local cleanup failure, and a failed retirement is never dressed up as
        // a clear.
        let remoteSummary: String
        switch outcome.remote {
        case .revoked(let receipt):
            remoteSummary = "Remote session \(receipt.sessionID) was revoked at \(receipt.revokedAt)."
        case .localOnly(let detail):
            remoteSummary = "No original remote session credential existed on this Mac, so no logout was dispatched and nothing is claimed revoked remotely. \(detail)"
        case .remoteUnknown(let detail):
            remoteSummary = "Remote session revocation is outcome-unknown: \(detail) This Mac discarded its local session authority; reconnect to verify the server session."
        }
        let retirementSummary: String
        switch outcome.localRetirement {
        case .completed(let detail):
            retirementSummary = "Local operation retirement completed. \(detail)"
        case .pending(let detail):
            retirementSummary = "Local operation retirement is still pending and retryable: \(detail)"
        case .notRequired(let detail):
            retirementSummary = "Local operation retirement was not required. \(detail)"
        }
        deletionReceipt = "\(remoteSummary) \(localSummary) \(retirementSummary) Visible local context and relationship authority were cleared."

        if let reminderRecoveryFailure {
            errorMessage = "Signed out locally, but reminder recovery deletion needs attention: \(reminderRecoveryFailure.localizedDescription)"
        } else if let localFailure {
            errorMessage = "Signed out locally, but local recovery deletion needs attention: \(localFailure.localizedDescription)"
        } else if case .pending(let detail) = outcome.localRetirement {
            errorMessage = "Signed out locally, but durable operation retirement is still pending and retryable: \(detail)"
        } else if case .remoteUnknown = outcome.remote {
            errorMessage = "Signed out locally; remote revocation outcome is unknown: \(remoteOutcomeDescription(outcome))"
        } else if case .localOnly(let detail) = outcome.remote {
            errorMessage = "Signed out locally with no remote session to revoke: \(detail)"
        } else {
            errorMessage = nil
        }
    }

    private static func signOutLocalSummary(
        localReceipt: LocalCapsuleDeletionReceipt?,
        localFailure: Error?,
        reminderRecoveryDeleted: Bool,
        reminderRecoveryFailure: Error?
    ) -> String {

            return "Encrypted recovery deletion could not be verified."

            ? (reminderRecoveryDeleted ? "Deleted encrypted reminder recovery." : "No reminder recovery file remained.")

    }

            localDraftStatus = .awaitingDecision
            errorMessage = "No supported editable draft is available to copy. Nothing was copied or sent."
            return

            localDraftStatus = .copied
            companionTrialMetrics.completedActions.insert(.draftCopied)
            companionTrialExportReceipt = nil
            errorMessage = nil
        } else {
            localDraftStatus = .prepared
            errorMessage = "The draft remains prepared, but macOS did not confirm the clipboard write. Copy again; nothing was sent."
        }
    }

    func prepareDraft(kind: PreparedDraftKind) {
        guard let insight = provisionalInsight,
              let draft = EvidenceBoundDraftComposer.compose(kind: kind, insight: insight) else {
            let language = provisionalInsight?.language ?? .english
            let purpose = kind.title(language: language)
            errorMessage = language == .chinese
                ? "选择的证据不足以支持“\(purpose)”。系统没有准备或发送任何内容。"
                : "The selected evidence does not support a \(purpose.lowercased()). Nothing was prepared or sent."
            return

            companionTrialMetrics.actionWasEdited = true

        preparedDraftKind = draft.kind

        editableMailSubject = draft.subject
        preparedDraftBaseline = (draft.kind, draft.subject, draft.body)
        mailDraftHandoffStatus = .notOpened
        localDraftStatus = .prepared
        recordDraftPreparedMetric()
        persistPreparedDraftRecoveryIfNeeded()
        errorMessage = nil
    }

    func beginReminderPreparation(now: Date = Date()) {
        guard !hasReminderExternalRecoveryState else {
            errorMessage = "Finish checking or removing the earlier reminder before preparing another one. Its external state remains visible in Needs your review."
            return
        }
        guard let insight = provisionalInsight, insight.canPrepareAction else {
            errorMessage = "No supported follow-up signal is available for a reminder."
            return
        }
        if reminderTitle.trimmingCharacters(in: .whitespacesAndNewlines).isEmpty {
            reminderTitle = "Follow up · \(insight.change)"
        }
        if reminderDueAt <= now {
            var components = Calendar.current.dateComponents([.year, .month, .day], from: now)
            components.day = (components.day ?? 0) + 1
            components.hour = 9
            components.minute = 0
            reminderDueAt = Calendar.current.date(from: components) ?? now.addingTimeInterval(24 * 60 * 60)
        }
        reminderBaseline = (reminderTitle, reminderDueAt)
        errorMessage = nil
    }

    func loadReminderDestinationPreview() async {
        guard canPreviewReminderDestination else {
            reminderOperationState = .failed("Confirm the relationship and candidate attribution, review any existing recruiter-owned action, then provide a future date and non-empty title before previewing the destination.")
            return
        }
        let ticket = authority.ticket("reminder-destination-preview")
        let capturedAccountID = accountID
        reminderOperationState = .loadingDestination
        switch await followUpReminderService.previewDestination() {
        case .success(let destination):
            // A delayed preview is read-only: across account, sign-out, or
            // scope changes it neither lands old state nor writes recovery.
            guard reminderLandingIsCurrent(ticket, capturedAccountID: capturedAccountID) else { return }
            reminderDestination = destination
            reminderOperationState = .readyForApproval
            if let proposal = currentReminderProposal {
                _ = persistReminderRecovery(
                    proposal: proposal,
                    stage: .proposalPrepared,
                    identity: ReminderOperationIdentity(
                        accountID: capturedAccountID,
                        idempotencyKey: proposal.idempotencyKey
                    ),
                    requireExistingOperationOwnership: false,
                    updatesCurrentWorkspace: true
                )
            }
        case .failure(let failure):
            guard reminderLandingIsCurrent(ticket, capturedAccountID: capturedAccountID) else { return }
            reminderDestination = nil
            reminderOperationState = .failed(reminderFailureMessage(failure, afterWriteAttempt: false))
        }
    }

    func approveAndCreateReminder() async {
        guard let proposal = currentReminderProposal else {
            reminderOperationState = .failed("The exact reminder preview is incomplete. Review its title, date, destination, relationship, and source attribution.")
            return
        }
        let ticket = authority.ticket("reminder-create")
        // The original account and the exact reminder operation are captured
        // before any await; nothing else may own this recovery.
        let identity = ReminderOperationIdentity(
            accountID: accountID,
            idempotencyKey: proposal.idempotencyKey
        )
        guard persistReminderRecovery(
            proposal: proposal,
            stage: .executionPending,
            identity: identity,
            requireExistingOperationOwnership: false,
            updatesCurrentWorkspace: true
        ) else {
            reminderOperationState = .failed("Protected local recovery could not be saved, so nothing was created.")
            return
        }
        reminderOperationState = .executing
        switch await followUpReminderService.execute(proposal) {
        case .success(let receipt):
            guard reminderLandingIsCurrent(ticket, identity: identity) else {
                // The system write may already have happened. Keep the
                // original account's governed recovery truthful where it still
                // owns this operation, and never land old state in a newer
                // workspace or write A's source under B.
                _ = persistReminderRecovery(
                    proposal: proposal,
                    stage: .verified,
                    receipt: receipt,
                    identity: identity,
                    requireExistingOperationOwnership: true,
                    updatesCurrentWorkspace: false
                )
                return
            }
            reminderOperationState = .saved(receipt)
            _ = persistReminderRecovery(
                proposal: proposal,
                stage: .verified,
                receipt: receipt,
                identity: identity,
                requireExistingOperationOwnership: false,
                updatesCurrentWorkspace: true
            )
            companionTrialMetrics.reminderVerifiedMilliseconds = Date()
                .timeIntervalSince(companionTrialMetrics.inputAcceptedAt) * 1_000
            companionTrialMetrics.completedActions.insert(.reminderVerified)
            companionTrialExportReceipt = nil
        case .failure(let failure):
            let outcomeUnknown = switch failure {
            case .saveFailed, .readbackMissing, .readbackMismatch: true
            case .previewOnly, .permissionDenied, .noDefaultList, .destinationChanged, .recoveryUnavailable: false
            }
            guard reminderLandingIsCurrent(ticket, identity: identity) else {
                // Never report a possibly written external reminder as a
                // withdrawn write: the original account keeps its governed
                // unknown outcome where its exact operation still owns it.
                _ = persistReminderRecovery(
                    proposal: proposal,
                    stage: outcomeUnknown ? .outcomeUnknown : .proposalPrepared,
                    identity: identity,
                    requireExistingOperationOwnership: true,
                    updatesCurrentWorkspace: false
                )
                return
            }
            if outcomeUnknown {
                _ = persistReminderRecovery(
                    proposal: proposal,
                    stage: .outcomeUnknown,
                    identity: identity,
                    requireExistingOperationOwnership: false,
                    updatesCurrentWorkspace: true
                )
                reminderOperationState = .unknown(reminderFailureMessage(failure, afterWriteAttempt: true))
            } else {
                _ = persistReminderRecovery(
                    proposal: proposal,
                    stage: .proposalPrepared,
                    identity: identity,
                    requireExistingOperationOwnership: false,
                    updatesCurrentWorkspace: true
                )
                reminderOperationState = .failed(reminderFailureMessage(failure, afterWriteAttempt: true))
            }
        }
    }

    func reconcileReminderOutcome() async {
        guard !isReminderRecoveryUnreadable else {
            reminderOperationState = .unknown(
                "The protected reminder recovery cannot be opened. Inspect Apple Reminders and sign out to clear this account's local recovery before creating anything else."
            )
            return
        }
        guard let proposal = reminderProposalForReconciliation else {
            reminderOperationState = .failed("The reminder proposal changed, so the earlier outcome cannot be reconciled from this preview.")
            return
        }
        let ticket = authority.ticket("reminder-reconcile")
        let identity = ReminderOperationIdentity(
            accountID: accountID,
            idempotencyKey: proposal.idempotencyKey
        )
        reminderOperationState = .executing
        switch await followUpReminderService.reconcile(proposal) {
        case .success(let receipt):
            guard reminderLandingIsCurrent(ticket, identity: identity) else {
                _ = persistReminderRecovery(
                    proposal: proposal,
                    stage: .verified,
                    receipt: receipt,
                    identity: identity,
                    requireExistingOperationOwnership: true,
                    updatesCurrentWorkspace: false
                )
                return
            }
            reminderOperationState = .saved(receipt)
            _ = persistReminderRecovery(
                proposal: proposal,
                stage: .verified,
                receipt: receipt,
                identity: identity,
                requireExistingOperationOwnership: false,
                updatesCurrentWorkspace: true
            )
            companionTrialMetrics.reminderVerifiedMilliseconds = elapsedTrialMilliseconds()
            companionTrialMetrics.completedActions.insert(.reminderVerified)
            companionTrialExportReceipt = nil
        case .failure(.readbackMissing):
            guard reminderLandingIsCurrent(ticket, identity: identity) else {
                _ = persistReminderRecovery(
                    proposal: proposal,
                    stage: .proposalPrepared,
                    identity: identity,
                    requireExistingOperationOwnership: true,
                    updatesCurrentWorkspace: false
                )
                return
            }
            _ = persistReminderRecovery(
                proposal: proposal,
                stage: .proposalPrepared,
                identity: identity,
                requireExistingOperationOwnership: false,
                updatesCurrentWorkspace: true
            )
            reminderOperationState = .failed("No matching reminder was found in the reviewed list. The exact proposal may be tried again; the recovery reference still prevents duplicates.")
        case .failure(let failure):
            guard reminderLandingIsCurrent(ticket, identity: identity) else {
                _ = persistReminderRecovery(
                    proposal: proposal,
                    stage: .outcomeUnknown,
                    identity: identity,
                    requireExistingOperationOwnership: true,
                    updatesCurrentWorkspace: false
                )
                return
            }
            _ = persistReminderRecovery(
                proposal: proposal,
                stage: .outcomeUnknown,
                identity: identity,
                requireExistingOperationOwnership: false,
                updatesCurrentWorkspace: true
            )
            reminderOperationState = .unknown(reminderFailureMessage(failure, afterWriteAttempt: true))
        }
    }

    func removeVerifiedReminder(_ receipt: FollowUpReminderReceipt) async {
        guard let proposal = reminderProposal(for: receipt) else {
            reminderOperationState = .removalFailed(
                receipt,
                "Protected removal recovery could not be saved, so the reminder was not removed."
            )
            return
        }
        let ticket = authority.ticket("reminder-remove")
        let identity = ReminderOperationIdentity(
            accountID: accountID,
            idempotencyKey: receipt.idempotencyKey
        )
        guard persistReminderRecovery(
            proposal: proposal,
            stage: .removalPending,
            receipt: receipt,
            identity: identity,
            requireExistingOperationOwnership: false,
            updatesCurrentWorkspace: true
        ) else {
            reminderOperationState = .removalFailed(
                receipt,
                "Protected removal recovery could not be saved, so the reminder was not removed."
            )
            return
        }
        reminderOperationState = .removing(receipt)
        switch await followUpReminderService.remove(receipt) {
        case .success(let removalReceipt):
            guard reminderLandingIsCurrent(ticket, identity: identity) else {
                // Cleanup compares the exact captured operation: a stale
                // removal success clears only its own recovery and never a
                // newer operation's receipt.
                clearReminderRecovery(matching: identity, updatesCurrentWorkspace: false)
                return
            }
            reminderOperationState = .removed(removalReceipt)
            clearReminderRecovery(matching: identity)
        case .failure(.saveFailed):
            let updatesCurrentWorkspace = reminderLandingIsCurrent(ticket, identity: identity)
            _ = persistReminderRecovery(
                proposal: proposal,
                stage: .removalUnknown,
                receipt: receipt,
                identity: identity,
                requireExistingOperationOwnership: !updatesCurrentWorkspace,
                updatesCurrentWorkspace: updatesCurrentWorkspace
            )
            guard updatesCurrentWorkspace else { return }
            reminderOperationState = .removalUnknown(
                receipt,
                "Apple Reminders returned an uncertain removal result. Reconcile before trying to remove it again."
            )
        case .failure(let failure):
            let updatesCurrentWorkspace = reminderLandingIsCurrent(ticket, identity: identity)
            _ = persistReminderRecovery(
                proposal: proposal,
                stage: .verified,
                receipt: receipt,
                identity: identity,
                requireExistingOperationOwnership: !updatesCurrentWorkspace,
                updatesCurrentWorkspace: updatesCurrentWorkspace
            )
            guard updatesCurrentWorkspace else { return }
            reminderOperationState = .removalFailed(
                receipt,
                reminderRemovalFailureMessage(failure)
            )
        }
    }

    func reconcileReminderRemoval(_ receipt: FollowUpReminderReceipt) async {
        guard let proposal = reminderProposal(for: receipt) else {
            reminderOperationState = .removalFailed(
                receipt,
                "The protected reminder proposal is unavailable. Inspect Apple Reminders before taking another action."
            )
            return
        }
        let ticket = authority.ticket("reminder-removal-reconcile")
        let identity = ReminderOperationIdentity(
            accountID: accountID,
            idempotencyKey: receipt.idempotencyKey
        )
        reminderOperationState = .removing(receipt)
        switch await followUpReminderService.reconcileRemoval(receipt) {
        case .success(let removalReceipt):
            guard reminderLandingIsCurrent(ticket, identity: identity) else {
                clearReminderRecovery(matching: identity, updatesCurrentWorkspace: false)
                return
            }
            reminderOperationState = .removed(removalReceipt)
            clearReminderRecovery(matching: identity)
        case .failure(.readbackMismatch):
            let updatesCurrentWorkspace = reminderLandingIsCurrent(ticket, identity: identity)
            _ = persistReminderRecovery(
                proposal: proposal,
                stage: .verified,
                receipt: receipt,
                identity: identity,
                requireExistingOperationOwnership: !updatesCurrentWorkspace,
                updatesCurrentWorkspace: updatesCurrentWorkspace
            )
            guard updatesCurrentWorkspace else { return }
            reminderOperationState = .removalFailed(
                receipt,
                "Apple Reminders readback shows that the reminder still exists. It is safe to review and remove it again."
            )
        case .failure(let failure):
            let updatesCurrentWorkspace = reminderLandingIsCurrent(ticket, identity: identity)
            _ = persistReminderRecovery(
                proposal: proposal,
                stage: .removalUnknown,
                receipt: receipt,
                identity: identity,
                requireExistingOperationOwnership: !updatesCurrentWorkspace,
                updatesCurrentWorkspace: updatesCurrentWorkspace
            )
            guard updatesCurrentWorkspace else { return }
            reminderOperationState = .removalUnknown(
                receipt,
                reminderRemovalFailureMessage(failure)
            )
        }
    }

              proposal.idempotencyKey == receipt.idempotencyKey else {

    private func reminderFailureMessage(_ failure: FollowUpReminderFailure, afterWriteAttempt: Bool) -> String {
        switch failure {
        case .previewOnly:
            return "Synthetic preview never writes to Apple Reminders. Connect a live workspace to create a real reminder."
        case .permissionDenied:
            return "Apple Reminders access was not granted. Nothing was created."
        case .noDefaultList:
            return "Apple Reminders has no default destination list. Choose a default list, then preview again."
        case .destinationChanged:
            return "The reviewed Reminders list is no longer available. Nothing was created; preview the destination again."
        case .recoveryUnavailable:
            return "The local duplicate-prevention reference could not be verified, so nothing was created."
        case .saveFailed:
            return afterWriteAttempt
                ? "Apple Reminders returned an uncertain write result. Check or reconcile before trying again."
                : "The Reminders destination could not be read. Nothing was created."
        case .readbackMissing:
            return "The write returned, but the reminder could not be read back. Its outcome is unknown; reconcile before trying again."
        case .readbackMismatch:
            return "The saved reminder did not match the approved title, date, or destination. Treat the outcome as unknown and inspect Apple Reminders."
        }
    }

    private func reminderRemovalFailureMessage(_ failure: FollowUpReminderFailure) -> String {
        switch failure {

            "Synthetic preview has no Apple Reminder to remove."
        case .permissionDenied:
            "Apple Reminders access is unavailable, so the verified reminder was not removed."
        case .noDefaultList:
            "The Reminders destination is unavailable; the verified reminder was not removed."
        case .destinationChanged:
            "The original Reminders list is no longer available. Inspect Apple Reminders before taking another action."
        case .recoveryUnavailable:
            "The reminder recovery reference is unavailable. Inspect Apple Reminders before taking another action."
        case .saveFailed:
            "Apple Reminders returned an uncertain removal result. Reconcile before trying again."
        case .readbackMissing:
            "The reminder could not be read back after removal. Reconcile before trying again."
        case .readbackMismatch:
            "The reminder identity did not match the verified receipt, so nothing was removed."
        }
    }

    func saveIdentityForReview() {
        guard mode == .ambiguousIdentity else { return }
        identityReviewReceipt = IdentityReviewReceipt(
            id: "identity-review-\(UUID().uuidString.lowercased())",
            taskID: currentTaskID,
            summary: isSyntheticFixture
                ? "Synthetic identity review saved. No person was bound and no fact or action authority was granted."
                : "The canonical Agent Task remains unresolved. No person was bound and no fact or action authority was granted.",
            occurredAt: Date()

    }

    func setDecision(itemID: String, choice: CanonicalDecisionChoice) {
        guard pendingDecision?.items.contains(where: { $0.id == itemID }) == true else { return }
        decisionSelections[itemID] = choice
    }

    func decideNextCanonicalItem(_ choice: CanonicalDecisionChoice) {
        guard let item = pendingDecision?.items.first(where: { decisionSelections[$0.id] == nil }) else { return }
        setDecision(itemID: item.id, choice: choice)
    }

    var canResolveCanonicalDecision: Bool {
        guard let pendingDecision else { return false }
        return !pendingDecision.items.isEmpty &&
            pendingDecision.items.allSatisfy { decisionSelections[$0.id] != nil }
    }

    func resolveCanonicalDecision() async {
        guard let pendingDecision, canResolveCanonicalDecision else {
            errorMessage = "Choose Confirm, Reject, or Keep unresolved for every proposal item."
            return
        }
        let decisions = pendingDecision.items.compactMap { item in
            decisionSelections[item.id].map {
                CanonicalDecisionRequest.Decision(itemID: item.id, choice: $0)
            }
        }
        // The exact decision operation identity. A late decision callback can
        // only land over the exact bundle, task, and revisions CURRENT state
        // still holds; a superseding submission or replacement drops it.
        let capturedReview = pendingDecision
        let ticket = authority.ticket("resolve-decision")
        do {
            mode = .working
            errorMessage = nil
            let response = try await service.resolveDecision(.init(
                bundleID: capturedReview.bundleID,
                taskID: capturedReview.taskID,
                taskRevision: capturedReview.taskRevision,
                bundleRevision: capturedReview.bundleRevision,
                proposalID: capturedReview.proposalID,
                baseRevision: capturedReview.baseRevision,
                reason: "The recruiter reviewed exact evidence, before and proposed values, effect, and authority in capri for Mac.",
                decisions: decisions
            ))
            guard authority.preserves(ticket),
                  self.pendingDecision?.bundleID == capturedReview.bundleID,
                  self.pendingDecision?.taskID == capturedReview.taskID,
                  self.pendingDecision?.taskRevision == capturedReview.taskRevision,
                  self.pendingDecision?.bundleRevision == capturedReview.bundleRevision else { return }
            apply(response)
        } catch {
            guard authority.preserves(ticket),
                  self.pendingDecision?.bundleID == capturedReview.bundleID,
                  self.pendingDecision?.taskID == capturedReview.taskID,
                  self.pendingDecision?.taskRevision == capturedReview.taskRevision,
                  self.pendingDecision?.bundleRevision == capturedReview.bundleRevision else { return }
            fail(error)
        }
    }

    func reconcileCanonicalDecision() async {
        let ticket = authority.ticket("reconcile-decision")
        let capturedAccountID = accountID
        do {
            mode = .working
            errorMessage = nil
            let response = try await service.reconcileDecisionOutcome()
            guard authority.preserves(ticket), accountID == capturedAccountID else { return }
            apply(response)
        } catch {
            guard authority.preserves(ticket), accountID == capturedAccountID else { return }
            fail(error)
        }
    }

            presentation = scope.presentation
            pendingDecision = nil
            runAudit = nil
            canonicalReceipt = nil
            identityReviewReceipt = nil
            decisionSelections = [:]
            scopeReviewStatus = .proposed
            mode = .ready
        case .syntheticFixture(let fixture):
            let syntheticOption = Self.syntheticScopeOptionForCurrentProcess
            let preflightPreview = ProcessInfo.processInfo.arguments.contains("--consequence-preflight-preview")
            isSyntheticFixture = true
            isSignedOut = false
            mode = fixture.mode

                : fixture.presentation
            factReviewStatus = .proposed
            editableFollowUpDraft = ""
            editableMailSubject = ""
            preparedDraftKind = nil
            mailDraftHandoffStatus = .notOpened
            localDraftStatus = .awaitingDecision
            activeReminderRecovery = nil
            isReminderRecoveryUnreadable = false
            reminderRecoveryNotice = nil
            pendingDecision = fixture.pendingDecision
            canonicalReceipt = fixture.receipt
            runAudit = nil
            identityReviewReceipt = fixture.mode == .identityReviewSaved
                ? FixtureRelationshipService.identityReviewReceiptFixture()
                : nil
            decisionSelections = [:]
            scopeReviewStatus = [.empty, .ready].contains(fixture.mode) ? .proposed : .confirmed
            relationshipScopeOptions = [syntheticOption]
            todayAttention = showsSyntheticTodayPreview
                ? Self.syntheticTodayAttention(for: fixture.mode)
                : .empty
            selectedScopeOptionID = preflightPreview || ![.empty, .ready].contains(fixture.mode)
                ? syntheticOption.id
                : nil
            reminderDuplicateActionDecision = Self.duplicateActionDecision(
                for: selectedScopeOptionID == nil ? nil : syntheticOption.consequencePreflight
            )
            if selectedScopeOptionID != nil {
                pursuitID = syntheticOption.pursuitID
                personID = syntheticOption.personID
                relationshipContextID = syntheticOption.relationshipContextID
            }
            recordRelationshipCompletionMetricIfNeeded(mode: fixture.mode)
        case .canonical(let readback):
            guard readback.provesCanonicalSafeReadback else {
                fail(RelationshipServiceError.canonicalReadbackIncomplete)
                return
            }
            // Canonical-first recovery shares the exact account/runtime
            // transition and ownership adoption BEFORE terminal cleanup.
            admitAuthenticatedResponse(accountID: readback.accountID)
            pursuitID = readback.pursuitID
            personID = readback.personID
            relationshipContextID = readback.relationshipContextID
            currentTaskID = readback.taskID
            presentation = readback.presentation
            mode = readback.displayMode
            pendingDecision = readback.pendingDecision
            canonicalReceipt = readback.receipt
            runAudit = readback.runAudit
            identityReviewReceipt = nil
            decisionSelections = [:]
            scopeReviewStatus = .confirmed
            relationshipScopeOptions = relationshipScopeOptions.filter {
                $0.pursuitID == readback.pursuitID &&
                    $0.personID == readback.personID &&
                    $0.relationshipContextID == readback.relationshipContextID
            }
            selectedScopeOptionID = relationshipScopeOptions.first?.id
            resetReminderDuplicateActionDecisionForSelectedScope()
            recordRelationshipCompletionMetricIfNeeded(mode: readback.displayMode)
            if [.noAction, .receipt, .failed].contains(readback.displayMode) {
                // Exact workflow-owned taskOnly cleanup: only the retiring
                // workflow's unamended, still-owned copies are removed. New
                // attachments, amended sources, and sources a newer operation
                // owns all survive; unrelated local-only additions are never
                // erased and cleanup is never gated on global capsule version.
                retireWorkflowOwnedTaskOnlyItems(
                    workflowID: readback.taskID,
                    accountID: readback.accountID
                )
            }
        }
    }

    private var hasReminderExternalRecoveryState: Bool {

    /// Compare-before-write by captured account and exact operation identity.
    /// A late callback must never write its evidence or proposal into a newer
    /// account's or a newer operation's encrypted recovery, and a recovery
    /// already cleared at sign-out is never recreated. The original account's
    /// governed recovery is kept only while that exact operation still owns it,
    /// so truthful reconciliation survives a replacement without ever landing
    /// old state in the newer workspace.
    @discardableResult
    private func persistReminderRecovery(
        proposal: FollowUpReminderProposal,
        stage: ReminderOperationRecoveryStage,
        receipt: FollowUpReminderReceipt? = nil,
        identity: ReminderOperationIdentity,
        requireExistingOperationOwnership: Bool,
        updatesCurrentWorkspace: Bool,
        now: Date = Date()
    ) -> Bool {
        let stored = try? reminderRecoveryStore.load(accountID: identity.accountID, now: now)
        if let stored, stored.proposal.idempotencyKey != identity.idempotencyKey {
            return false
        }
        if stored == nil, requireExistingOperationOwnership {
            return false
        }
        let recovery = ReminderOperationRecovery(
            proposal: proposal,
            stage: stage,
            receipt: receipt,
            now: now
        )
        do {
            try reminderRecoveryStore.save(recovery, accountID: identity.accountID)
            if updatesCurrentWorkspace {
                activeReminderRecovery = recovery
                isReminderRecoveryUnreadable = false
                reminderRecoveryNotice = nil
            }
            return true
        } catch {
            if updatesCurrentWorkspace {
                reminderRecoveryNotice = "The exact reminder recovery could not be protected on this Mac: \(error.localizedDescription)"
            }
            return false
        }
    }

    /// Compare-before-delete by exact operation identity: an old cleanup can
    /// never delete a newer operation's recovery receipt, and a stale defer or
    /// error path can never remove newly saved recovery.
    private func clearReminderRecovery(
        matching identity: ReminderOperationIdentity,
        updatesCurrentWorkspace: Bool = true
    ) {
        let stored = try? reminderRecoveryStore.load(accountID: identity.accountID, now: Date())
        if let stored {
            guard stored.proposal.idempotencyKey == identity.idempotencyKey else { return }
            do {
                _ = try reminderRecoveryStore.clear(accountID: identity.accountID)
            } catch {
                if updatesCurrentWorkspace {
                    reminderRecoveryNotice = "The local reminder recovery could not be deleted: \(error.localizedDescription)"
                }
                return
            }
        }
        guard updatesCurrentWorkspace, accountID == identity.accountID else { return }
        activeReminderRecovery = nil
        isReminderRecoveryUnreadable = false
        reminderRecoveryNotice = nil
    }

    /// True only while the captured account may land reminder state in the
    /// current workspace UI. Delayed responses across account replacement,
    /// sign-out, or scope changes never land old state in a newer workspace
    /// and never persist the old account's source under the new account.
    private func reminderLandingIsCurrent(
        _ ticket: MacRelationshipAuthorityTicket,
        capturedAccountID: String
    ) -> Bool {
        authority.preservesAuthority(ticket) &&
            accountID == capturedAccountID &&
            !isSignedOut &&
            !isSignOutInProgress
    }

    /// Extends the account landing check with the exact operation: an older
    /// callback must never replace a newer reminder operation's UI state, and a
    /// storage compare-write that refuses an old operation must also prevent
    /// that old operation's UI success from landing.
    private func reminderLandingIsCurrent(
        _ ticket: MacRelationshipAuthorityTicket,
        identity: ReminderOperationIdentity
    ) -> Bool {
        guard reminderLandingIsCurrent(ticket, capturedAccountID: identity.accountID) else {
            return false
        }
        if let active = activeReminderRecovery,
           active.proposal.idempotencyKey != identity.idempotencyKey {
            return false
        }
        let stored = try? reminderRecoveryStore.load(accountID: identity.accountID, now: Date())
        if let stored, stored.proposal.idempotencyKey != identity.idempotencyKey {
            return false
        }
        return true
    }

    private func currentReminderOperationIdentity() -> ReminderOperationIdentity? {
        if let recovery = activeReminderRecovery {
            return ReminderOperationIdentity(
                accountID: accountID,
                idempotencyKey: recovery.proposal.idempotencyKey
            )
        }
        return currentReminderProposal.map {
            ReminderOperationIdentity(accountID: accountID, idempotencyKey: $0.idempotencyKey)
        }
    }

    private func recoverReminderOperationIfNeeded(accountID: String, now: Date = Date()) {
        guard !recoveredReminderAccountIDs.contains(accountID) else { return }
        recoveredReminderAccountIDs.insert(accountID)
        do {
            guard let recovery = try reminderRecoveryStore.load(accountID: accountID, now: now) else {
                isReminderRecoveryUnreadable = false
                return
            }
            activeReminderRecovery = recovery
            isReminderRecoveryUnreadable = false
            reminderTitle = recovery.title
            reminderDueAt = recovery.dueAt
            reminderBaseline = (recovery.title, recovery.dueAt)
            reminderDestination = .init(
                identifier: recovery.destinationIdentifier,
                title: recovery.destinationTitle
            )
            switch recovery.stage {
            case .proposalPrepared:
                reminderDestination = nil
                reminderOperationState = .notPrepared
                reminderRecoveryNotice = "Recovered the exact reminder draft. Preview its Apple Reminders destination again before approval."
            case .executionPending, .outcomeUnknown:
                reminderOperationState = .unknown(
                    "capri found an unfinished reminder operation after relaunch. Check Apple Reminders before creating anything else."
                )
                reminderRecoveryNotice = "Recovered an unfinished reminder check. capri will reconcile the original operation instead of retrying it."
            case .verified:
                guard let receipt = recovery.receipt else { return }
                reminderOperationState = .saved(receipt)
                reminderRecoveryNotice = "Recovered a verified Apple Reminder and its removal path."
            case .removalPending, .removalUnknown:
                guard let receipt = recovery.receipt else { return }
                reminderOperationState = .removalUnknown(
                    receipt,
                    "capri found an unfinished reminder removal after relaunch. Check the original reminder before trying again."
                )
                reminderRecoveryNotice = "Recovered an unfinished reminder removal. capri will verify absence before claiming it was removed."
            }
        } catch {
            activeReminderRecovery = nil
            isReminderRecoveryUnreadable = true
            reminderOperationState = .unknown(
                "The protected reminder recovery cannot be opened. Inspect Apple Reminders before creating anything else."
            )
            reminderRecoveryNotice = "Encrypted reminder recovery could not be opened. New reminder writes remain blocked for this account."
        }
    }

    /// Records exact account, runtime, operation, and workflow ownership over
    /// the frozen selected sources of one submission.
    /// Registers exact account/workflow/operation/source ownership at
    /// freeze/admission, BEFORE the first await, so a newer in-flight operation
    /// already protects its selected source from an older terminal workflow.
    /// The minimal record is persisted in the same encrypted Capsule payload so
    /// restart terminal and reconciliation cleanup can still target exactly
    /// this workflow; the canonical workflow ID is attached when the readback
    /// returns. A failed persistence keeps the on-disk record intact, so
    /// cleanup retry authority survives failure and restart.
    /// Throwing durable PRE-DISPATCH barrier. If the encrypted ownership write
    /// fails, nothing is submitted to the service and the failure notice is
    /// retained: no model, task, or service dispatch may proceed without this
    /// exact recovery authority on disk.
    @discardableResult
    private func registerWorkflowOwnership(
        for manifest: SubmittedContextManifest
    ) throws -> CapsuleWorkflowOwnership {
        guard var record = capsule.capturedWorkflowOwnershipRecord(
            accountID: manifest.accountID,
            operationID: manifest.idempotencyKey,
            selectedSourceIDs: manifest.selectedItems.compactMap({ UUID(uuidString: $0.sourceID) }),
            now: manifest.submittedAt
        ) else {
            throw RelationshipServiceError.invalidResponse(
                "The frozen submission lost a selected source before its exact ownership could be recorded."
            )
        }
        record.localAttemptID = UUID()
        record.originalSubmissionScope = CapsuleWorkflowSubmissionScope(
            pursuitID: manifest.pursuitID, personID: manifest.personID,
            relationshipContextID: relationshipContextID
        )
        let ownership = CapsuleWorkflowOwnership(record: record, runtimeID: sessionRuntimeID)
        var updated = capsule
        updated.workflowOwnerships = (updated.workflowOwnerships ?? []).filter {
            $0.operationID != record.operationID
        }
        updated.workflowOwnerships?.append(record)
        try capsuleStore.save(updated, accountID: accountID, now: Date())
        capsule = updated
        workflowOwnerships.removeAll { $0.operationID == record.operationID }
        workflowOwnerships.append(ownership)
        return ownership
    }

    /// Both lifecycle barriers use the original operation owner and exact
    /// source copies. No unrelated Capsule version churn invalidates them.
    private func persistTaskLifecycle(
        _ event: MacTaskLifecycleEvent,
        manifest: SubmittedContextManifest,
        ticket: MacRelationshipAuthorityTicket,
        runtimeID: UUID,
        attemptID: UUID
    ) throws {
        let reference: MacTaskAdmissionReference
        let taskID: String
        let phase: CapsuleWorkflowPhase
        switch event {
        case .prepared(let value): reference = value; taskID = ""; phase = .taskOutcomeUnknown
        case .accepted(let value, let id): reference = value; taskID = id; phase = .canonicalTask
        }
        let admitted = try authority.synchronized {
            guard authority.preserves(ticket), sessionRuntimeID == runtimeID,
                  accountID == manifest.accountID, !isSignedOut,
                  capsule.matchesSubmittedSnapshot(manifest),
                  reference.accountID == accountID,
                  reference.operationID == manifest.idempotencyKey,
                  reference.pursuitID == manifest.pursuitID,
                  reference.originalRuntimeID == runtimeID,
                  reference.originalAttemptID == attemptID,
                  reference.expiresAt > Date(),
                  let index = workflowOwnerships.firstIndex(where: {
                      $0.operationID == manifest.idempotencyKey && $0.runtimeID == runtimeID && $0.record.localAttemptID == attemptID
                  }),
                  let diskIndex = capsule.workflowOwnerships?.firstIndex(where: {
                      $0.operationID == manifest.idempotencyKey && $0.localAttemptID == attemptID
                  }) else { return false }
            let previous = workflowOwnerships[index].record
            guard reference.expiresAt == previous.expiresAt,
                  previous.sources.allSatisfy({ binding in capsule.items.contains { binding.matchesCurrentState(of: $0) } }),
                  previous.taskAdmission == nil || previous.taskAdmission == reference,
                  previous.workflowID.isEmpty || previous.workflowID == taskID else { return false }
            if case .accepted = event, previous.taskAdmission != reference { return false }
            let record = CapsuleWorkflowOwnershipRecord(
                accountID: previous.accountID, workflowID: taskID,
                operationID: previous.operationID, phase: phase, taskAdmission: reference, localAttemptID: previous.localAttemptID,
                originalSubmissionScope: previous.originalSubmissionScope,
                sources: previous.sources, capturedAt: previous.capturedAt, expiresAt: previous.expiresAt
            )
            var updated = capsule
            updated.workflowOwnerships?[diskIndex] = record
            try capsuleStore.save(updated, accountID: accountID, now: Date())
            capsule = updated
            workflowOwnerships[index] = CapsuleWorkflowOwnership(record: record, runtimeID: runtimeID)
            return true
        }
        guard admitted else {
            throw RelationshipServiceError.staleAuthority("The original task lifecycle owner, source or authorization is obsolete. No old correlation is recreated.")
        }
    }

    /// Recovery only reads exact source-free admissions. Original provenance
    /// stays immutable; this legitimate runtime adopts a new current ticket.
    private func recoverExactTaskAdmissions() async {
        let ticket = authority.ticket("recover-exact-task-admissions")
        let runtimeID = sessionRuntimeID
        let ownerAccountID = accountID
        let references = workflowOwnerships.compactMap { $0.record.taskAdmission }
        for reference in references {
            guard authority.preservesAuthentication(ticket), sessionRuntimeID == runtimeID,
                  accountID == ownerAccountID, !isSignedOut else { return }
            guard reference.accountID == ownerAccountID, reference.expiresAt > Date(),
                  workflowOwnerships.contains(where: {
                      $0.record.taskAdmission == reference && $0.runtimeID == runtimeID &&
                          $0.record.localAttemptID == reference.originalAttemptID
                  }) else { continue }
            do {
                let readback = try await service.lookupTaskAdmission(reference)
                guard authority.preservesAuthentication(ticket), sessionRuntimeID == runtimeID,
                      accountID == ownerAccountID, !isSignedOut, reference.expiresAt > Date(),
                      let index = workflowOwnerships.firstIndex(where: {
                          $0.record.taskAdmission == reference && $0.runtimeID == runtimeID &&
                              $0.record.localAttemptID == reference.originalAttemptID
                      }),
                      capsule.workflowOwnerships?.contains(where: { $0.taskAdmission == reference && $0.localAttemptID == reference.originalAttemptID }) == true else { continue }
                guard readback.admissionState == .accepted, let taskID = readback.taskID else {
                    localRecoveryNotice = "The original task admission remains \(readback.admissionState.rawValue). No task was retried and the original lifetime was preserved."
                    continue
                }
                let previous = workflowOwnerships[index].record
                guard previous.workflowID.isEmpty || previous.workflowID == taskID else {
                    throw RelationshipServiceError.invalidResponse("Exact task admission changed identity; no association or cleanup was applied.")
                }
                let record = CapsuleWorkflowOwnershipRecord(
                    accountID: previous.accountID, workflowID: taskID,
                    operationID: previous.operationID, phase: .canonicalTask, taskAdmission: reference, localAttemptID: previous.localAttemptID,
                    originalSubmissionScope: previous.originalSubmissionScope,
                    sources: previous.sources, capturedAt: previous.capturedAt, expiresAt: previous.expiresAt
                )
                var updated = capsule
                guard let diskIndex = updated.workflowOwnerships?.firstIndex(where: { $0.taskAdmission == reference && $0.localAttemptID == reference.originalAttemptID }) else { continue }
                updated.workflowOwnerships?[diskIndex] = record
                try capsuleStore.save(updated, accountID: ownerAccountID, now: Date())
                capsule = updated
                workflowOwnerships[index] = CapsuleWorkflowOwnership(record: record, runtimeID: runtimeID)
                localRecoveryNotice = "Recovered exact task \(taskID) by its original admission key. No source or effect authority was restored and no task was retried."
                if readback.taskStatus?.isTerminal == true {
                    retireWorkflowOwnedTaskOnlyItems(workflowID: taskID, accountID: ownerAccountID)
                }
            } catch {
                guard authority.preservesAuthentication(ticket), sessionRuntimeID == runtimeID,
                      accountID == ownerAccountID, !isSignedOut else { return }
                localRecoveryNotice = "Original task admission remains unverified and was not replayed: \(error.localizedDescription)"
            }
        }
    }

    /// Attaches the canonical workflow identity returned by the readback to the
    /// operation recorded at admission.
    /// Durably attaches the canonical task/operation association BEFORE the
    /// authority transition it enables. A failed save keeps the exact
    /// recoverable pending correlation and its original lifetime: the task is
    /// never reported canceled and its authority is never deleted. Restart
    /// reconciliation resolves the already-created task by exact operation
    /// identity with no automatic POST retry.
    private func attachWorkflowID(_ workflowID: String, toAttempt captured: CapsuleWorkflowOwnership) {
        let operationID = captured.operationID
        guard let index = workflowOwnerships.firstIndex(where: {
            $0.operationID == operationID && $0.accountID == captured.accountID &&
                $0.runtimeID == captured.runtimeID && $0.record.localAttemptID == captured.record.localAttemptID
        }), let recordIndex = capsule.workflowOwnerships?.firstIndex(where: {
            $0.operationID == operationID && $0.localAttemptID == captured.record.localAttemptID
        }) else {
            return
        }
        let previous = workflowOwnerships[index]
        if previous.record.workflowID == workflowID && previous.record.phase == .canonicalTask { return }
        let record = CapsuleWorkflowOwnershipRecord(
            accountID: previous.record.accountID,
            workflowID: workflowID,
            operationID: previous.record.operationID,
            phase: .canonicalTask,
            taskAdmission: previous.record.taskAdmission,
            localAttemptID: previous.record.localAttemptID,
            originalSubmissionScope: previous.record.originalSubmissionScope,
            sources: previous.record.sources,
            capturedAt: previous.record.capturedAt,
            expiresAt: previous.record.expiresAt
        )
        var updated = capsule
        updated.workflowOwnerships?[recordIndex] = record
        do {
            try capsuleStore.save(updated, accountID: accountID, now: Date())
            capsule = updated
            workflowOwnerships[index] = CapsuleWorkflowOwnership(record: record, runtimeID: previous.runtimeID)
        } catch {
            // The exact pending task/operation correlation stays recoverable in
            // memory and its failure stays visible until it is persisted.
            workflowOwnerships[index] = CapsuleWorkflowOwnership(record: record, runtimeID: previous.runtimeID)
            localRecoveryNotice = "The canonical task association \(workflowID) could not be protected on this Mac yet; its exact operation \(operationID) stays recoverable and is never retried blindly: \(error.localizedDescription)"
        }
    }

    /// Settles one workflow registration's lifecycle from its real outcome,
    /// even when the UI landing itself is obsolete. A superseded pre-dispatch
    /// registration is retired ONLY on positive host proof that no task was
    /// dispatched or accepted; an entered or unknown task keeps its exact
    /// correlation and is never presumed canceled.
    private func settleWorkflowRegistration(_ captured: CapsuleWorkflowOwnership, outcome: Result<Void, Error>) {
        let operationID = captured.operationID
        guard let index = workflowOwnerships.firstIndex(where: {
            $0.operationID == operationID && $0.accountID == captured.accountID &&
                $0.runtimeID == captured.runtimeID && $0.record.localAttemptID == captured.record.localAttemptID
        }), capsule.workflowOwnerships?.contains(where: {
            $0.operationID == operationID && $0.localAttemptID == captured.record.localAttemptID
        }) == true else {
            return
        }
        switch outcome {
        case .success:
            return
        case .failure(let error):
            guard case RelationshipServiceError.sourceWorkflowStopped = error else {
                // Unknown dispatch: keep exact operation/task idempotency
                // correlation for governed read reconciliation.
                let previous = workflowOwnerships[index]
                guard previous.record.phase == .admittedPreDispatch else { return }
                let unknownRecord = CapsuleWorkflowOwnershipRecord(
                    accountID: previous.record.accountID,
                    workflowID: previous.record.workflowID,
                    operationID: previous.record.operationID,
                    phase: .taskOutcomeUnknown,
                    taskAdmission: previous.record.taskAdmission,
                    localAttemptID: previous.record.localAttemptID,
                    originalSubmissionScope: previous.record.originalSubmissionScope,
                    sources: previous.record.sources,
                    capturedAt: previous.record.capturedAt,
                    expiresAt: previous.record.expiresAt
                )
                workflowOwnerships[index] = CapsuleWorkflowOwnership(
                    record: unknownRecord,
                    runtimeID: previous.runtimeID
                )
                var updated = capsule
                if let recordIndex = updated.workflowOwnerships?.firstIndex(where: { $0.operationID == operationID && $0.localAttemptID == captured.record.localAttemptID }) {
                    updated.workflowOwnerships?[recordIndex] = unknownRecord
                }
                do {
                    try capsuleStore.save(updated, accountID: accountID, now: Date())
                    capsule = updated
                } catch {
                    localRecoveryNotice = "The unknown task outcome for operation \(operationID) could not be protected yet; it stays recoverable and is never retried blindly: \(error.localizedDescription)"
                }
                return
            }
            // Caller nondispatch never negates a canonical association already
            // read for this original key. Keep its exact task retirement owner.
            guard workflowOwnerships[index].record.phase != .canonicalTask else { return }
            // Positive host proof applies only to this exact local attempt.
            let retiring = workflowOwnerships.remove(at: index)
            var updated = capsule
            updated.workflowOwnerships = (updated.workflowOwnerships ?? []).filter {
                $0.operationID != retiring.operationID || $0.localAttemptID != captured.record.localAttemptID
            }
            if updated.workflowOwnerships?.isEmpty == true {
                updated.workflowOwnerships = nil
            }
            do {
                try capsuleStore.save(updated, accountID: accountID, now: Date())
                capsule = updated
            } catch {
                // Keep exact retry authority: the registration stays pending on
                // disk and its failure stays visible until successfully
                // persisted.
                workflowOwnerships.insert(retiring, at: index)
                localRecoveryNotice = "The proven-never-dispatched registration \(retiring.operationID) could not be retired on disk yet; retry remains available: \(error.localizedDescription)"
            }
        }
    }

    /// Explicit legitimate recovery adoption: only the runtime that recovered
    /// this account's encrypted payload adopts its persisted ownership, and a
    /// sign-out/new session never adopts retired ownership.
    private func adoptPersistedWorkflowOwnerships() {
        guard let records = capsule.workflowOwnerships else { return }
        for record in records where record.accountID == accountID {
            guard !workflowOwnerships.contains(where: { $0.operationID == record.operationID }) else { continue }
            workflowOwnerships.append(
                CapsuleWorkflowOwnership(record: record, runtimeID: sessionRuntimeID)
            )
        }
    }

    /// Terminal cleanup is scoped to the retiring workflow's captured source
    /// ownership. A different account or runtime, an amended source, a source a
    /// newer operation still owns, and every unrelated item survive; cleanup
    /// failure stays visible through the normal local recovery reporting and
    /// the spent record is retired in the same encrypted save, so a failed save
    /// keeps restart retry authority on disk.
    private func retireWorkflowOwnedTaskOnlyItems(
        workflowID: String,
        accountID readbackAccountID: String
    ) {
        guard readbackAccountID == accountID,
              let index = workflowOwnerships.firstIndex(where: {
                  $0.workflowID == workflowID &&
                      $0.accountID == accountID &&
                      $0.runtimeID == sessionRuntimeID
              }) else {
            return
        }
        let retiring = workflowOwnerships.remove(at: index)
        let surviving = workflowOwnerships.filter { $0.operationID != retiring.operationID }
        let previousCapsule = capsule
        var updated = capsule
        let removed = updated.retireWorkflowTaskOnlyItems(
            ownedBy: retiring,
            survivingOwnerships: surviving
        )
        updated.workflowOwnerships = (updated.workflowOwnerships ?? []).filter {
            $0.operationID != retiring.operationID
        }
        if updated.workflowOwnerships?.isEmpty == true {
            updated.workflowOwnerships = nil
        }
        do {
            // Terminal save failure retains exact local retry authority: the
            // in-memory retirement is rolled back and stays visible until the
            // save succeeds; it is never hidden as ordinary success.
            try capsuleStore.save(updated, accountID: accountID, now: Date())
            capsule = updated
        } catch {
            capsule = previousCapsule
            workflowOwnerships.insert(retiring, at: index)
            localRecoveryNotice = "Terminal taskOnly cleanup for workflow \(workflowID) could not be persisted; exact retry authority is retained: \(error.localizedDescription)"
            return
        }
        if removed > 0 {
            refreshProvisionalInsight()
            localRecoveryNotice = "Deleted \(removed) task-only local Capsule item\(removed == 1 ? "" : "s") after the canonical task ended."
        }
    }

    /// Shared authenticated-response admission for BOTH connected and
    /// canonical-first responses: the exact account/runtime transition and
    /// encrypted Capsule ownership adoption happen BEFORE any terminal
    /// retirement, and a retired login never adopts old ownership. Canonical
    /// recovery therefore works with the placeholder bootstrap account too.
    private func admitAuthenticatedResponse(accountID responseAccountID: String) {
        let priorAccountID = accountID
        let crossesAuthenticationBoundary = priorAccountID != responseAccountID ||
            isSignedOut || isSignOutInProgress
        if crossesAuthenticationBoundary {
            // A new authenticated session (account change or a login that
            // supersedes a racing sign-out) is an authentication boundary:
            // every older ticket dies here, and the new session gets a
            // fresh runtime identity so an old workflow can never clean up
            // its local items. The service's own fence was already
            // invalidated by the racing sign-out; a legitimate login that
            // this very load established is preserved.
            authority.invalidate(.authentication)
            sessionRuntimeID = UUID()
            workflowOwnerships.removeAll()
            isSignOutInProgress = false
            isPaused = false
        }
        if priorAccountID != responseAccountID {
            invalidateActiveLocalIntake()
            // An authenticated account change is an authority boundary.
            // Never carry the previous account's in-memory Capsule,
            // manifest, decisions, or receipts into the new workspace.
            capsule = ContextCapsuleDraft()
            provisionalInsight = nil
            editableFollowUpDraft = ""
            editableMailSubject = ""
            preparedDraftKind = nil
            mailDraftHandoffStatus = .notOpened
            localDraftStatus = .awaitingDecision
            resetReminderPreparation()
            activeReminderRecovery = nil
            isReminderRecoveryUnreadable = false
            reminderRecoveryNotice = nil
            reminderDuplicateActionDecision = .unavailable
            companionTrialMetrics = CompanionTrialMetrics()
            companionTrialExportReceipt = nil
            quickPanelNavigationRequest = nil
            lastSubmittedManifest = nil
            pendingDecision = nil
            runAudit = nil
            canonicalReceipt = nil
            identityReviewReceipt = nil
            decisionSelections = [:]
            localRecoveryNotice = nil
            deletionReceipt = nil
            recoveredAccountIDs.remove(responseAccountID)
            recoveredReminderAccountIDs.remove(responseAccountID)
        }
        isSyntheticFixture = false
        isSignedOut = false
        accountID = responseAccountID
        followUpReminderService.setLocalRecoveryAccount(responseAccountID)
        recoverCapsuleIfNeeded(accountID: responseAccountID)
        recoverReminderOperationIfNeeded(accountID: responseAccountID)
        adoptPersistedWorkflowOwnerships()
    }

    private func persistCapsule() {
        do {
            try capsuleStore.save(capsule, accountID: accountID, now: Date())
        } catch {
            errorMessage = "Local recovery was not updated: \(error.localizedDescription)"
        }
    }

    private func recoverCapsuleIfNeeded(accountID: String) {
        guard !recoveredAccountIDs.contains(accountID), capsule.items.isEmpty else { return }
        recoveredAccountIDs.insert(accountID)
        do {
            let recovery = try capsuleStore.load(accountID: accountID, now: Date())
            capsule = recovery.draft
            adoptPersistedWorkflowOwnerships()
            refreshProvisionalInsight()
            if !capsule.items.isEmpty {
                mode = .ready
                localRecoveryNotice = "Recovered \(capsule.items.count) encrypted local Capsule item\(capsule.items.count == 1 ? "" : "s") for review. Nothing was submitted automatically."
            } else if recovery.expiredItemCount > 0 {
                localRecoveryNotice = "Expired and deleted \(recovery.expiredItemCount) local Capsule item\(recovery.expiredItemCount == 1 ? "" : "s") before recovery."
            }
            recoverPreparedDraftIfAvailable()
        } catch {
            errorMessage = "Local Capsule recovery failed closed: \(error.localizedDescription)"
        }
    }
