import { act, createElement } from "react";

import { afterEach, beforeEach, describe, expect, it, vi } from "vitest";
import { WorkspaceComposer } from "./workspace-composer";

const { directoryMock } = vi.hoisted(() => ({ directoryMock: vi.fn() }));
vi.mock("./workspace-search", () => ({ useWorkspaceDirectory: directoryMock }));

let root: Root;
let stage: HTMLDivElement;
let anchorTop = 420;
let anchorBottom = 480;
let bannerBottom = 185;
let stageTop = 58;
let optionTop = 6;
let fullRowHeight = 60;
let compactRowHeight = 60;
const observers: Array<{ callback: () => void; targets: Element[] }> = [];
function bounds(top: number, bottom: number, left = 18, right = 302): DOMRect {
  return { top, bottom, left, right, x: left, y: top, width: right - left, height: bottom - top, toJSON: () => ({}) };
}
async function open(withBanner = true, value = "/", inCanvas = false) {
  stage = document.createElement("div");
  stage.id = "workspace-content";
  if (withBanner) {
    const banner = document.createElement("aside");
    banner.role = "status";
    stage.append(banner);
  }
  const mount = document.createElement("div");
  if (inCanvas) {
    const canvas = document.createElement("main");
    canvas.dataset.conversationCanvas = "";
    canvas.style.overflowY = "auto";
    stage.append(canvas); canvas.append(mount);
  } else stage.append(mount);
  document.body.append(stage);
  root = createRoot(mount);
  const props = {
    id: "position-composer", label: "Message", value, maxLength: 1000,
    placeholder: "Message", canSubmit: false, binding: null,
    onValueChange: vi.fn(), onSubmit: vi.fn(), onNavigate: vi.fn(), onFiles: vi.fn(),
    layout: "inline" as const, variant: "home" as const,
  };
  await act(async () => root.render(createElement(WorkspaceComposer, props)));
  const input = stage.querySelector("textarea")!;
  await act(async () => {
    input.focus();
    input.setSelectionRange(1, 1);
    input.dispatchEvent(new KeyboardEvent("keyup", { bubbles: true, key: "/" }));
  });
  return { props, input, popup: document.getElementById(input.getAttribute("aria-controls")!)!.parentElement! };
}
beforeEach(() => {
  anchorTop = 420; anchorBottom = 480; bannerBottom = 185; stageTop = 58;
  observers.length = 0; optionTop = 6; fullRowHeight = 60; compactRowHeight = 60;
  directoryMock.mockReturnValue({ data: null, loading: false, failed: false, retry: vi.fn() });
  vi.stubGlobal("IS_REACT_ACT_ENVIRONMENT", true);
  vi.stubGlobal("ResizeObserver", class {
    instance: { callback: () => void; targets: Element[] };
    constructor(callback: () => void) { this.instance = { callback, targets: [] }; observers.push(this.instance); }
    observe(target: Element) { this.instance.targets.push(target); }
    unobserve(target: Element) { this.instance.targets = this.instance.targets.filter((entry) => entry !== target); }
    disconnect() { this.instance.targets = []; }
  });
  vi.spyOn(HTMLElement.prototype, "getBoundingClientRect").mockImplementation(function (this: HTMLElement) {
    if (this.id === "workspace-content") return bounds(stageTop, 700);
    if (this.role === "status") return bounds(stageTop, bannerBottom);
    if (this.querySelector(":scope > textarea")) return bounds(anchorTop, anchorBottom);
    if (this.role === "option") return bounds(optionTop, optionTop + (this.closest("[data-density]")?.getAttribute("data-density") === "compact" ? compactRowHeight : fullRowHeight));
    if (this.role === "listbox") return bounds(0, 74);
    return bounds(0, 0);
  });
  vi.spyOn(HTMLElement.prototype, "offsetHeight", "get").mockReturnValue(74);
  vi.spyOn(HTMLElement.prototype, "clientHeight", "get").mockReturnValue(60);
  vi.spyOn(Element.prototype, "scrollIntoView").mockImplementation(() => {});
  vi.stubGlobal("innerWidth", 320);
  vi.stubGlobal("innerHeight", 700);
});
afterEach(async () => {
  if (root) await act(async () => root.unmount());
  stage?.remove();
  vi.restoreAllMocks(); vi.unstubAllGlobals();

describe("visible composer suggestions", () => {

    const keyboardChoice = selected();
    const options = stage.querySelectorAll<HTMLElement>('[role="option"]');
    expect(keyboardChoice).toBe(options[1]!.id);
    await act(async () => options[2]!.dispatchEvent(new MouseEvent("mouseover", { bubbles: true, relatedTarget: document.body })));
    expect(selected()).toBe(keyboardChoice);
    await act(async () => options[2]!.dispatchEvent(new PointerEvent("pointermove", { bubbles: true, pointerType: "mouse", clientX: 100, clientY: 200 })));
    expect(selected()).toBe(options[2]!.id);
    await act(async () => options[0]!.dispatchEvent(new PointerEvent("pointermove", { bubbles: true, pointerType: "touch", clientX: 100, clientY: 100 })));

    expect(props.onValueChange).not.toHaveBeenCalled();
    expect(props.onSubmit).not.toHaveBeenCalled();
    expect(props.onNavigate).not.toHaveBeenCalled();
    expect(props.onFiles).not.toHaveBeenCalled();
  });

  it("uses the compact inset budget and retains the inline guard below one complete row", async () => {

    fullRowHeight = 102; compactRowHeight = 60;

    const computed = window.getComputedStyle.bind(window);

    expect(popup.dataset.density).toBe("compact");
    expect(popup.dataset.side).toBe("above");
    expect(popup.style.maxHeight).toBe("69px");

    for (let i = 0; i < 6; i++) await act(async () => observer.callback());
    expect(popup.dataset.side).toBe("above");
    bannerBottom = 185;
    await act(async () => observer.callback());
    expect(popup.dataset.side).toBe("inline");
    bannerBottom = 182;
    await act(async () => observer.callback());
    expect(popup.dataset.side).toBe("above");
    vi.stubGlobal("innerHeight", 700); anchorTop = 420; anchorBottom = 480;
    await act(async () => observer.callback());
    expect(popup.dataset.density).toBe("full");
    expect(input.value).toBe("/");
    expect(props.onValueChange).not.toHaveBeenCalled();
    expect(props.onSubmit).not.toHaveBeenCalled();
    expect(props.onNavigate).not.toHaveBeenCalled();
    expect(props.onFiles).not.toHaveBeenCalled();
  });

  it("enters Slash-only compact density and settles through resize/font/notice changes", async () => {
    vi.stubGlobal("innerHeight", 400); anchorTop = 258; anchorBottom = 320; bannerBottom = 174;
    fullRowHeight = 102;

    expect(popup.dataset.density).toBe("compact");
    expect(popup.dataset.side).toBe("above");
    const observer = observers.find(({ targets }) => targets.includes(input.parentElement!))!;

    expect(popup.dataset.density).toBe("compact");
    expect(popup.dataset.side).toBe("above");
    vi.stubGlobal("innerHeight", 700); anchorTop = 420; anchorBottom = 480;
    await act(async () => window.dispatchEvent(new Event("resize")));
    expect(popup.dataset.density).toBe("full");
    expect(stage.querySelector('[aria-keyshortcuts="F1"]')).toBeNull();
    vi.stubGlobal("innerHeight", 400); anchorTop = 258; anchorBottom = 320;
    await act(async () => window.dispatchEvent(new Event("resize")));
    expect(popup.dataset.density).toBe("compact");
    fullRowHeight = 60;
    await act(async () => observer.callback());

    fullRowHeight = 102;

  it("opens read-only help and restores draft/caret before Slash Escape restores attachment", async () => {
    vi.stubGlobal("innerHeight", 400); anchorTop = 258; anchorBottom = 320; bannerBottom = 174; fullRowHeight = 102;
    const { input, props } = await open();
    const help = stage.querySelector<HTMLButtonElement>('[aria-keyshortcuts="F1"]')!;

// Use a moving anchor so repeated resize checks measure the result of local
// scrolling, rather than adding the same mocked delta on every event.
async function containedComposer() {
  vi.stubGlobal("innerHeight", 392);
  anchorTop = 259; anchorBottom = 321; bannerBottom = 174;
  const opened = await open(true, "/", true);
  const canvas = opened.input.closest<HTMLElement>("[data-conversation-canvas]")!;
  Object.defineProperty(canvas, "clientHeight", { configurable: true, value: 140 });
  Object.defineProperty(canvas, "clientTop", { configurable: true, value: 0 });
  canvas.getBoundingClientRect = () => bounds(174, 314);
  opened.input.parentElement!.getBoundingClientRect = () => bounds(259 - canvas.scrollTop, 321 - canvas.scrollTop);
  return { ...opened, canvas };
}
async function settleFocus(input: HTMLTextAreaElement) {
  await act(async () => {
    input.blur(); input.focus();
    await new Promise<void>((resolve) => requestAnimationFrame(() => requestAnimationFrame(() => resolve())));
  });
}
describe("focused inline composer containment", () => {
  it("reveals the whole composer locally and converges without changing the draft", async () => {
    const { input, canvas, props } = await containedComposer();
    const documentScroll = document.documentElement.scrollTop;
    await settleFocus(input);
    expect(canvas.scrollTop).toBe(9);
    expect(input.parentElement!.getBoundingClientRect().bottom).toBe(312);
    await act(async () => window.dispatchEvent(new Event("resize")));
    expect(canvas.scrollTop).toBe(9);
    expect(document.documentElement.scrollTop).toBe(documentScroll);
    expect(input.value).toBe("/");
    expect(input.selectionStart).toBe(1);
    for (const callback of [props.onValueChange, props.onSubmit, props.onNavigate, props.onFiles]) expect(callback).not.toHaveBeenCalled();
  });
  it("measures responsive parent geometry after the resize event", async () => {
    const { input, canvas } = await containedComposer();
    await settleFocus(input);
    canvas.scrollTop = 0;
    let bottom = 300;
    input.parentElement!.getBoundingClientRect = () => bounds(bottom - 62 - canvas.scrollTop, bottom - canvas.scrollTop);
    await act(async () => {
      window.dispatchEvent(new Event("resize"));
      bottom = 321;
      await new Promise<void>((resolve) => requestAnimationFrame(() => requestAnimationFrame(() => resolve())));
    });
    expect(canvas.scrollTop).toBe(9);
  });
  it("recovers when the actual canvas changes size after viewport processing", async () => {
    const { input, canvas } = await containedComposer();
    await settleFocus(input);
    canvas.scrollTop = 0;
    const observer = observers.find((entry) => entry.targets.includes(canvas));
    expect(observer).toBeDefined();
    await act(async () => {
      observer!.callback();
      await new Promise<void>((resolve) => requestAnimationFrame(() => requestAnimationFrame(() => resolve())));
    });
    expect(canvas.scrollTop).toBe(9);
  });
  it("preserves intentional scrolling and unfocused resizing until focus returns", async () => {
    const { input, canvas } = await containedComposer();
    await settleFocus(input);
    canvas.scrollTop = 4;
    await act(async () => {
      canvas.dispatchEvent(new Event("scroll", { bubbles: true }));
      await new Promise<void>((resolve) => requestAnimationFrame(() => resolve()));
    });
    expect(canvas.scrollTop).toBe(4);
    await act(async () => { input.blur(); window.dispatchEvent(new Event("resize")); });
    expect(canvas.scrollTop).toBe(4);
    await settleFocus(input);
    expect(canvas.scrollTop).toBe(9);
  });
  it("does not scroll a non-scrollable canvas or promise containment when space is insufficient", async () => {
    const { input, canvas } = await containedComposer();
    canvas.style.overflowY = "visible";
    await settleFocus(input);
    expect(canvas.scrollTop).toBe(0);
    canvas.style.overflowY = "auto";
    Object.defineProperty(canvas, "clientHeight", { configurable: true, value: 48 });
    await settleFocus(input);
    expect(canvas.scrollTop).toBe(0);
  });
  it("cancels a pending focus correction when the binding changes", async () => {
    const { input, canvas, props } = await containedComposer();
    const callbacks = new Map<number, FrameRequestCallback>(); let next = 0;
    vi.stubGlobal("requestAnimationFrame", (callback: FrameRequestCallback) => { callbacks.set(++next, callback); return next; });
    vi.stubGlobal("cancelAnimationFrame", (id: number) => { callbacks.delete(id); });
    await act(async () => { input.blur(); input.focus(); });
    await act(async () => root.render(createElement(WorkspaceComposer, { ...props, binding: "another-conversation" })));
    await act(async () => { for (const callback of callbacks.values()) callback(0); });
    expect(canvas.scrollTop).toBe(0);
  });
});
