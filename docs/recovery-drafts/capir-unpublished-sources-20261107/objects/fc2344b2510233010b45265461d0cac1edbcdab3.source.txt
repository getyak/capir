const json=(res,value,status=200)=>{res.writeHead(status,{'content-type':'application/json'});res.end(JSON.stringify(value));};
const list=['models','list','--remote','--json'];

const successful=r=>{assert.equal(r.code,0,r.stdout+r.stderr);return JSON.parse(r.stdout);};

// These use the built production CLI and real owned HTTP, not a fake adapter.
test('native traverses six full pages with long exact Unicode cursors and >1MiB aggregate',async t=>{
  const pages=Array.from({length:6},(_,p)=>Array.from({length:150},(_,i)=>({id:`${p}-${i}-${longID}`,display_name:longLabel})));
  const f=await setup(t,(req,res)=>{
    const page=f.requests.length-1;
    assert.equal(req.method,'GET'); assert.equal(req.headers['x-api-key'],f.env.CAPIR_SYNTHETIC_KEY);
    assert.equal(req.headers['anthropic-version'],'2023-06-01'); assert.equal(req.headers['anthropic-beta'],undefined);
    const url=new URL(req.url,'http://localhost');assert.equal(url.pathname,'/prefix/v1/models');
    assert.equal(url.searchParams.get('after_id'),page===0?null:pages[page-1].at(-1).id);
    json(res,{data:pages[page],has_more:page<5,last_id:pages[page].at(-1).id});
  },'anthropic');
  const r=await cli(list,await withoutDefaultTaskDeadline(f));const doc=successful(r);
  assert.ok(Buffer.byteLength(r.stdout)>1024*1024); assert.deepEqual(doc.models,pages.flat());
  assert.equal(doc.complete,true);assert.equal(doc.pagination_mode,'native_complete');assert.equal(doc.has_more,false);
  assert.equal(doc.cursor,pages.at(-1).at(-1).id);assert.equal(f.requests.length,6);assert.ok(f.requests.every(x=>x.method==='GET'&&x.body===null));
});

for(const mode of ['timeout','SIGINT'])test(`actual models list later ${mode} closes its owned pending HTTP response`,{timeout:15000},async t=>{
  let entered,disconnected;const ready=new Promise(r=>entered=r);const closed=new Promise(r=>disconnected=r);
  const f=await setup(t,(_,res)=>{if(f.requests.length===1)return json(res,{data:[{id:'a'}],has_more:true,last_id:'a'});res.once('close',disconnected);res.writeHead(200,{'content-type':'application/json'});res.flushHeaders();entered();},'anthropic');
  const child=spawn(process.execPath,[binary,...list,...mode==='timeout'?['--timeout','1']:[]],{env:f.env,stdio:['pipe','pipe','pipe']});t.after(()=>{if(child.exitCode===null)child.kill('SIGKILL');});child.stdin.end();let out='',err='';child.stdout.on('data',x=>out+=x);child.stderr.on('data',x=>err+=x);const exit=new Promise(r=>child.once('close',r));await ready;if(mode==='SIGINT')child.kill('SIGINT');assert.equal(await exit,mode==='timeout'?3:130,err+out);await closed;assert.equal(f.requests.length,2);assert.equal(JSON.parse(out).models,undefined);
});
