export const MAIL_PROVIDERS = ["gmail", "outlook"] as const;
export type MailProvider = (typeof MAIL_PROVIDERS)[number];
export type MailItem = {
  id: string;
  subject: string;
  participants: string;
  occurredAt: string | null;
  sourceUrl: string | null;
};
export type MailPreview = {
  provider: MailProvider;
  account: string;
  checkedAt: string;
  expiresAt: string;
  items: MailItem[];
  status: "verified" | "empty";
  scope: "inbox_metadata_sample";
  connectionVersion?: string;
};
export type MailConnectionStatus = {
  configured: boolean;
  connected: boolean;
  expiresAt: string | null;
  connectionVersion?: string | null;
};
export function isMailProvider(value: string): value is MailProvider {
  return MAIL_PROVIDERS.some(provider => provider === value);
}
