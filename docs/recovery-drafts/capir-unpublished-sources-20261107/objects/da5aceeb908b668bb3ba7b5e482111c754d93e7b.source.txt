function context(provider: string, operation: string) { return { params: Promise.resolve({ provider, operation }) }; }
function request(provider: string, operation: string, method: string, cookie = "", query = "", body?: unknown) {
  return new NextRequest(`${origin}/api/integrations/mail/${provider}/${operation}${query}`, {
    method, headers: { host: "localhost:3000", origin, "x-workspace-session": contactHandoffSessionVersion(identity as never),
      "x-talent-signal-workspace": identity.backendAccountId, cookie, ...(body ? { "content-type": "application/json" } : {}) },
    ...(body ? { body: JSON.stringify(body) } : {}),
  });
}
function cookie(response: NextResponse, suffix: string) {
  const found = response.cookies.getAll().find(c => c.name.endsWith(suffix));
  expect(found).toBeDefined();
  return `${found!.name}=${found!.value}`;
}
function respond(value: unknown, status = 200) { return Response.json(value, { status }); }

async function start(provider = "gmail") {
  const response = await POST(request(provider, "start", "POST", "", "", { returnTo: "/workspace/sessions" }), context(provider, "start"));
  expect(response.status).toBe(200);
  const authorization = new URL((await response.json()).authorizationUrl);
  return { attemptCookie: cookie(response, "attempt"), state: authorization.searchParams.get("state")!, authorization, response };
}
async function connect(provider = "gmail") {
  const attempt = await start(provider);
  const response = await GET(request(provider, "callback", "GET", attempt.attemptCookie, `?state=${attempt.state}&code=synthetic-code`), context(provider, "callback"));
  expect(response.status).toBe(303);
  expect(response.headers.get("location")).toContain("result=authorized");
  return { ...attempt, grantCookie: cookie(response, "grant"), callbackResponse: response };
}

it("keeps unavailable providers honest and refuses cross-origin or stale sessions before provider IO", async () => {
  vi.stubEnv("TALENT_SIGNAL_MAIL_TEST_ENABLED", "false");
  const status = await GET(request("gmail", "status", "GET"), context("gmail", "status"));
  expect(await status.json()).toEqual({ configured: false, connected: false, expiresAt: null, connectionVersion: null });
  expect((await POST(request("gmail", "start", "POST", "", "", {}), context("gmail", "start"))).status).toBe(503);
  const hostile = request("gmail", "start", "POST", "", "", {}); hostile.headers.set("origin", "https://hostile.invalid");
  expect((await POST(hostile, context("gmail", "start"))).status).toBe(403);
  const stale = request("gmail", "status", "GET"); stale.headers.set("x-workspace-session", "old");
  expect((await GET(stale, context("gmail", "status"))).status).toBe(409);
  expect(calls.every(c => c.url.endsWith("/v1/auth/session"))).toBe(true);
});

it("runs the actual Gmail start/callback/read path without body access, refresh or token exposure", async () => {
  const connection = await connect();
  expect(connection.authorization.hostname).toBe("accounts.google.com");
  expect(connection.authorization.searchParams.get("scope")).toBe(MAIL_SCOPES.gmail[0]);
  expect(connection.authorization.searchParams.get("code_challenge_method")).toBe("S256");
  expect(connection.authorization.searchParams.get("access_type")).toBe("online");
  expect(connection.callbackResponse.headers.get("location")).not.toContain("synthetic-mail-token");
  const response = await POST(request("gmail", "test", "POST", connection.grantCookie), context("gmail", "test"));
  expect(response.status).toBe(200);
  expect(response.headers.get("cache-control")).toContain("no-store");
  const preview = await response.json();
  expect(preview).toMatchObject({ provider: "gmail", status: "verified", account: "synthetic@example.com", scope: "inbox_metadata_sample" });
  expect(preview.items).toHaveLength(1);
  expect(preview.items[0].sourceUrl).toContain("authuser=synthetic%40example.com");
  expect(JSON.stringify(preview)).not.toContain("synthetic-mail-token");
  const gmailCalls = calls.filter(c => c.url.startsWith("https://gmail.googleapis.com"));
  expect(gmailCalls).toHaveLength(3);
  expect(gmailCalls.find(c => c.url.includes("/messages?"))!.url).not.toMatch(/[?&]q=/);
  expect(gmailCalls.find(c => c.url.includes("/messages/a1?"))!.url).toContain("format=metadata");
  expect(gmailCalls.every(c => c.init?.redirect === "error")).toBe(true);
  expect(connection.callbackResponse.cookies.getAll().find(c => c.name.endsWith("grant"))).toMatchObject({ httpOnly: true, sameSite: "lax", maxAge: expect.any(Number) });
});

it("rejects wrong state, wrong provider, replay, cancelled consent and partial/overbroad scope", async () => {
  const attempt = await start();
  const wrong = await GET(request("gmail", "callback", "GET", attempt.attemptCookie, "?state=wrong&code=synthetic"), context("gmail", "callback"));
  expect(wrong.headers.get("location")).toContain("result=expired");
  const foreign = await GET(request("outlook", "callback", "GET", attempt.attemptCookie, `?state=${attempt.state}&code=synthetic`), context("outlook", "callback"));
  expect(foreign.headers.get("location")).toContain("result=expired");
  const cancel = await GET(request("gmail", "callback", "GET", attempt.attemptCookie, `?state=${attempt.state}&error=access_denied`), context("gmail", "callback"));
  expect(cancel.headers.get("location")).toContain("result=cancelled");
  const replay = await GET(request("gmail", "callback", "GET", attempt.attemptCookie, `?state=${attempt.state}&code=synthetic`), context("gmail", "callback"));
  expect(replay.headers.get("location")).toContain("result=expired");
  expect(calls.filter(c => c.url.includes("/token"))).toHaveLength(0);
  for (const permissions of ["openid", `${MAIL_SCOPES.gmail[0]} https://www.googleapis.com/auth/gmail.send`]) {
    scope = permissions;
    const retry = await start();
    const declined = await GET(request("gmail", "callback", "GET", retry.attemptCookie, `?state=${retry.state}&code=synthetic`), context("gmail", "callback"));
    expect(declined.headers.get("location")).toContain("result=failed");
    expect(declined.cookies.getAll().some(c => c.name.endsWith("grant"))).toBe(false);
  }
});

it("a replacement attempt survives an old in-flight callback response", async () => {
  const first = await start();
  let release!: (response: Response) => void;
  let entered!: () => void;
  const reached = new Promise<void>(resolve => { entered = resolve; });
  override = url => url.includes("/token") ? new Promise<Response>(resolve => { release = resolve; entered(); }) : null;
  const pending = GET(request("gmail", "callback", "GET", first.attemptCookie, `?state=${first.state}&code=first`), context("gmail", "callback"));
  await reached;
  const newer = await start();
  release(respond({ access_token: "synthetic-mail-token", token_type: "Bearer", expires_in: 3600, scope }));
  const oldResponse = await pending;
  expect(oldResponse.headers.get("location")).toContain("result=expired");
  expect(oldResponse.cookies.getAll()).toHaveLength(0);
  override = null;
  const newResponse = await GET(request("gmail", "callback", "GET", newer.attemptCookie, `?state=${newer.state}&code=newer`), context("gmail", "callback"));
  expect(newResponse.headers.get("location")).toContain("result=authorized");
});
it("disconnect during the final live-session read suppresses an old preview", async () => {
  const connection = await connect();
  let release!: (response: Response) => void;
  let entered!: () => void;
  const reached = new Promise<void>(resolve => { entered = resolve; });
  let count = 0;
  override = url => url.endsWith("/v1/auth/session") && ++count === 2 ? new Promise<Response>(resolve => { release = resolve; entered(); }) : null;
  const pending = POST(request("gmail", "test", "POST", connection.grantCookie), context("gmail", "test"));
  await reached;
  const cleared = await POST(request("gmail", "disconnect", "POST", connection.grantCookie), context("gmail", "disconnect"));
  expect(cleared.status).toBe(200);
  release(respond({ account: { id: identity.backendAccountId }, user: { id: identity.backendUserId } }));
  const response = await pending;
  expect(response.status).toBe(409);
  expect((await response.json()).code).toBe("MAIL_GRANT_EXPIRED");
});
