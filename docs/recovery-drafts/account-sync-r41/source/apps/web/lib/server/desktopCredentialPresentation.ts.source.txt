import "server-only";

import {
  authenticatedBackendClient,
  readPrimaryBackendSessionClaims,
} from "@/lib/server/backendAuth";
import { resolveActiveCredentialGrant } from "@/lib/server/desktopAuth";
import {
  flowIsLive,
  readAuthOperation,
  sessionFingerprint,
  type AuthOperation,
} from "@/lib/server/stagedAuth";

/**
 * Non-secret presentation of the one active credential continuation
 * (ADR 0020). The staged Web pages are the single continuation surface: before
 * anything is rendered, the active flow-owned sealed relay grant — or the
 * ordinary Web sealed attempt — is resolved together with the CURRENT
 * account/user, session fingerprint, revisions and absolute deadlines. A
 * stale, expired or foreign flow fails closed and renders nothing.
 *
 * Only `operationRef`, the frozen rendered scope, the intent/provider and the
 * grant's UI status ever leave this module. The credential attempt id, its
 * server-held secret, provider challenge, nonce and verifier stay server-side
 * and never enter DOM, query or page state.
 */

/** Fixed first-party completion handoff intercepted by capable native hosts. */
export const DESKTOP_CHECK_RESULT_PATH = "/desktop-auth/check-result";

export type CredentialContinuationStep = "password" | "target";

export type CredentialContinuationView = {
  /** Frozen operation reference: the only operation identifier in the DOM. */
  operationRef: string;
  accountId: string;
  userId: string;
  accountRevision: number;
  userRevision: number;
  intent: "set_password" | "change_password" | "link_provider";
  targetProvider?: "apple" | "google";
  /**
   * Which server-side grant authorizes the eventual consumer. This is UI
   * status only; the grant secrets themselves never leave the server.
   */
  grant: "relay" | "sealed-attempt";
};

/**
 * Resolve the active continuation for exactly one staged step. The relay grant
 * (ADR 0020) takes precedence; the ordinary Web sealed attempt stays supported.
 */
export async function presentCredentialContinuation(
  step: CredentialContinuationStep,
  now = Date.now(),
): Promise<CredentialContinuationView | null> {
  const relay = await resolveActiveCredentialGrant();
  const operation: AuthOperation | null = relay?.operation ?? (await readAuthOperation());
  if (!operation) return null;
  const grant = relay ? ("relay" as const) : operation.attempt ? ("sealed-attempt" as const) : null;
  if (!grant) return null;
  // Absolute flow deadline: the frozen creation time plus the fixed TTL.
  if (!flowIsLive(operation, now)) return null;
  if (relay) {
    // The relay grant carries its own absolute deadline and immutable intent;
    // resealing or returning can never extend or re-target it.
    if (relay.continuation.flow_ref !== operation.ref) return null;
    if (relay.continuation.intent !== operation.intent) return null;
    const grantDeadline = Date.parse(relay.continuation.expires_at ?? "");
    if (relay.continuation.expires_at !== undefined && (!Number.isFinite(grantDeadline) || grantDeadline <= now)) {
      return null;
    }
  }
  if (step === "password") {
    if (operation.intent !== "set_password" && operation.intent !== "change_password") return null;
  } else {
    // The target continuation is only renderable when its own provider proof
    // round could actually be entered from here.
    if (operation.intent !== "link_provider" || !operation.targetProvider) return null;
    if (relay) {
      const challenge = relay.continuation.provider_challenge;
      if (!challenge?.challenge_id) return null;
      const challengeDeadline = Date.parse(challenge.expires_at);
      if (!Number.isFinite(challengeDeadline) || challengeDeadline <= now) return null;
      if (
        relay.continuation.target_provider &&
        relay.continuation.target_provider !== operation.targetProvider
      ) {
        return null;
      }
    } else if (operation.step !== "awaiting-target" || !operation.attempt) {
      return null;
    }
  }
  // The exact original session, actor and frozen revisions must still be the
  // caller's; expected DOM values are comparators, never authorization.
  const claims = await readPrimaryBackendSessionClaims();
  if (!claims) return null;
  if (sessionFingerprint(claims.backendAccessToken) !== operation.sessionFingerprint) return null;
  if (
    claims.backendAccountId !== operation.accountId ||
    claims.backendUserId !== operation.userId
  ) {
    return null;
  }
  try {
    const client = await authenticatedBackendClient();
    if (!client) return null;
    const settings = await client.accountSettings();
    if (
      settings.workspace.revision !== operation.accountRevision ||
      settings.user.revision !== operation.userRevision
    ) {
      return null;
    }
  } catch {
    return null;
  }
  return {
    operationRef: operation.ref,
    accountId: operation.accountId,
    userId: operation.userId,
    accountRevision: operation.accountRevision,
    userRevision: operation.userRevision,
    intent: operation.intent,
    ...(operation.intent === "link_provider" && operation.targetProvider
      ? { targetProvider: operation.targetProvider }
      : {}),
    grant,
  };
}
