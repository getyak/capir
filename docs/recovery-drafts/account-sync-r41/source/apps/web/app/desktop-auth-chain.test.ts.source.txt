function treeNodes(node: unknown, type: string): Array<{ props?: Record<string, unknown> }> {
  if (!node || typeof node !== "object") return [];
  if (Array.isArray(node)) return node.flatMap((child) => treeNodes(child, type));
  const element = node as { type?: unknown; props?: { children?: unknown } };
  return [
    ...(element.type === type ? [element as { props?: Record<string, unknown> }] : []),
    ...treeNodes(element.props?.children, type),
  ];
}
const attempts = new Map<string, Record<string, unknown>>();
const consumeBodies = new Map<string, Record<string, unknown>>();
let calls: Array<{ path: string; attempt_id?: string; expected_account_id?: string | null }> = [];

function jsonPost(path: string, body: unknown) {
  return new Request(ORIGIN + path, {
    method: "POST",
    headers: { origin: ORIGIN, "content-type": "application/json", "sec-fetch-dest": "document" },
    body: JSON.stringify(body),
  });
}

function formPost(path: string, fields: Record<string, string>) {
  const form = new FormData();
  for (const [key, value] of Object.entries(fields)) form.set(key, value);
  return new Request(ORIGIN + path, {
    method: "POST",
    headers: { origin: ORIGIN, "sec-fetch-dest": "document" },
    body: form,
  });
}

async function wkPrepare(options: {
  provider?: "google" | "apple";
  purpose?: "login" | "credential_round";
  role?: "login" | "current" | "target";
  intent?: string;
  target_provider?: string;
  expected_account_id?: string;
} = {}) {
  transport.active = "wk";
  const response = await preparePOST(
    jsonPost("/api/desktop-auth/prepare", {
      provider: options.provider ?? "google",
      purpose: options.purpose ?? "login",
      role: options.role ?? (options.purpose === "credential_round" ? "current" : "login"),
      code_challenge: "h".repeat(43),
      state: randomUUID().replaceAll("-", ""),
      ...(options.intent ? { intent: options.intent } : {}),
      ...(options.target_provider ? { target_provider: options.target_provider } : {}),
      ...(options.expected_account_id ? { expected_account_id: options.expected_account_id } : {}),
    }),
  );
  expect(response.status, response.headers.get("location") ?? "").toBe(303);
  expect(response.headers.get("location") ?? "", "prepare target").toContain("/desktop-auth/pending");
  const pending = new URL(response.headers.get("location")!, ORIGIN);
  const attempt = attempts.get(pending.searchParams.get("attempt")!)!;
  expect(transport.wk.has(pairingCookieName(attempt.attempt_id as string))).toBe(true);
  return attempt;
}

async function systemRound(attempt: Record<string, unknown>) {
  transport.active = "system";
  const response = await authorizeGET(
    new Request(
      `${ORIGIN}/desktop-auth/authorize?attempt=${attempt.attempt_id}&state=${attempt.state}`,
    ),
  );
  expect(response.status).toBe(303);
  expect(new URL(response.headers.get("location")!).hostname).toBe(
    attempt.provider === "apple" ? "appleid.apple.com" : "accounts.google.com",
  );
  const sealed = await readDesktopBrowserAttempt(attempt.attempt_id as string);
  expect(sealed?.attempt_id).toBe(attempt.attempt_id);
  expect(sealed?.state).toBe(attempt.state);
  transport.applePost = attempt.provider === "apple";
  await expect(
    buildAuthConfig().callbacks!.jwt!({
      token: {},
      user: { id: "synthetic-provider-user" },
      account: {
        provider: attempt.provider,
        id_token: `synthetic-provider-token-${attempt.provider_challenge_id}`,
      },
      profile: {},
    } as never),
  ).rejects.toThrow("desktop_staged");
  transport.applePost = false;
  try {
    await LoginPage({ searchParams: Promise.resolve({ error: "AccessDenied" }) });
    throw new Error("Expected redirect");
  } catch (error) {
    expect((error as { target?: string }).target).toBe(
      `/desktop-auth/confirm?attempt=${attempt.attempt_id}`,
    );
  }
  // The system jar never receives an ordinary session cookie.
  expect(transport.system.has(AUTH_SESSION_COOKIE)).toBe(false);
  return ConfirmPage({ searchParams: Promise.resolve({ attempt: attempt.attempt_id as string }) });
}

it("completes the Google login chain through confirmation and the WK consume", async () => {
  const attempt = await wkPrepare({ provider: "google" });
  const page = await systemRound(attempt);
  const forms = treeNodes(page, "form").map((form) => form.props?.action);
  expect(forms).toContain("/desktop-auth/complete");
  const response = await completePOST(
    formPost("/desktop-auth/complete", { attempt_id: attempt.attempt_id as string }),
  );
  const callback = new URL(response.headers.get("location")!);
  expect(callback.protocol).toBe("com.talentsignal.macos.auth:");
  expect(callback.searchParams.get("attempt")).toBe(attempt.attempt_id);
  transport.active = "wk";
  const result = await consumePOST(
    jsonPost("/api/desktop-auth/consume", {
      attempt_id: attempt.attempt_id,
      code: callback.searchParams.get("code"),
      verifier: "v".repeat(43),
    }),
  );
  expect(result.headers.get("location")).toBe("/workspace");
  expect(transport.wk.has(AUTH_SESSION_COOKIE)).toBe(true);
  expect(transport.wk.has(pairingCookieName(attempt.attempt_id as string))).toBe(false);
});

it("a stale A complete form with current B never mints a code and preserves B", async () => {
  const a = await wkPrepare({ provider: "google" });
  await systemRound(a);
  const b = await wkPrepare({ provider: "apple" });
  await systemRound(b);
  const response = await completePOST(formPost("/desktop-auth/complete", { attempt_id: a.attempt_id as string }));
  expect(response.status).toBe(409);
  const minted = calls.filter((call) => call.path === "/v1/desktop-auth/complete");
  expect(minted.some((call) => call.attempt_id === b.attempt_id)).toBe(false);
  expect(minted.some((call) => call.attempt_id === a.attempt_id)).toBe(false);
  // B's sealed round and cookies survive; its own form still works.
  expect(transport.system.has(browserAttemptCookieName(b.attempt_id as string))).toBe(true);
  const fresh = await completePOST(formPost("/desktop-auth/complete", { attempt_id: b.attempt_id as string }));
  const callback = new URL(fresh.headers.get("location")!);
  expect(callback.searchParams.get("attempt")).toBe(b.attempt_id);
});

it("system cancel cancels the owned attempt and emits the fixed cancelled callback", async () => {
  const attempt = await wkPrepare({ provider: "google" });
  await systemRound(attempt);
  const response = await cancelPOST(
    formPost("/api/desktop-auth/cancel", { attempt_id: attempt.attempt_id as string }),
  );
  const cancelCalls = calls.filter((call) => call.path === "/v1/desktop-auth/cancel");
  expect(cancelCalls).toHaveLength(1);
  const location = new URL(response.headers.get("location")!);
  expect(location.protocol).toBe("com.talentsignal.macos.auth:");
  expect(location.searchParams.get("outcome")).toBe("cancelled");
  expect(location.searchParams.get("attempt")).toBe(attempt.attempt_id);
  expect(location.searchParams.get("code")).toBeNull();
  // The system jar's sealed round is cleared; nothing claims success for a
  // consumed result.
  expect(transport.system.has(browserAttemptCookieName(attempt.attempt_id as string))).toBe(false);
});

it("the real Server Action start reaches prepare and a password round seals both continuations", async () => {
  // Real Server Action -> actual prepare handler: the frozen operation and
  // round are created by `startProviderReauth` itself (reauthChallengeId and
  // the sealed round challenge agree), never by invented recovery state.
  const { startProviderReauth } = await import(
    "@/app/workspace/settings/login-methods/actions"
  );
  const { sessionFingerprint } = await import("@/lib/server/stagedAuth");
  const { AUTH_SESSION_COOKIE: sessionCookie, authSecret } = await import(
    "@/lib/server/backendAuth"
  );
  const accountId = bodyAccountId;
  const userId = bodyUserId;
  const claims = {
    backendAccessToken: "synthetic-actor",
    backendAccountId: accountId,
    backendAccountName: "Actor",
    backendAccountSlug: "actor",
    backendUserId: userId,
    backendExpiresAt: new Date(Date.now() + 3_600_000).toISOString(),
    backendRole: "member" as const,
    backendUsername: null,
  };
  transport.wk.set(sessionCookie, {
    value: await encode({ secret: authSecret(), salt: sessionCookie, maxAge: 3600, token: claims }),
    options: { httpOnly: true, sameSite: "lax", secure: true },
  });
  void sessionFingerprint;
  const originalSessionCookie = transport.wk.get(sessionCookie)?.value;

  const form = new FormData();
  form.set("intent", "set_password");
  form.set("reauthProvider", "google");
  form.set("accountId", accountId);
  form.set("userId", userId);
  form.set("accountRevision", "3");
  form.set("userRevision", "2");
  form.set("operationRef", "");
  form.set("nativeHost", "1");
  form.set("hasPassword", "false");
  let redirected: string | null = null;
  try {
    await startProviderReauth(form);
    throw new Error("Expected native redirect");
  } catch (error) {
    redirected = (error as { target?: string }).target ?? null;
  }
  // The capable host gets the fixed native request link and embedded signIn is
  // never invoked for this flow.
  expect(redirected).toContain("/desktop-auth/request?provider=google&purpose=credential_round&role=current");

  const attempt = await wkPrepare({
    purpose: "credential_round",
    role: "current",
    intent: "set_password",
    expected_account_id: accountId,
  });
  const prepareCall = calls.find((call) => call.path === "/v1/desktop-auth/prepare");
  // The frozen rendered scope travels as an expected comparator.
  expect(prepareCall?.expected_account_id).toBe(accountId);
  const page = await systemRound(attempt);
  expect(JSON.stringify(page)).toContain("密码");
  const response = await completePOST(
    formPost("/desktop-auth/complete", { attempt_id: attempt.attempt_id as string }),
  );
  const callback = new URL(response.headers.get("location")!);
  transport.active = "wk";
  const consume = await consumePOST(
    jsonPost("/api/desktop-auth/consume", {
      attempt_id: attempt.attempt_id,
      code: callback.searchParams.get("code"),
      verifier: "v".repeat(43),
    }),
  );
  expect(consume.headers.get("location")).toContain("/workspace/settings?desktop-auth=password");
  // The server-sealed per-flow continuation resolves the SAME grant for the
  // real password consumer; relay state is never mirrored into a shared
  // mutable credential cookie (a held older response cannot overwrite it).
  expect([...transport.wk.keys()].some((name) => name.includes("desktop-continuation"))).toBe(true);
  const { resolveActiveCredentialGrant } = await import("@/lib/server/desktopAuth");
  const resolved = await resolveActiveCredentialGrant();
  expect(resolved?.continuation.credential_attempt_id).toBeTruthy();
  expect(resolved?.continuation.intent).toBe("set_password");
  expect(transport.wk.has("talent-signal.credential-attempt")).toBe(false);
  // The pairing authority survives the successful consume until the REAL
  // server acknowledgment (a WK navigation is not an acknowledgment).

it("R2: target consume injects the SERVER-held grant secret, never a client value", async () => {
  // Password-first target: the Action seals the operation + grant; the native
  // target round then consumes with only D/code/verifier and the Web injects
  // the sealed grant secret server-to-server.
  const { startPasswordStepUpLink } = await import(
    "@/app/workspace/settings/login-methods/actions"
  );
  const { AUTH_SESSION_COOKIE: sessionCookie, authSecret } = await import(
    "@/lib/server/backendAuth"
  );
  const claims = {
    backendAccessToken: "synthetic-actor",
    backendAccountId: bodyAccountId,
    backendAccountName: "Actor",
    backendAccountSlug: "actor",
    backendUserId: bodyUserId,
    backendExpiresAt: new Date(Date.now() + 3_600_000).toISOString(),
    backendRole: "member" as const,
    backendUsername: null,
  };
  transport.wk.set(sessionCookie, {
    value: await encode({ secret: authSecret(), salt: sessionCookie, maxAge: 3600, token: claims }),
    options: { httpOnly: true, sameSite: "lax", secure: true },
  });
  const form = new FormData();
  form.set("currentPassword", "quiet-current-password-1");
  form.set("targetProvider", "google");
  form.set("accountId", bodyAccountId);
  form.set("userId", bodyUserId);
  form.set("accountRevision", "3");
  form.set("userRevision", "2");
  form.set("nativeHost", "1");
  try {
    await startPasswordStepUpLink({} as never, form);
  } catch {
    /* native redirect */
  }
  // The target prepare resolves the sealed operation + grant (server-held).
  transport.active = "wk";
  const response = await preparePOST(
    jsonPost("/api/desktop-auth/prepare", {
      provider: "google",
      purpose: "credential_round",
      role: "target",
      intent: "link_provider",
      target_provider: "google",
      code_challenge: "h".repeat(43),
      state: "s".repeat(32),
    }),
  );
  expect(response.status, response.headers.get("location") ?? "").toBe(303);
  expect(response.headers.get("location")).toContain("/desktop-auth/pending");
  // The sealed target continuation is installed so the final ACK can match.
  expect([...transport.wk.keys()].some((name) => name.includes("desktop-continuation"))).toBe(true);
  const consumeSecretBefore = calls.filter((call) => call.path === "/v1/desktop-auth/consume");
  expect(consumeSecretBefore).toHaveLength(0);
  // A client-supplied secret is ignored: the Web resolves the sealed one.
  const pending = new URL(response.headers.get("location")!, ORIGIN);
  const attemptId = pending.searchParams.get("attempt")!;
  const fakeSecret = "client-supplied-secret-should-be-ignored-aaaaaa";
  transport.wk.set(pairingCookieName(attemptId), {
    value: "p".repeat(43),
    options: { httpOnly: true, sameSite: "lax", secure: true },
  });
  const ref = "xchg-target-1-1-abcdefgh";
  await consumePOST(
    new Request(`${ORIGIN}/api/desktop-auth/consume?ref=${ref}`, {
      method: "POST",
      headers: {
        origin: ORIGIN,
        "content-type": "application/json",
        "sec-fetch-dest": "document",
        "x-desktop-transport": "capable-v2",
        "x-request-ref": ref,
      },
      body: JSON.stringify({
        attempt_id: attemptId,
        code: "c".repeat(43),
        verifier: "v".repeat(43),
        attempt_secret: fakeSecret,
        role: "target",
        purpose: "credential_round",
        request_ref: ref,
      }),
    }),
  );
  const consumeCall = calls.find((call) => call.path === "/v1/desktop-auth/consume");
  expect(consumeCall).toBeTruthy();
  const body = consumeBodies.get(consumeCall!.attempt_id ?? attemptId);
  // The SERVER-held sealed grant secret travels; the client value never does.
  expect(body?.attempt_secret).toBe("fixture-attempt-secret-abcdefghijklmnopqrstuvwxyz");
  expect(body?.attempt_secret).not.toBe(fakeSecret);
});

it("provider-first target resolves the SAME flow-owned continuation through the real consumers", async () => {
  // Real current Server Action -> current round -> sealed continuation ->
  // real continueTargetLink Action -> target prepare -> target consume, all
  // against the actual route consumers with separate WK jars.
  const { startProviderReauth, continueTargetLink } = await import(
    "@/app/workspace/settings/login-methods/actions"
  );
  const { AUTH_SESSION_COOKIE: sessionCookie, authSecret } = await import(
    "@/lib/server/backendAuth"
  );
  const claims = {
    backendAccessToken: "synthetic-actor",
    backendAccountId: bodyAccountId,
    backendAccountName: "Actor",
    backendAccountSlug: "actor",
    backendUserId: bodyUserId,
    backendExpiresAt: new Date(Date.now() + 3_600_000).toISOString(),
    backendRole: "member" as const,
    backendUsername: null,
  };
  transport.wk.set(sessionCookie, {
    value: await encode({ secret: authSecret(), salt: sessionCookie, maxAge: 3600, token: claims }),
    options: { httpOnly: true, sameSite: "lax", secure: true },
  });

  // The real current-provider start seals the operation and its target round.
  const form = new FormData();
  form.set("intent", "link_provider");
  form.set("reauthProvider", "google");
  form.set("targetProvider", "apple");
  form.set("accountId", bodyAccountId);
  form.set("userId", bodyUserId);
  form.set("accountRevision", "3");
  form.set("userRevision", "2");
  form.set("nativeHost", "1");
  try {
    await startProviderReauth(form);
    throw new Error("Expected native redirect");
  } catch (error) {
    expect((error as { target?: string }).target).toContain("purpose=credential_round");
  }

  // Current round through the actual consumers.
  const current = await wkPrepare({
    purpose: "credential_round",
    role: "current",
    intent: "link_provider",
    expected_account_id: bodyAccountId,
    provider: "google",
  });
  const page = await systemRound(current);
  expect(JSON.stringify(page)).toContain("连接");
  const completeResponse = await completePOST(
    formPost("/desktop-auth/complete", { attempt_id: current.attempt_id as string }),
  );

  transport.active = "wk";
  const consume = await consumePOST(
    jsonPost("/api/desktop-auth/consume", {
      attempt_id: current.attempt_id,
      code: callback.searchParams.get("code"),
      verifier: "v".repeat(43),
      role: "current",
      purpose: "credential_round",
      intent: "link_provider",
    }),

  expect(consume.headers.get("location")).toContain("desktop-auth=target");

  // The pending ACK for the current round.
  const { POST: ackPOST } = await import("@/app/api/desktop-auth/ack/route");
  const acked = await ackPOST(
    jsonPost("/api/desktop-auth/ack", {
      attempt_id: current.attempt_id,
      verifier: "v".repeat(43),
    }),
  );
  expect([200, 303]).toContain(acked.status);

  // The REAL target Action resolves the flow-owned continuation and hands the
  // native target link (no embedded signIn). The form carries the rendered
  // immutable operation reference as its exact scope comparator.
  const { readAuthOperation } = await import("@/lib/server/stagedAuth");
  const sealedOperation = await readAuthOperation();
  const targetForm = new FormData();
  targetForm.set("operationRef", sealedOperation?.ref ?? "");
  targetForm.set("nativeHost", "1");
  targetForm.set("accountId", bodyAccountId);
  targetForm.set("userId", bodyUserId);
  targetForm.set("accountRevision", "3");
  targetForm.set("userRevision", "2");
  let targetRedirect: string | null = null;
  try {
    await continueTargetLink(targetForm);

    targetRedirect = (error as { target?: string }).target ?? null;
  }
  expect(targetRedirect).toContain("/desktop-auth/request?provider=apple&purpose=credential_round&role=target");

  // Target prepare resolves the SAME flow-owned sealed continuation.
  transport.active = "wk";
  const { resolveCredentialFlow: resolveFlow } = await import("@/lib/server/desktopAuth");
  const flowDebug = await resolveFlow("target");
  expect(flowDebug?.expected.attempt_secret, "the flow-owned continuation resolves the sealed secret").toBeTruthy();
  const before = consumeBodies.size;
  const target = await wkPrepare({
    purpose: "credential_round",
    role: "target",
    intent: "link_provider",
    target_provider: "apple",
    provider: "apple",
    expected_account_id: bodyAccountId,
  });
  expect(target.attempt_id).toBeTruthy();
  // Target consume sends the captured semantic context; the Web injects the
  // sealed grant secret (never a client value).
  const ref = "xchg-provider-target-1-abcdefgh";
  const targetResponse = await consumePOST(
    new Request(`${ORIGIN}/api/desktop-auth/consume?ref=${ref}`, {
      method: "POST",
      headers: {
        origin: ORIGIN,
        "content-type": "application/json",
        "sec-fetch-dest": "document",
        "x-desktop-transport": "capable-v2",
        "x-request-ref": ref,
      },
      body: JSON.stringify({
        attempt_id: target.attempt_id,
        code: "c".repeat(43),
        verifier: "v".repeat(43),
        role: "target",
        purpose: "credential_round",
        intent: "link_provider",
        target_provider: "apple",
        request_ref: ref,
      }),
    }),
  );
  expect(targetResponse.status, targetResponse.headers.get("location") ?? "").toBe(200);
  expect(consumeBodies.size).toBeGreaterThan(before);
  const targetBody = consumeBodies.get(target.attempt_id as string)!;
  // The Web->backend body carries only the contract fields; the semantic
  // context (role/intent) is validated by the Web against the sealed flow and
  // echoed in the receipt headers the native controller enforces.
  expect(targetResponse.headers.get("x-desktop-role")).toBe("target");
  expect(targetResponse.headers.get("x-desktop-outcome")).toBe("committed");
  expect(targetResponse.headers.get("x-desktop-intent")).toBe("link_provider");
  // The server-injected secret is the sealed continuation's, never a client one.
  expect(Object.keys(targetBody).sort().join(","), "backend body keys").toContain("attempt_secret");
  expect(targetBody.attempt_secret).toBe("s".repeat(43));
});
